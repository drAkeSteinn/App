    94→
    95→export function LiveAdminView({
    96→  tournament,
    97→  players,
    98→  bracket,
    99→  loadingBracket,
   100→}: {
   101→  tournament: Tournament | null;
   102→  players: Player[];
   103→  bracket: Bracket | null;
   104→  loadingBracket: boolean;
   105→}) {
   106→  const [sel, setSel] = useState<{ r: number; m: number } | null>(null);
   107→  const [pickBank, setPickBank] = useState<{ r: number; m: number; slotIdx: number } | null>(null);
   108→  const [confirm, setConfirm] = useState<null | { kind: "remix" | "finish" | "stop" | "wipe" | "reopen" | "results" | "fill" | "demos" }>(null);
   109→  const [busy, setBusy] = useState(false);
   110→  const [view, setView] = useState<AdminView>("tree");
   111→  const [panelOpen, setPanelOpen] = useState(true);
   112→  const [zoom, setZoom] = useState<number | null>(null); // null = auto-fit
   113→  const [startMins, setStartMins] = useState(1);
   114→  const [forceModal, setForceModal] = useState<{ blockingName: string } | null>(null);
   115→  const { data: allTournaments } = useTournaments();
   116→  const { data: allContests } = useContests();
   117→
   118→  /* banco efectivo: reservas del registro antes de la primera eliminatoria;
   119→     después, los eliminados de la última ronda completada (banco rotativo) */
   120→  const bankInfo = useMemo(() => computeBank(bracket, players), [bracket, players]);
   121→  const arenaState = useArenaState(tournament?.id ?? null);
   122→  const schedOpen = arenaState?.schedOpen ?? false;
   123→  /** Emite la señal de sonido al visor. NO reproduce localmente en el admin
   124→      para evitar duplicación — el visor es el único que debe sonar. */
   125→  const fireSound = (event: SoundEvent) => {
   126→    if (tournament) emitSound(tournament.id, event).catch(() => {});
   127→  };
   128→  /* Torneo activo para transmisión (visor + OBS). Solo uno a la vez. */
   129→  const active = useActiveTournament();
   130→  const isActive = !!tournament && active.tid === tournament.id;
   131→  const anotherActive = !!active.tid && active.tid !== tournament?.id;
   132→  const activeTournamentName = active.tournament?.name ?? null;
   133→  const participants = useMemo(
   134→    () => (tournament ? groupParticipants(players, tournament.modality) : []),
   135→    [players, tournament]
   136→  );
   137→  const officialsCount = useMemo(() => players.filter((p) => p.seat === "off").length, [players]);
   138→  const demoCount = useMemo(() => players.filter((p) => p.demo).length, [players]);
   139→  const missingPlayers = tournament ? playersCapacity(tournament) - officialsCount : 0;
   140→  const schedule = useMemo(
   141→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map()),
   142→    [bracket, tournament]
   143→  );
   144→  /* capacidad del escenario: matches simultáneos según las 4 PCs */
   145→  const maxSim = tournament ? maxSimultaneous(tournament.modality) : 1;
   146→  const liveCount = useMemo(
   147→    () => (bracket ? bracket.rounds.reduce((acc, round) => acc + round.filter((x) => x.status === "live").length, 0) : 0),
   148→    [bracket]
   149→  );
   150→
   151→  /* navegación secuencial de matches (para editar rápido sin recorrer todo) */
   152→  const flatMatches = useMemo(() => {
   153→    if (!bracket) return [];
   154→    const out: { r: number; m: number }[] = [];
   155→    bracket.rounds.forEach((round, r) => round.forEach((_, m) => out.push({ r, m })));
   156→    if (bracket.thirdPlace) out.push({ r: -1, m: 0 });
   157→    return out;
   158→  }, [bracket]);
   159→  const flatIdx = useMemo(() => {
   160→    if (!sel || !bracket) return -1;
   161→    return flatMatches.findIndex((x) => x.r === sel.r && x.m === sel.m);
   162→  }, [flatMatches, sel, bracket]);
   163→
   164→  if (!tournament) {
   165→    return (
   166→      <div className="panel clip-card">
   167→        <EmptyState
   168→          icon={<Radio size={26} />}
   169→          title="Selecciona un torneo"
   170→          message="Elige un torneo en el selector superior para administrar el bracket en vivo: ganadores, descalificaciones y reemplazos del banco."
   171→        />
   172→      </div>
   173→    );
   174→  }
   175→
   176→  const guard = async (fn: () => Promise<void>) => {
   177→    setBusy(true);
   178→    try {
   179→      await fn();
   180→    } catch (e) {
   181→      toast.error(e instanceof Error ? e.message : "Error de operación");
   182→    } finally {
   183→      setBusy(false);
   184→    }
   185→  };
   186→
   187→  const selMatch = sel && bracket ? (sel.r === -1 ? bracket.thirdPlace : bracket.rounds[sel.r]?.[sel.m]) ?? null : null;
   188→
   189→  const stepMatch = (dir: 1 | -1) => {
   190→    const nx = flatMatches[flatIdx + dir];
   191→    if (nx) setSel(nx);
   192→  };
   193→
   194→  const zoomStep = (dir: 1 | -1) => {
   195→    setZoom((z) => {
   196→      const base = z ?? 0.8;
   197→      return Math.min(1.8, Math.max(0.3, Math.round((base + dir * 0.1) * 100) / 100));
   198→    });
   199→  };
   200→
   201→  return (
   202→    <div className="flex flex-col gap-3 min-h-0">
   203→      {/* ============ barra de info del torneo ============ */}
   204→      <div className="panel clip-card p-4 flex flex-wrap items-center gap-3">
   205→        <div className="flex items-center gap-3 min-w-0 mr-auto">
   206→          <span className="slashes w-8 h-4 inline-block shrink-0" aria-hidden />
   207→          <div className="min-w-0">
   208→            <div className="font-display italic text-[17px] uppercase truncate">{tournament.name}</div>
   209→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   210→              <StatusChip status={tournament.status} />
   211→              <ModalityChip modality={tournament.modality} />
   212→              {bracket ? (
   213→                <span
   214→                  className={`chip clip-tag border ${
   215→                    liveCount >= maxSim
   216→                      ? "bg-[#e8102e]/15 text-[#ff8095] border-[#e8102e]/40"
   217→                      : "bg-[#e8102e]/12 text-[#ff8095] border-[#e8102e]/35"
   218→                  }`}
   219→                  title="Matches en juego / máximo simultáneo según las PCs del escenario (4 PCs)"
   220→                >
   221→                  <Monitor size={10} />
   222→                  EN JUEGO {liveCount}/{maxSim}
   223→                </span>
   224→              ) : null}
   225→              {bracket ? (
   226→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   227→                  {doneMatches(bracket)}/{totalMatches(bracket)} MATCHES
   228→                </span>
   229→              ) : (
   230→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   231→                  {officialsCount}/{playersCapacity(tournament)} JUGADORES
   232→                </span>
   233→              )}
   234→              {tournament.startAt ? (
   235→                <span className="chip clip-tag bg-[#ffb830]/12 text-[#ffb830] border border-[#ffb830]/35">
   236→                  <TimerReset size={10} />
   237→                  INICIA ≈ {fmtTime(tournament.startAt)}
   238→                </span>
   239→              ) : null}
   240→              {isActive ? (
   241→                <span className="chip clip-tag bg-[#e8102e] text-white text-[9px] blink">
   242→                  <span className="w-1.5 h-1.5 bg-white rounded-full" aria-hidden />
   243→                  EN TRANSMISIÓN
   244→                </span>
   245→              ) : null}
   246→            </div>
   247→          </div>
   248→        </div>
   249→        {/* control de transmisión: Abrir / Cerrar torneo para el visor + OBS */}
   250→        <div className="flex items-center gap-2 shrink-0">
   251→          {isActive ? (
   252→            <Btn
   253→              small
   254→              variant="dark"
   255→              onClick={() =>
   256→                guard(async () => {
   257→                  await closeTournament();
   258→                  toast.success("Torneo cerrado — el visor y las cards de OBS dejan de transmitir");
   259→                })
   260→              }
   261→              title="Cerrar torneo: el visor y las cards de OBS dejan de mostrar este torneo"
   262→            >
   263→              <Lock size={12} />
   264→              Cerrar torneo
   265→            </Btn>
   266→          ) : (
   267→            <Btn
   268→              small
   269→              variant={anotherActive ? "dark" : "red"}
   270→              title={
   271→                anotherActive
   272→                  ? `Ya hay otra transmisión activa: ${activeTournamentName}. Clic para forzar.`
   273→                  : "En vivo: el visor y las cards de OBS mostrarán este torneo"
   274→              }
   275→              onClick={() =>
   276→                guard(async () => {
   277→                  if (!tournament) return;
   278→                  /* verificar si hay otra transmisión activa */
   279→                  const info = await getActiveInfo();
   280→                  const hasOther = (info.tid && info.tid !== tournament.id) || info.contestId;
   281→                  if (hasOther) {
   282→                    /* determinar el nombre del bloqueante */
   283→                    let blockingName = "otra transmisión";
   284→                    if (info.tid) {
   285→                      const t = allTournaments.find((x) => x.id === info.tid);
   286→                      blockingName = t?.name ?? "un torneo";
   287→                    } else if (info.contestId) {
   288→                      const c = allContests.find((x) => x.id === info.contestId);
   289→                      blockingName = c?.name ?? "un concurso";
   290→                    }
   291→                    setForceModal({ blockingName });
   292→                    return;
   293→                  }
   294→                  await openTournament(tournament.id, tournament.name);
   295→                  toast.success("Torneo en vivo — el visor y las cards de OBS lo muestran");
   296→                })
   297→              }
   298→            >
   299→              <Radio size={12} />
   300→              {anotherActive ? `Ocupado: ${activeTournamentName}` : "En vivo"}
   301→            </Btn>
   302→          )}
   303→        </div>
   304→      </div>
   305→
   306→      {/* ============ FLUJO POR FASES ============ */}
   307→      <PhaseBar
   308→        tournament={tournament}
   309→        bracket={bracket}
   310→        players={players}
   311→        participantsCount={participants.length}
   312→        officialsCount={officialsCount}
   313→        missingPlayers={missingPlayers}
   314→        demoCount={demoCount}
   315→        bankCount={bankInfo.entries.length}
   316→        startMins={startMins}
   317→        setStartMins={setStartMins}
   318→        busy={busy}
   319→        guard={guard}
   320→        onConfirm={(kind) => setConfirm({ kind })}
   321→        onFireSound={fireSound}
   322→        onLaunchMix={() =>
   323→          guard(async () => {
   324→            await launchMix(tournament.id, players, tournament);
   325→            setSel(null);
   326→            fireSound("mixLaunch");
   327→            toast.success("Mix match lanzado — el visor reproduce la animación");
   328→          })
   329→        }
   330→      />
   331→
   332→      {/* ============ cuerpo ============ */}
   333→      {loadingBracket ? (
   334→        <div className="panel clip-card p-14 flex items-center justify-center text-[#8e919c]">
   335→          <Loader2 size={20} className="animate-spin" />
   336→        </div>
   337→      ) : !bracket ? (
   338→        <div className="panel clip-card">
   339→          <EmptyState
   340→            icon={<Zap size={26} />}
   341→            title="Brackets no creados"
   342→            message={`Cuando estés listo, lanza el MIX MATCH: se randomizarán los ${participants.length} participantes y el visor de espectadores mostrará la animación de llenado de brackets.`}
   343→            action={
   344→              <div className="flex flex-wrap gap-2 justify-center">
   345→                <Btn
   346→                  variant="red"
   347→                  disabled={participants.length < 2}
   348→                  onClick={() =>
   349→                    guard(async () => {
   350→                      await launchMix(tournament.id, players, tournament);
   351→                      fireSound("mixLaunch");
   352→                      toast.success("Mix match lanzado");
   353→                    })
   354→                  }
   355→                >
   356→                  <Rocket size={13} />
   357→                  Lanzar mix match
   358→                </Btn>
   359→                <Btn
   360→                  variant="dark"
   361→                  onClick={() =>
   362→                    guard(async () => {
   363→                      await setStartAt(tournament.id, Date.now() + startMins * 60_000);
   364→                      toast.success(`Inicio configurado en ${startMins} min — el visor cambia a brackets con el cronómetro`);
   365→                    })
   366→                  }
   367→                >
   368→                  <TimerReset size={13} />
   369→                  Inicio en {startMins} min
   370→                </Btn>
   371→              </div>
   372→            }
   373→          />
   374→          {/* vista previa de participantes */}
   375→          {participants.length > 0 ? (
   376→            <div className="px-5 pb-5">
   377→              <div className="label-cap mb-2 flex items-center gap-2">
   378→                <Users size={11} />
   379→                Participantes que entrarán al sorteo ({participants.length})
   380→              </div>
   381→              <div className="flex flex-wrap gap-1.5 max-h-36 overflow-y-auto">
   382→                {participants.map((p, i) => (
   383→                  <span key={p.pid} className="chip clip-tag bg-white/[0.05] text-[#c9cbd3] border border-white/10">
   384→                    {i + 1}. {p.label}
   385→                  </span>
   386→                ))}
   387→              </div>
   388→            </div>
   389→          ) : null}
   390→        </div>
   391→      ) : (
   392→        <>
   393→          {/* ============ toolbar de vistas ============ */}
   394→          <div className="panel clip-card-sm px-3 py-2 flex flex-wrap items-center gap-2">
   395→            <div className="flex gap-1 p-0.5 bg-black/40 border border-white/10" role="radiogroup" aria-label="Vista del bracket">
   396→              <button
   397→                type="button"
   398→                role="radio"
   399→                aria-checked={view === "tree"}
   400→                onClick={() => setView("tree")}
   401→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   402→                  view === "tree" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   403→                }`}
   404→              >
   405→                <LayoutPanelLeft size={12} />
   406→                Bracket
   407→              </button>
   408→              <button
   409→                type="button"
   410→                role="radio"
   411→                aria-checked={view === "rounds"}
   412→                onClick={() => setView("rounds")}
   413→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   414→                  view === "rounds" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   415→                }`}
   416→              >
   417→                <Columns3 size={12} />
   418→                Rondas
   419→              </button>
   420→            </div>
   421→
   422→            {selMatch ? (
   423→              <span className="chip clip-tag bg-[#e8102e]/15 text-[#ff8095] border border-[#e8102e]/40 text-[9px] flex items-center gap-1.5">
   424→                <Hand size={10} />
   425→                {matchTag(bracket, sel!.r, sel!.m)} seleccionado
   426→              </span>
   427→            ) : (
   428→              <span className="text-[9px] font-extrabold uppercase tracking-[0.16em] text-[#6b6e78] hidden sm:inline">
   429→                Clic en un match para administrarlo
   430→              </span>
   431→            )}
   432→
   433→            <div className="ml-auto flex items-center gap-1.5">
   434→              {/* navegación prev/next */}
   435→              <div className="flex items-center gap-1">
   436→                <IconBtn title="Match anterior" disabled={flatIdx <= 0} onClick={() => stepMatch(-1)}>
   437→                  <ChevronLeft size={14} />
   438→                </IconBtn>
   439→                <IconBtn title="Match siguiente" disabled={flatIdx < 0 || flatIdx >= flatMatches.length - 1} onClick={() => stepMatch(1)}>
   440→                  <ChevronRight size={14} />
   441→                </IconBtn>
   442→              </div>
   443→
   444→              {/* zoom (solo vista bracket) */}
   445→              {view === "tree" ? (
   446→                <div className="flex items-center gap-1 pl-2 border-l border-white/10">
   447→                  <IconBtn title="Alejar" onClick={() => zoomStep(-1)}>
   448→                    <ZoomOut size={14} />
   449→                  </IconBtn>
   450→                  <span className="text-[9px] font-extrabold tabular-nums text-[#8e919c] w-10 text-center uppercase">
   451→                    {zoom ? `${Math.round(zoom * 100)}%` : "AUTO"}
   452→                  </span>
   453→                  <IconBtn title="Acercar" onClick={() => zoomStep(1)}>
   454→                    <ZoomIn size={14} />
   455→                  </IconBtn>
   456→                  <IconBtn title="Ajustar a la pantalla" active={zoom === null} onClick={() => setZoom(null)}>
   457→                    <Maximize2 size={13} />
   458→                  </IconBtn>
   459→                </div>
   460→              ) : null}
   461→
   462→              {/* toggle cartelera en el visor (proyectado) */}
   463→              <button
   464→                type="button"
   465→                onClick={() => {
   466→                  if (!tournament) return;
   467→                  setSchedOpen(tournament.id, !schedOpen).catch(() => {});
   468→                }}
   469→                title={schedOpen ? "Ocultar cartelera en el visor" : "Mostrar cartelera de horarios en el visor"}
   470→                aria-pressed={schedOpen}
   471→                className={`btn-press clip-tag px-2.5 py-2 text-[10px] font-extrabold uppercase tracking-[0.1em] flex items-center gap-1.5 border transition-colors ${
   472→                  schedOpen
   473→                    ? "border-[#ffb830]/60 bg-[#ffb830]/12 text-[#ffb830]"
   474→                    : "border-white/12 text-[#8e919c] hover:text-white hover:border-[#ffb830]/50"
   475→                }`}
   476→              >
   477→                <CalendarClock size={13} />
   478→                <span className="hidden md:inline">Cartelera</span>
   479→              </button>
   480→
   481→              {/* editar posición de cards de OBS (abre el editor draggable) */}
   482→              <button
   483→                type="button"
   484→                onClick={() => window.open("/?obs=1&edit=1", "_blank", "noopener")}
   485→                title="Arrastrar las cards de OBS (PC1-PC4) a posiciones personalizadas — se guarda para todos los torneos"
   486→                className="btn-press clip-tag px-2.5 py-2 text-[10px] font-extrabold uppercase tracking-[0.1em] flex items-center gap-1.5 border border-white/12 text-[#8e919c] hover:text-white hover:border-[#e8102e]/60 transition-colors"
   487→              >
   488→                <MonitorPlay size={13} />
   489→                <span className="hidden md:inline">Editar cards OBS</span>
   490→              </button>
   491→
   492→              {/* toggle panel lateral */}
   493→              <button
   494→                type="button"
   495→                onClick={() => setPanelOpen((v) => !v)}
   496→                title={panelOpen ? "Ocultar panel del match" : "Mostrar panel del match"}
   497→                aria-pressed={panelOpen}
   498→                className={`btn-press p-2 border transition-colors hidden xl:inline-flex ${
   499→                  panelOpen
   500→                    ? "border-[#e8102e]/60 bg-[#e8102e]/10 text-white"
   501→                    : "border-white/12 text-[#8e919c] hover:text-white"
   502→                }`}
   503→              >
   504→                {panelOpen ? <PanelRightClose size={14} /> : <PanelRightOpen size={14} />}
   505→              </button>
   506→            </div>
   507→          </div>
   508→
   509→          {/* ============ stage + panel lateral colapsable ============ */}
   510→          <div className="flex gap-3 items-stretch min-h-0">

   864→/* ============================================================
   865→   PC DOCK — monitor del escenario: qué match y quién ocupa
   866→   cada una de las 4 PCs (● EN JUEGO / PLANIFICADA / LIBRE).
   867→   Clic en una PC ocupada abre ese match en el panel.
   868→   ============================================================ */
   869→function PcDock({
   870→  tournament,
   871→  bracket,
   872→  onSelectMatch,
   873→}: {
   874→  tournament: Tournament;
   875→  bracket: Bracket;
   876→  onSelectMatch: (r: number, m: number) => void;
   877→}) {
   878→  const cells = useMemo(() => pcOccupancy(bracket, tournament.modality), [bracket, tournament.modality]);
   879→  const maxSim = maxSimultaneous(tournament.modality);
   880→  return (
   881→    <div
   882→      className="shrink-0 mb-2 mx-1 clip-tag border border-white/10 bg-black/40 px-2.5 py-1.5 flex items-stretch gap-1.5 overflow-x-auto"
   883→      role="group"
   884→      aria-label="Asignación de PCs del escenario"
   885→    >
   886→      <div className="flex flex-col justify-center gap-0.5 shrink-0 pr-2.5 border-r border-white/10 mr-1">
   887→        <span className="label-cap flex items-center gap-1.5 whitespace-nowrap">
   888→          <Monitor size={12} className="text-[#e8102e]" />
   889→          PCs del escenario
   890→        </span>
   891→        <span className="text-[8px] font-bold text-[#6b6e78] uppercase tracking-[0.1em] whitespace-nowrap">
   892→          Máx {maxSim} simultáneo{maxSim > 1 ? "s" : ""}
   893→        </span>
   894→      </div>
   895→      {cells.map((c, i) => {
   896→        const key = `PC${i + 1}`;
   897→        const live = c?.status === "live";
   898→        return (
   899→          <button
   900→            key={key}
   901→            type="button"
   902→            disabled={!c}
   903→            onClick={() => c && onSelectMatch(c.r, c.m)}
   904→            title={c ? `${c.tag} · ${key}${c.label ? ` · ${c.label}` : ""} — clic para administrar el match` : `${key} libre`}
   905→            className={`btn-press min-w-[138px] flex-1 text-left px-2 py-1 border transition-colors ${
   906→              live
   907→                ? "border-[#e8102e]/60 bg-[#e8102e]/[0.08]"
   908→                : c
   909→                  ? "border-white/12 bg-white/[0.03] hover:border-[#ffb830]/50"
   910→                  : "border-dashed border-white/10 bg-transparent cursor-default"
   911→            }`}
   912→          >
   913→            <div className="flex items-center justify-between gap-2">
   914→              <span
   915→                className={`flex items-center gap-1 text-[9.5px] font-extrabold tracking-[0.14em] ${
   916→                  live ? "text-[#ff2440]" : c ? "text-[#c9cbd3]" : "text-[#6b6e78]"
   917→                }`}
   918→              >
   919→                <Monitor size={9} />
   920→                {key}
   921→              </span>
   922→              {c ? (
   923→                <span className={`text-[8.5px] font-extrabold uppercase tracking-[0.08em] truncate ${live ? "text-[#ff8095]" : "text-[#8e919c]"}`}>
   924→                  {c.tag}
   925→                </span>
   926→              ) : (
   927→                <span className="text-[8px] font-extrabold uppercase tracking-[0.2em] text-white/25">libre</span>
   928→              )}
   929→            </div>
   930→            {c ? (
   931→              <>
   932→                <div className={`text-[10.5px] font-extrabold uppercase truncate leading-tight mt-0.5 ${live ? "text-white" : "text-[#a9adb8]"}`}>
   933→                  {c.label || "—"}
   934→                </div>
   935→                <div className={`text-[7.5px] font-extrabold uppercase tracking-[0.14em] mt-0.5 ${live ? "text-[#ff2440] blink" : "text-[#6b6e78]"}`}>
   936→                  {live ? "● EN JUEGO" : "PLANIFICADA"}
   937→                </div>
   938→              </>
   939→            ) : (
   940→              <div className="text-[9px] font-bold text-white/15 mt-1">—</div>
   941→            )}
   942→          </button>
   943→        );
   944→      })}
   945→    </div>
   946→  );
   947→}

   949→/* ============================================================
   950→   Panel lateral de edición del match seleccionado
   951→   ============================================================ */
   952→function MatchPanel({
   953→  tournament,
   954→  bracket,
   955→  sel,
   956→  setSel,
   957→  selMatch,
   958→  schedule,
   959→  bankInfo,
   960→  onClose,
   961→  guard,
   962→  setPickBank,
   963→  busy,
   964→  stepMatch,
   965→  onSound,
   966→  onShowPodium,
   967→}: {
   968→  tournament: Tournament;
   969→  bracket: Bracket;
   970→  sel: { r: number; m: number } | null;
   971→  setSel: (v: { r: number; m: number } | null) => void;
   972→  selMatch: Bracket["rounds"][number][number] | null;
   973→  schedule: Map<string, number>;
   974→  bankInfo: BankInfo;
   975→  onClose: () => void;
   976→  guard: (fn: () => Promise<void>) => Promise<void>;
   977→  setPickBank: (v: { r: number; m: number; slotIdx: number } | null) => void;
   978→  busy: boolean;
   979→  stepMatch: (dir: 1 | -1) => void;
   980→  onSound: (event: SoundEvent) => void;
   981→  onShowPodium: () => void;
   982→}) {
   983→  /* capacidad del escenario (4 PCs) + PCs del match seleccionado */
   984→  const maxSim = maxSimultaneous(tournament.modality);
   985→  const liveCount = useMemo(
   986→    () => bracket.rounds.reduce((acc, round) => acc + round.filter((x) => x.status === "live").length, 0)
   987→      + (bracket.thirdPlace?.status === "live" ? 1 : 0),
   988→    [bracket]
   989→  );
   990→  const planned = useMemo(() => plannedPcs(bracket, tournament.modality), [bracket, tournament.modality]);
   991→  const selPcsStr = selMatch ? pcsLabel(selMatch.pcs?.length ? selMatch.pcs : planned.get(selMatch.id)) : "";
   992→  const blockedByPcs = !!selMatch && selMatch.status !== "live" && liveCount >= maxSim;
   993→
   994→  return (
   995→    <div className="panel clip-card p-4 h-full flex flex-col gap-3 overflow-y-auto">
   996→      {!selMatch || !sel ? (
   997→        <div className="py-10 text-center">
   998→          <div className="w-12 h-12 mx-auto mb-3 border border-white/12 bg-white/[0.03] clip-card-sm flex items-center justify-center text-[#e8102e]">
   999→            <Hand size={20} />
  1000→          </div>
  1001→          <p className="text-[11px] font-bold text-[#8e919c] uppercase tracking-wider">Selecciona un match</p>
  1002→          <p className="text-[10px] text-[#6b6e78] font-semibold mt-1">
  1003→            Podrás marcar ganador, descalificar, reemplazar desde el banco y llevar el marcador
  1004→          </p>
  1005→        </div>
  1006→      ) : (

  1332→function MiniCountdown({ target }: { target: number }) {
  1333→  const [now, setNow] = useState(() => Date.now());
  1334→  useEffect(() => {
  1335→    const t = setInterval(() => setNow(Date.now()), 500);
  1336→    return () => clearInterval(t);
  1337→  }, []);
  1338→  const total = Math.max(0, Math.floor((target - now) / 1000));
  1339→  const h = Math.floor(total / 3600);
  1340→  const m = Math.floor((total % 3600) / 60);
  1341→  const s = total % 60;
  1342→  const pad = (x: number) => String(x).padStart(2, "0");
  1343→  return (
  1344→    <span className="font-display italic text-[20px] leading-none text-silver-grad tabular-nums">
  1345→      {h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`}
  1346→    </span>
  1347→  );
  1348→}
