   203→/** Visualizador de audio que reacciona al sonido REAL usando Web Audio API.
   204→    Cada barra mide la amplitud de una banda de frecuencia del audio en vivo. */
   205→function LiveWaveform({ audioEl }: { audioEl: HTMLAudioElement | null }) {
   206→  const canvasRef = useRef<HTMLCanvasElement>(null);
   207→  const rafRef = useRef<number>(0);
   208→
   209→  useEffect(() => {
   210→    if (!audioEl) return;
   211→    const AC = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
   212→    if (!AC) return;
   213→    let ctx: AudioContext | null = null;
   214→    let analyser: AnalyserNode | null = null;
   215→    let source: MediaElementAudioSourceNode | null = null;
   216→    try {
   217→      ctx = new AC();
   218→      analyser = ctx.createAnalyser();
   219→      analyser.fftSize = 64;
   220→      analyser.smoothingTimeConstant = 0.75;
   221→      source = ctx.createMediaElementSource(audioEl);
   222→      source.connect(analyser);
   223→      analyser.connect(ctx.destination);
   224→    } catch {
   225→      return;
   226→    }
   227→    const data = new Uint8Array(analyser.frequencyBinCount);
   228→    const canvas = canvasRef.current;
   229→    if (!canvas) return;
   230→    const c2d = canvas.getContext("2d");
   231→    if (!c2d) return;
   232→
   233→    const BARS = 24;
   234→    const draw = () => {
   235→      if (!analyser || !c2d) return;
   236→      analyser.getByteFrequencyData(data);
   237→      const w = canvas.width;
   238→      const h = canvas.height;
   239→      c2d.clearRect(0, 0, w, h);
   240→      const barW = w / BARS;
   241→      for (let i = 0; i < BARS; i++) {
   242→        // mapear los bins del analyser a las barras
   243→        const idx = Math.floor((i / BARS) * data.length);
   244→        const val = data[idx] / 255; // 0-1
   245→        const barH = Math.max(3, val * h * 0.95);
   246→        const x = i * barW;
   247→        const y = (h - barH) / 2;
   248→        // gradiente rojo→dorado
   249→        const grad = c2d.createLinearGradient(0, y, 0, y + barH);
   250→        grad.addColorStop(0, "#ff2440");
   251→        grad.addColorStop(1, "#ffb830");
   252→        c2d.fillStyle = grad;
   253→        c2d.fillRect(x + 1, y, barW - 2, barH);
   254→      }
   255→      rafRef.current = requestAnimationFrame(draw);
   256→    };
   257→    draw();
   258→    return () => {
   259→      cancelAnimationFrame(rafRef.current);
   260→      try {
   261→        source?.disconnect();
   262→        analyser?.disconnect();
   263→        ctx?.close();
   264→      } catch {
   265→        /* ignore */
   266→      }
   267→    };
   268→  }, [audioEl]);
   269→
   270→  return (
   271→    <canvas
   272→      ref={canvasRef}
   273→      width={400}
   274→      height={64}
   275→      className="w-full max-w-[400px] h-12 sm:h-16"
   276→      style={{ imageRendering: "auto" }}
   277→    />
   278→  );
   279→}

   514→function LeaderCard({
   515→  p,
   516→  rank,
   517→  isWinner,
   518→}: {
   519→  p: Participant;
   520→  rank: number;
   521→  isWinner: boolean;
   522→}) {
   523→  const prevScore = useRef(p.score);
   524→  const [flash, setFlash] = useState(false);
   525→  useEffect(() => {
   526→    if (p.score > prevScore.current) {
   527→      prevScore.current = p.score;
   528→      const r = requestAnimationFrame(() => setFlash(true));
   529→      const t = setTimeout(() => setFlash(false), 1200);
   530→      return () => {
   531→        cancelAnimationFrame(r);
   532→        clearTimeout(t);
   533→      };
   534→    }
   535→    prevScore.current = p.score;
   536→  }, [p.score]);
   537→
   538→  return (
   539→    <motion.div
   540→      initial={{ opacity: 0, y: 8 }}
   541→      animate={{ opacity: 1, y: 0 }}
   542→      className={`clip-card-sm px-4 py-3 flex items-center gap-3 transition-all relative overflow-hidden ${
   543→        flash
   544→          ? "bg-[linear-gradient(160deg,rgba(255,184,48,0.3),rgba(255,184,48,0.08))] border border-[#ffb830]"
   545→          : isWinner
   546→            ? "bg-[linear-gradient(160deg,rgba(255,184,48,0.2),rgba(255,184,48,0.05))] border border-[#ffb830]/60"
   547→            : rank === 0
   548→              ? "bg-white/[0.08] border border-white/12"
   549→              : "bg-black/40 border border-white/8"
   550→      }`}
   551→    >
   552→      {flash ? (
   553→        <>
   554→          <motion.div
   555→            initial={{ opacity: 0.8, scale: 1 }}
   556→            animate={{ opacity: 0, scale: 1.3 }}
   557→            transition={{ duration: 1.2, ease: "easeOut" }}
   558→            className="absolute inset-0 bg-[radial-gradient(60%_60%_at_50%_50%,rgba(255,184,48,0.35),transparent_70%)] pointer-events-none"
   559→          />
   560→          <motion.div
   561→            initial={{ opacity: 0, y: -8 }}
   562→            animate={{ opacity: [0, 1, 0], y: -20 }}
   563→            transition={{ duration: 1, ease: "easeOut" }}
   564→            className="absolute right-3 top-0 font-display italic text-[#ffb830] text-[14px] font-extrabold pointer-events-none"
   565→          >
   566→            +1
   567→          </motion.div>
   568→        </>
   569→      ) : null}
   570→      <span className={`w-9 h-9 shrink-0 clip-badge flex items-center justify-center font-display italic text-[16px] ${rank === 0 ? "bg-[#ffb830] text-[#241500]" : "bg-black/40 border border-white/10 text-[#8e919c]"}`}>
   571→        {rank + 1}
   572→      </span>
   573→      <div className="min-w-0 flex-1">
   574→        <FitLine>
   575→          <span className="font-display italic uppercase text-white truncate block" style={{ fontSize: "clamp(15px, 1.8vw, 22px)" }}>
   576→            {p.nick}
   577→          </span>
   578→        </FitLine>
   579→      </div>
   580→      {isWinner ? <Crown size={18} className="text-[#ffb830] shrink-0" /> : null}
   581→      <span
   582→        key={p.score}
   583→        className={`font-display italic tabular-nums leading-none ${flash ? "text-[#ffe9b0] obs-score-pop" : "text-[#ffb830]"}`}
   584→        style={{ fontSize: "clamp(22px, 3vw, 36px)" }}
   585→      >
   586→        {p.score}
   587→      </span>
   588→    </motion.div>
   589→  );
   590→}

   592→function SoundLeaderboard({ participants, winnerPid }: { participants: Participant[]; winnerPid: string | null }) {
   593→  const sorted = useMemo(
   594→    () => [...participants].sort((a, b) => b.score - a.score),
   595→    [participants]
   596→  );
   597→  if (sorted.length === 0) return null;
   598→  return (
   599→    <div className="w-full h-full flex flex-col">
   600→      <div className="label-cap mb-2 flex items-center gap-2 shrink-0">
   601→        <Trophy size={13} className="text-[#ffb830]" />
   602→        Marcador
   603→        <span className="text-[#6b6e78]">· {sorted.length} jugadores</span>
   604→      </div>
   605→      <div
   606→        className="grid gap-2 flex-1 content-center"
   607→        style={{
   608→          gridTemplateColumns: `repeat(auto-fit, minmax(min(100%, 180px), 1fr))`,
   609→        }}
   610→      >
   611→        {sorted.map((p, i) => (
   612→          <LeaderCard key={p.id} p={p} rank={i} isWinner={winnerPid === p.id} />
   613→        ))}
   614→      </div>
   615→    </div>
   616→  );
   617→}

   621→export function ContestViewer() {
   622→  const { contest } = useActiveTournament();
   623→  const { data: participants } = useParticipants(contest?.id ?? null);
   624→  const { data: sounds } = useSounds(contest?.id ?? null);
   625→  const state = useContestState(contest?.id ?? null);
   626→
   627→  /* ---- Audio: se reproduce una vez. El visor detecta cuando termina
   628→     y actualiza `soundActive` localmente (no depende de Firestore). ---- */
   629→  const audioRef = useRef<HTMLAudioElement | null>(null);
   630→  const [audioEl, setAudioEl] = useState<HTMLAudioElement | null>(null);
   631→  const [soundActive, setSoundActive] = useState(false);
   632→
   633→  const currentSoundId = state?.currentSoundId ?? null;
   634→  const fsSoundPlaying = state?.soundPlaying ?? false;
   635→  const currentSound = useMemo(() => {
   636→    if (!currentSoundId || !sounds.length) return null;
   637→    return sounds.find((s) => s.id === currentSoundId) ?? null;
   638→  }, [currentSoundId, sounds]);
   639→  const currentSoundData = currentSound?.data ?? null;
   640→
   641→  useEffect(() => {
   642→    const raf = requestAnimationFrame(() => {
   643→      // si no hay sonido o Firestore dice que no está sonando → limpiar
   644→      if (!currentSoundData || !fsSoundPlaying) {
   645→        if (audioRef.current) {
   646→          audioRef.current.pause();
   647→          audioRef.current = null;
   648→        }
   649→        setAudioEl(null);
   650→        setSoundActive(false);
   651→        return;
   652→      }
   653→      // recrear el Audio → suena una vez desde el inicio
   654→      if (audioRef.current) {
   655→        audioRef.current.pause();
   656→      }
   657→      const audio = new Audio(currentSoundData);
   658→      audio.loop = false;
   659→      audio.onended = () => {
   660→        // el audio terminó naturalmente → el visor deja de mostrar "Sonando"
   661→        setSoundActive(false);
   662→        setAudioEl(null);
   663→      };
   664→      audio.play().then(() => {
   665→        setSoundActive(true);
   666→        setAudioEl(audio);
   667→      }).catch(() => {
   668→        setSoundActive(false);
   669→      });
   670→      audioRef.current = audio;
   671→    });
   672→    return () => {
   673→      cancelAnimationFrame(raf);
   674→      if (audioRef.current) {
   675→        audioRef.current.pause();
   676→        audioRef.current = null;
   677→      }
   678→      setAudioEl(null);
   679→      setSoundActive(false);
   680→    };
   681→  }, [currentSoundData, fsSoundPlaying]);
   682→
   683→  if (!contest) {
   684→    return (
   685→      <div className="fixed inset-0 flex items-center justify-center" style={{ background: "#070708" }}>
   686→        <ShowBackground />
   687→        <div className="relative z-10 text-center px-6">
   688→          <span className="text-[11px] font-extrabold tracking-[0.5em] uppercase text-[#ff8095]">Visor de concursos</span>
   689→          <h1 className="font-display italic text-[34px] sm:text-[44px] uppercase leading-none mt-3">
   690→            <span className="text-silver-grad">Esperando</span> <span className="text-red-grad">concurso</span>
   691→          </h1>
   692→          <p className="text-[12px] font-bold tracking-[0.16em] uppercase text-[#8e919c] mt-6">
   693→            El administrador debe abrir un concurso
   694→          </p>
   695→        </div>
   696→      </div>
   697→    );
   698→  }
   699→
   700→  const isCosplay = contest.type === "cosplay";
   701→  const isSound = contest.type === "sound";
   702→  const spotlightPid = state?.spotlightPid ?? null;
   703→  const spotlight = participants.find((p) => p.id === spotlightPid) ?? null;
   704→  const winner = contest.winnerPid ? participants.find((p) => p.id === contest.winnerPid) : null;
   705→  const isFinished = contest.status === "finished";
   706→  const revealAnswer = state?.revealAnswer ?? false;
   707→  const revealQuestion = state?.revealQuestion ?? false;
   708→
   709→  /* El visor muestra el SoundStage cuando: hay sonido activo (realmente
   710→     reproduciéndose), o se pidió mostrar la pregunta, o se reveló la
   711→     respuesta. Si no → muestra el idle con branding HyperX. */
   712→  const showStage = soundActive || revealQuestion || revealAnswer;
   713→
   714→  return (
   715→    <div className="fixed inset-0 flex flex-col overflow-hidden" style={{ background: "#070708" }}>
   716→      <Backdrop variant="show" />
   717→      <ShowBackground />
   718→
   719→      {/* header con TÍTULO GRANDE del concurso */}
   720→      <header className="relative z-20 shrink-0 border-b border-white/10 bg-black/55 backdrop-blur-md flex flex-col items-center justify-center py-3 px-4">
   721→        <div className="flex items-center gap-3 mb-1">
   722→          <span className="text-[9px] sm:text-[10px] font-extrabold tracking-[0.4em] uppercase text-[#ffb830]">
   723→            {isCosplay ? "Concurso Cosplayer" : "Sonidos Gamer"}
   724→          </span>
   725→          <span className={`chip clip-tag ${isFinished ? "bg-[#ffb830] text-[#241500]" : contest.status === "live" ? "bg-[#e8102e] text-white" : "bg-white/[0.06] text-[#c9cbd3] border border-white/10"} !text-[8px] !px-2 !py-[2px]`}>
   726→            {isFinished ? "FINALIZADO" : contest.status === "live" ? "EN VIVO" : "REGISTRO"}
   727→          </span>
   728→        </div>
   729→        <FitLine>
   730→          <h1 className="font-display italic uppercase leading-none text-silver-grad" style={{ fontSize: "clamp(22px, 3.5vw, 40px)" }}>
   731→            {contest.name}
   732→          </h1>
   733→        </FitLine>
   734→        <div className="flex items-center gap-3 mt-1.5 text-[9px] font-bold tracking-[0.2em] uppercase text-[#6b6e78]">
   735→          <span className="flex items-center gap-1">
   736→            <Users size={11} />
   737→            {participants.length}
   738→          </span>
   739→          {isSound ? (
   740→            <span className="flex items-center gap-1">
   741→              <Music size={11} />
   742→              {sounds.length} sonidos
   743→            </span>
   744→          ) : null}
   745→        </div>
   746→      </header>
   747→
   748→      {/* cuerpo */}
   749→      <main className="relative flex-1 min-h-0 flex flex-col overflow-hidden">
   750→        {isCosplay ? (
   751→          <div className="flex-1 min-h-0 flex items-center justify-center py-4 overflow-y-auto">
   752→            <AnimatePresence mode="wait">
   753→              {winner ? (
   754→                <CosplaySpotlight key="winner" nick={winner.nick} name={winner.name} isWinner />
   755→              ) : spotlight ? (
   756→                <CosplaySpotlight key={spotlight.id} nick={spotlight.nick} name={spotlight.name} isWinner={false} />
   757→              ) : (
   758→                <CosplayGrid key="grid" participants={participants.filter((p) => !p.eliminated)} />
   759→              )}
   760→            </AnimatePresence>
   761→          </div>
   762→        ) : null}
   763→
   764→        {isSound ? (
   765→          <>
   766→            {/* ZONA SUPERIOR (50%): ganador / sonido + imagen + respuesta / idle */}
   767→            <div className="flex-[1_1_50%] min-h-0 flex items-center justify-center p-4 border-b border-white/8">
   768→              <AnimatePresence mode="wait">
   769→                {winner ? (
   770→                  <SoundWinner key="winner" nick={winner.nick} score={winner.score} />
   771→                ) : showStage && currentSound ? (
   772→                  <SoundStage
   773→                    key="stage"
   774→                    question={currentSound.question}
   775→                    answer={currentSound.answer}
   776→                    image={currentSound.image}
   777→                    revealAnswer={revealAnswer}
   778→                    revealQuestion={revealQuestion}
   779→                    audioEl={audioEl}
   780→                    soundActive={soundActive}
   781→                  />
   782→                ) : (
   783→                  <SoundIdle key="idle" isFinished={isFinished} />
   784→                )}
   785→              </AnimatePresence>
   786→            </div>
   787→            {/* ZONA INFERIOR (50%): marcador con cards grandes */}
   788→            <div className="flex-[1_1_50%] min-h-0 p-4 overflow-y-auto">
   789→              <SoundLeaderboard participants={participants} winnerPid={contest.winnerPid} />
   790→            </div>
   791→          </>
   792→        ) : null}
   793→      </main>
   794→
   795→      {/* footer */}
   796→      <footer className="relative z-20 shrink-0 h-9 border-t border-white/10 bg-black/70 flex items-center justify-center">
   797→        <span className="text-[9px] font-extrabold tracking-[0.3em] uppercase text-[#6b6e78]">
   798→          {isCosplay ? "Concurso Cosplayer · aplausos del público" : "Sonidos Gamer · adivina el juego"}
   799→        </span>
   800→      </footer>
   801→    </div>
   802→  );
   803→}
