    75→/* ============================================================
    76→   Vista TORNEO EN VIVO — administración del bracket
    77→   · Dos vistas: Bracket (árbol) y Rondas (columnas de cards)
    78→   · Panel lateral colapsable de edición del match

    82→
    83→type AdminView = "tree" | "rounds";
    84→
    85→export function LiveAdminView({
    86→  tournament,
    87→  players,
    88→  bracket,
    89→  loadingBracket,
    90→}: {
    91→  tournament: Tournament | null;
    92→  players: Player[];
    93→  bracket: Bracket | null;
    94→  loadingBracket: boolean;
    95→}) {
    96→  const [sel, setSel] = useState<{ r: number; m: number } | null>(null);
    97→  const [pickBank, setPickBank] = useState<{ r: number; m: number; slotIdx: number } | null>(null);
    98→  const [confirm, setConfirm] = useState<null | { kind: "remix" | "finish" | "wipe" | "reopen" | "results" | "fill" | "demos" }>(null);
    99→  const [busy, setBusy] = useState(false);
   100→  const [view, setView] = useState<AdminView>("tree");
   101→  const [panelOpen, setPanelOpen] = useState(true);
   102→  const [zoom, setZoom] = useState<number | null>(null); // null = auto-fit
   103→  const [startMins, setStartMins] = useState(1);
   104→
   105→  /* banco efectivo: reservas del registro antes de la primera eliminatoria;
   106→     después, los eliminados de la última ronda completada (banco rotativo) */
   107→  const bankInfo = useMemo(() => computeBank(bracket, players), [bracket, players]);
   108→  const participants = useMemo(
   109→    () => (tournament ? groupParticipants(players, tournament.modality) : []),
   110→    [players, tournament]
   111→  );
   112→  const officialsCount = useMemo(() => players.filter((p) => p.seat === "off").length, [players]);
   113→  const demoCount = useMemo(() => players.filter((p) => p.demo).length, [players]);
   114→  const missingPlayers = tournament ? playersCapacity(tournament) - officialsCount : 0;
   115→  const schedule = useMemo(
   116→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map()),
   117→    [bracket, tournament]
   118→  );
   119→  /* capacidad del escenario: matches simultáneos según las 4 PCs */
   120→  const maxSim = tournament ? maxSimultaneous(tournament.modality) : 1;
   121→  const liveCount = useMemo(
   122→    () => (bracket ? bracket.rounds.reduce((acc, round) => acc + round.filter((x) => x.status === "live").length, 0) : 0),
   123→    [bracket]
   124→  );
   125→
   126→  /* navegación secuencial de matches (para editar rápido sin recorrer todo) */
   127→  const flatMatches = useMemo(() => {
   128→    if (!bracket) return [];
   129→    const out: { r: number; m: number }[] = [];
   130→    bracket.rounds.forEach((round, r) => round.forEach((_, m) => out.push({ r, m })));
   131→    return out;
   132→  }, [bracket]);
   133→  const flatIdx = useMemo(() => {
   134→    if (!sel || !bracket) return -1;
   135→    return flatMatches.findIndex((x) => x.r === sel.r && x.m === sel.m);
   136→  }, [flatMatches, sel, bracket]);
   137→
   138→  if (!tournament) {
   139→    return (
   140→      <div className="panel clip-card">
   141→        <EmptyState
   142→          icon={<Radio size={26} />}
   143→          title="Selecciona un torneo"
   144→          message="Elige un torneo en el selector superior para administrar el bracket en vivo: ganadores, descalificaciones y reemplazos del banco."
   145→        />
   146→      </div>
   147→    );
   148→  }
   149→
   150→  const guard = async (fn: () => Promise<void>) => {
   151→    setBusy(true);
   152→    try {
   153→      await fn();
   154→    } catch (e) {
   155→      toast.error(e instanceof Error ? e.message : "Error de operación");
   156→    } finally {
   157→      setBusy(false);
   158→    }
   159→  };
   160→
   161→  const selMatch = sel && bracket ? bracket.rounds[sel.r]?.[sel.m] : null;
   162→
   163→  const stepMatch = (dir: 1 | -1) => {
   164→    const nx = flatMatches[flatIdx + dir];
   165→    if (nx) setSel(nx);
   166→  };
   167→
   168→  const zoomStep = (dir: 1 | -1) => {
   169→    setZoom((z) => {
   170→      const base = z ?? 0.8;
   171→      return Math.min(1.8, Math.max(0.3, Math.round((base + dir * 0.1) * 100) / 100));
   172→    });
   173→  };
   174→
   175→  return (
   176→    <div className="flex flex-col gap-3 min-h-0">
   177→      {/* ============ barra de info del torneo ============ */}
   178→      <div className="panel clip-card p-4 flex flex-wrap items-center gap-3">
   179→        <div className="flex items-center gap-3 min-w-0 mr-auto">
   180→          <span className="slashes w-8 h-4 inline-block shrink-0" aria-hidden />
   181→          <div className="min-w-0">
   182→            <div className="font-display italic text-[17px] uppercase truncate">{tournament.name}</div>
   183→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   184→              <StatusChip status={tournament.status} />
   185→              <ModalityChip modality={tournament.modality} />
   186→              {bracket ? (
   187→                <span
   188→                  className={`chip clip-tag border ${
   189→                    liveCount >= maxSim
   190→                      ? "bg-[#e8102e]/15 text-[#ff8095] border-[#e8102e]/40"
   191→                      : "bg-[#e8102e]/12 text-[#ff8095] border-[#e8102e]/35"
   192→                  }`}
   193→                  title="Matches en juego / máximo simultáneo según las PCs del escenario (4 PCs)"
   194→                >
   195→                  <Monitor size={10} />
   196→                  EN JUEGO {liveCount}/{maxSim}
   197→                </span>
   198→              ) : null}
   199→              {bracket ? (
   200→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   201→                  {doneMatches(bracket)}/{totalMatches(bracket)} MATCHES
   202→                </span>
   203→              ) : (
   204→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   205→                  {officialsCount}/{playersCapacity(tournament)} JUGADORES
   206→                </span>
   207→              )}
   208→              {tournament.startAt ? (
   209→                <span className="chip clip-tag bg-[#ffb830]/12 text-[#ffb830] border border-[#ffb830]/35">
   210→                  <TimerReset size={10} />
   211→                  INICIA ≈ {fmtTime(tournament.startAt)}
   212→                </span>
   213→              ) : null}
   214→            </div>
   215→          </div>
   216→        </div>
   217→        <span className="text-[9px] font-extrabold uppercase tracking-[0.18em] text-[#6b6e78] hidden md:inline">
   218→          Cada fase se refleja automáticamente en el visor
   219→        </span>
   220→      </div>
   221→
   222→      {/* ============ FLUJO POR FASES ============ */}
   223→      <PhaseBar
   224→        tournament={tournament}
   225→        bracket={bracket}
   226→        participantsCount={participants.length}
   227→        officialsCount={officialsCount}
   228→        missingPlayers={missingPlayers}
   229→        demoCount={demoCount}
   230→        bankCount={bankInfo.entries.length}
   231→        startMins={startMins}
   232→        setStartMins={setStartMins}
   233→        busy={busy}
   234→        guard={guard}
   235→        onConfirm={(kind) => setConfirm({ kind })}
   236→        onLaunchMix={() =>
   237→          guard(async () => {
   238→            await launchMix(tournament.id, players, tournament);
   239→            setSel(null);
   240→            toast.success("Mix match lanzado — el visor reproduce la animación");
   241→          })
   242→        }
   243→      />
   244→
   245→      {/* ============ cuerpo ============ */}
   246→      {loadingBracket ? (
   247→        <div className="panel clip-card p-14 flex items-center justify-center text-[#8e919c]">
   248→          <Loader2 size={20} className="animate-spin" />
   249→        </div>
   250→      ) : !bracket ? (
   251→        <div className="panel clip-card">
   252→          <EmptyState
   253→            icon={<Zap size={26} />}
   254→            title="Brackets no creados"
   255→            message={`Cuando estés listo, lanza el MIX MATCH: se randomizarán los ${participants.length} participantes y el visor de espectadores mostrará la animación de llenado de brackets.`}
   256→            action={
   257→              <div className="flex flex-wrap gap-2 justify-center">
   258→                <Btn
   259→                  variant="red"
   260→                  disabled={participants.length < 2}
   261→                  onClick={() =>
   262→                    guard(async () => {
   263→                      await launchMix(tournament.id, players, tournament);
   264→                      toast.success("Mix match lanzado");
   265→                    })
   266→                  }
   267→                >
   268→                  <Rocket size={13} />
   269→                  Lanzar mix match
   270→                </Btn>
   271→                <Btn
   272→                  variant="dark"
   273→                  onClick={() =>
   274→                    guard(async () => {
   275→                      await setStartAt(tournament.id, Date.now() + startMins * 60_000);
   276→                      toast.success(`Inicio configurado en ${startMins} min — el visor cambia a brackets con el cronómetro`);
   277→                    })
   278→                  }
   279→                >
   280→                  <TimerReset size={13} />
   281→                  Inicio en {startMins} min
   282→                </Btn>
   283→              </div>
   284→            }
   285→          />
   286→          {/* vista previa de participantes */}
   287→          {participants.length > 0 ? (
   288→            <div className="px-5 pb-5">
   289→              <div className="label-cap mb-2 flex items-center gap-2">
   290→                <Users size={11} />
   291→                Participantes que entrarán al sorteo ({participants.length})
   292→              </div>
   293→              <div className="flex flex-wrap gap-1.5 max-h-36 overflow-y-auto">
   294→                {participants.map((p, i) => (
   295→                  <span key={p.pid} className="chip clip-tag bg-white/[0.05] text-[#c9cbd3] border border-white/10">
   296→                    {i + 1}. {p.label}
   297→                  </span>
   298→                ))}
   299→              </div>
   300→            </div>
   301→          ) : null}
   302→        </div>
   303→      ) : (
   304→        <>
   305→          {/* ============ toolbar de vistas ============ */}
   306→          <div className="panel clip-card-sm px-3 py-2 flex flex-wrap items-center gap-2">
   307→            <div className="flex gap-1 p-0.5 bg-black/40 border border-white/10" role="radiogroup" aria-label="Vista del bracket">
   308→              <button
   309→                type="button"
   310→                role="radio"
   311→                aria-checked={view === "tree"}
   312→                onClick={() => setView("tree")}
   313→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   314→                  view === "tree" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   315→                }`}
   316→              >
   317→                <LayoutPanelLeft size={12} />
   318→                Bracket
   319→              </button>
   320→              <button
   321→                type="button"
   322→                role="radio"
   323→                aria-checked={view === "rounds"}
   324→                onClick={() => setView("rounds")}
   325→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   326→                  view === "rounds" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   327→                }`}
   328→              >
   329→                <Columns3 size={12} />
   330→                Rondas
   331→              </button>
   332→            </div>
   333→
   334→            {selMatch ? (
   335→              <span className="chip clip-tag bg-[#e8102e]/15 text-[#ff8095] border border-[#e8102e]/40 text-[9px] flex items-center gap-1.5">
   336→                <Hand size={10} />
   337→                {matchTag(bracket, sel!.r, sel!.m)} seleccionado
   338→              </span>
   339→            ) : (
   340→              <span className="text-[9px] font-extrabold uppercase tracking-[0.16em] text-[#6b6e78] hidden sm:inline">
   341→                Clic en un match para administrarlo
   342→              </span>
   343→            )}
   344→
   345→            <div className="ml-auto flex items-center gap-1.5">
   346→              {/* navegación prev/next */}
   347→              <div className="flex items-center gap-1">
   348→                <IconBtn title="Match anterior" disabled={flatIdx <= 0} onClick={() => stepMatch(-1)}>
   349→                  <ChevronLeft size={14} />
   350→                </IconBtn>
   351→                <IconBtn title="Match siguiente" disabled={flatIdx < 0 || flatIdx >= flatMatches.length - 1} onClick={() => stepMatch(1)}>
   352→                  <ChevronRight size={14} />
   353→                </IconBtn>
   354→              </div>
   355→
   356→              {/* zoom (solo vista bracket) */}
   357→              {view === "tree" ? (
   358→                <div className="flex items-center gap-1 pl-2 border-l border-white/10">
   359→                  <IconBtn title="Alejar" onClick={() => zoomStep(-1)}>
   360→                    <ZoomOut size={14} />
   361→                  </IconBtn>
   362→                  <span className="text-[9px] font-extrabold tabular-nums text-[#8e919c] w-10 text-center uppercase">
   363→                    {zoom ? `${Math.round(zoom * 100)}%` : "AUTO"}
   364→                  </span>
   365→                  <IconBtn title="Acercar" onClick={() => zoomStep(1)}>
   366→                    <ZoomIn size={14} />
   367→                  </IconBtn>
   368→                  <IconBtn title="Ajustar a la pantalla" active={zoom === null} onClick={() => setZoom(null)}>
   369→                    <Maximize2 size={13} />
   370→                  </IconBtn>
   371→                </div>
   372→              ) : null}
   373→
   374→              {/* toggle panel lateral */}
   375→              <button
   376→                type="button"
   377→                onClick={() => setPanelOpen((v) => !v)}
   378→                title={panelOpen ? "Ocultar panel del match" : "Mostrar panel del match"}
   379→                aria-pressed={panelOpen}
   380→                className={`btn-press p-2 border transition-colors hidden xl:inline-flex ${
   381→                  panelOpen
   382→                    ? "border-[#e8102e]/60 bg-[#e8102e]/10 text-white"
   383→                    : "border-white/12 text-[#8e919c] hover:text-white"
   384→                }`}
   385→              >
   386→                {panelOpen ? <PanelRightClose size={14} /> : <PanelRightOpen size={14} />}
   387→              </button>
   388→            </div>
   389→          </div>
   390→
   391→          {/* ============ stage + panel lateral colapsable ============ */}
   392→          <div className="flex gap-3 items-stretch min-h-0">
   393→            {/* área principal */}
   394→            <div className="flex-1 min-w-0 min-h-0">
   395→              <div className="panel clip-card p-3 min-h-[420px] h-[calc(100vh-330px)] max-h-[80vh] flex flex-col">
   396→                <div className="flex items-center justify-between px-2 pb-2">
   397→                  <span className="label-cap">
   398→                    {view === "tree" ? "Bracket · clic en un match para administrar" : "Rondas · cards seleccionables con botón de ganador directo"}
   399→                  </span>
   400→                  <span className="label-cap">{MODALITY_LABEL[tournament.modality]} · a {tournament.winsNeeded} wins</span>
   401→                </div>
   402→
   403→                {/* monitor del escenario: asignación de las 4 PCs por match */}
   404→                <PcDock tournament={tournament} bracket={bracket} onSelectMatch={(r, m) => setSel({ r, m })} />
   405→
   406→                {view === "tree" ? (
   407→                  <div className="flex-1 min-h-0 border border-white/8 bg-black/30 clip-card-sm">
   408→                    <FitStage deps={[bracket, tournament.modality, zoom]} maxScale={0.95} minScale={0.4} zoom={zoom} className="w-full h-full">
   409→                      <BracketTree
   410→                        bracket={bracket}
   411→                        modality={tournament.modality}
   412→                        mode="admin"
   413→                        selectedId={selMatch?.id ?? null}
   414→                        onSelectMatch={(r, m) => setSel({ r, m })}
   415→                        schedule={schedule}
   416→                        showTimes
   417→                      />
   418→                    </FitStage>
   419→                  </div>
   420→                ) : (
   421→                  <RoundColumns
   422→                    bracket={bracket}
   423→                    modality={tournament.modality}
   424→                    times={schedule}
   425→                    selectedId={selMatch?.id ?? null}
   426→                    onSelectMatch={(r, m) => setSel({ r, m })}
   427→                    onPickWinner={(r, m, slotIdx) =>
   428→                      guard(async () => {
   429→                        await setWinner(tournament.id, r, m, slotIdx);
   430→                        setSel({ r, m });
   431→                        toast.success(`${bracket.rounds[r][m].slots[slotIdx].label} avanza`);
   432→                      })
   433→                    }
   434→                  />
   435→                )}
   436→              </div>
   437→            </div>
   438→
   439→            {/* panel lateral (desktop, colapsable) */}
   440→            <aside
   441→              className={`hidden xl:block shrink-0 overflow-hidden transition-[width] duration-300 ease-out ${
   442→                panelOpen ? "w-[376px]" : "w-0"
   443→              }`}
   444→              aria-hidden={!panelOpen}
   445→            >
   446→              <div className="w-[376px] h-full">
   447→                <MatchPanel
   448→                  tournament={tournament}
   449→                  bracket={bracket}
   450→                  sel={sel}
   451→                  setSel={setSel}
   452→                  selMatch={selMatch}
   453→                  schedule={schedule}
   454→                  bankInfo={bankInfo}
   455→                  onClose={() => setPanelOpen(false)}
   456→                  guard={guard}
   457→                  setPickBank={setPickBank}
   458→                  busy={busy}
   459→                  stepMatch={stepMatch}
   460→                />
   461→              </div>
   462→            </aside>
   463→          </div>
   464→
   465→          {/* panel del match (móvil/tablet: debajo) */}
   466→          <div className="xl:hidden">
   467→            <MatchPanel
   468→              tournament={tournament}
   469→              bracket={bracket}
   470→              sel={sel}
   471→              setSel={setSel}
   472→              selMatch={selMatch}
   473→              schedule={schedule}
   474→              bankInfo={bankInfo}
   475→              onClose={() => setSel(null)}
   476→              guard={guard}
   477→              setPickBank={setPickBank}
   478→              busy={busy}
   479→              stepMatch={stepMatch}
   480→            />
   481→          </div>
   482→
   483→          {/* pestaña flotante para reabrir el panel en desktop */}
   484→          {!panelOpen && selMatch ? (
   485→            <button
   486→              type="button"
   487→              onClick={() => setPanelOpen(true)}
   488→              className="btn-press fixed right-0 top-1/2 -translate-y-1/2 z-40 hidden xl:flex flex-col items-center gap-2 red-badge clip-badge-l px-2 py-4 text-[10px] font-extrabold uppercase tracking-[0.2em] shadow-[0_10px_30px_rgba(232,16,46,0.4)]"
   489→              style={{ writingMode: "vertical-rl" }}
   490→            >
   491→              <ChevronLeft size={12} />
   492→              Panel del match
   493→            </button>
   494→          ) : null}
   495→        </>
   496→      )}
   497→
   498→      {/* ============ modal banco ============ */}
   499→      <Modal open={!!pickBank} onClose={() => setPickBank(null)} title="Reemplazar desde el banco">
   500→        <p className="text-[11px] font-bold text-[#8e919c] uppercase tracking-wider mb-1">
   501→          Selecciona la reserva que entrará al match
   502→        </p>
   503→        <p className="text-[10px] font-semibold text-[#6b6e78] mb-3 leading-snug">
   504→          {bankInfo.fromRound === null
   505→            ? "Reservas del registro · el banco se renueva con los eliminados de cada eliminatoria"
   506→            : `Banco renovado: reservas eliminadas en ${bankInfo.sourceLabel}`}
   507→        </p>
   508→        <div className="space-y-2 max-h-72 overflow-y-auto">
   509→          {bankInfo.entries.length === 0 ? (
   510→            <p className="text-[11px] font-bold text-[#6b6e78] uppercase">
   511→              {bankInfo.fromRound === null
   512→                ? "No hay jugadores en el banco"
   513→                : "Banco vacío — se llena al terminar la ronda anterior"}
   514→            </p>
   515→          ) : (
   516→            bankInfo.entries.map((e) => (
   517→              <button
   518→                key={e.key}
   519→                type="button"
   520→                onClick={async () => {
   521→                  if (!pickBank || !tournament) return;
   522→                  await guard(async () => {
   523→                    await replaceFromBank(
   524→                      tournament.id,
   525→                      pickBank.r,
   526→                      pickBank.m,
   527→                      pickBank.slotIdx,

  1108→function MiniCountdown({ target }: { target: number }) {
  1109→  const [now, setNow] = useState(() => Date.now());
  1110→  useEffect(() => {
  1111→    const t = setInterval(() => setNow(Date.now()), 500);
  1112→    return () => clearInterval(t);
  1113→  }, []);
  1114→  const total = Math.max(0, Math.floor((target - now) / 1000));
  1115→  const h = Math.floor(total / 3600);
  1116→  const m = Math.floor((total % 3600) / 60);
  1117→  const s = total % 60;
  1118→  const pad = (x: number) => String(x).padStart(2, "0");
  1119→  return (
  1120→    <span className="font-display italic text-[20px] leading-none text-silver-grad tabular-nums">
  1121→      {h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`}
  1122→    </span>
  1123→  );
  1124→}
