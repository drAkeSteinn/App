    43→    43→/* Detecta el primer evento "dramático" entre dos versiones del bracket:
    44→    44→   match que pasa a DONE (ganador marcado o cambiado) o a LIVE (en juego). */
    45→    45→function findSpotlightEvent(cur: Bracket, prev: Bracket, winsNeeded: number): SpotlightData | null {
    46→    46→  for (let r = 0; r < cur.rounds.length; r++) {
    47→    47→    const round = cur.rounds[r];
    48→    48→    for (let m = 0; m < round.length; m++) {
    49→    49→      const match = round[m];
    50→    50→      const old = prev.rounds[r]?.[m];
    51→    51→      if (!match || !old || match.bye) continue;
    52→    52→      const becameDone = match.status === "done" && old.status !== "done";
    53→    53→      const winnerChanged = match.status === "done" && old.status === "done" && match.w !== old.w;
    54→    54→      const becameLive = match.status === "live" && old.status !== "live";
    55→    55→      if (becameDone || winnerChanged || becameLive) {
    56→    56→        return {
    57→    57→          id: `${match.id}-${match.status}-${match.w}-${Date.now()}`,
    58→    58→          kind: match.status === "live" ? "live" : "win",
    59→    59→          tag: matchTag(cur, r, m),
    60→    60→          a: match.slots[0]?.label || "—",
    61→    61→          b: match.slots[1]?.label || "—",
    62→    62→          aScore: match.slots[0]?.score ?? 0,
    63→    63→          bScore: match.slots[1]?.score ?? 0,
    64→    64→          winner: match.w,
    65→    65→          winsNeeded,
    66→    66→        };
    67→    67→      }
    68→    68→    }
    69→    69→  }
    70→    70→  return null;
    71→    71→}

   309→   309→export function SpectatorView({ tid }: { tid: string | null }) {
   310→   310→  const { data: tournaments, loading, error } = useTournaments();
   311→   311→  const tournament: Tournament | null = useMemo(
   312→   312→    () => tournaments.find((t) => t.id === tid) ?? null,
   313→   313→    [tournaments, tid]
   314→   314→  );
   315→   315→  const { data: players } = usePlayers(tournament ? tid : null);
   316→   316→  const { data: bracket } = useBracket(tournament ? tid : null);
   317→   317→
   318→   318→  const status = tournament?.status ?? "open";
   319→   319→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   320→   320→  const [schedOpen, setSchedOpen] = useState(false);
   321→   321→  const [revealCount, setRevealCount] = useState(0);
   322→   322→  const [showPodium, setShowPodium] = useState(true);
   323→   323→  const [spotlight, setSpotlight] = useState<SpotlightData | null>(null);
   324→   324→  const prevBracketRef = useRef<{ tid: string | null; bracket: Bracket } | null>(null);
   325→   325→  const rootRef = useRef<HTMLDivElement | null>(null);
   326→   326→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   327→   327→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   328→   328→
   329→   329→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   330→   330→     se refleja sola en el visor). El toggle manual gana hasta que el status
   331→   331→     vuelva a cambiar, porque la anulación queda atada al status vigente.
   332→   332→     CUENTA REGRESIVA: en cuanto el admin programa el inicio (startAt), el
   333→   333→     visor salta solo a la pestaña de brackets, donde el cronómetro de inicio
   334→   334→     se ve en grande; si cancela la cuenta regresiva regresa al modo registro. */
   335→   335→  const inRegisterPhase = status === "open" || status === "closed";
   336→   336→  const autoMode: ViewerMode = inRegisterPhase ? (tournament?.startAt ? "brackets" : "reg") : "brackets";
   337→   337→  const mode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   338→   338→
   339→   339→  /* sincroniza el modo en la URL para compartir la vista correcta */
   340→   340→  useEffect(() => {
   341→   341→    try {
   342→   342→      const url = new URL(window.location.href);
   343→   343→      if (mode === "reg") url.searchParams.set("mode", "reg");
   344→   344→      else url.searchParams.delete("mode");
   345→   345→      if (url.toString() !== window.location.href) window.history.replaceState(null, "", url.toString());
   346→   346→    } catch {
   347→   347→      /* ignore */
   348→   348→    }
   349→   349→  }, [mode]);
   350→   350→
   351→   351→  const changeMode = (m: ViewerMode) => {
   352→   352→    setOverride({ status, mode: m });
   353→   353→  };
   354→   354→
   355→   355→  // animación de mix
   356→   356→  const totalSlots = bracket ? bracket.rounds[0].length * 2 : 0;
   357→   357→  const pool = useMemo(() => {
   358→   358→    if (!bracket) return [];
   359→   359→    const labels = bracket.rounds[0].flatMap((m) => m.slots.map((s) => s.label || "TBD"));
   360→   360→    return Array.from(new Set(labels));
   361→   361→  }, [bracket]);
   362→   362→
   363→   363→  useEffect(() => {
   364→   364→    if (status !== "mixing" || !bracket || mode !== "brackets") return;
   365→   365→    const r0 = requestAnimationFrame(() => {
   366→   366→      setRevealCount(0);
   367→   367→      setShowPodium(false);
   368→   368→    });
   369→   369→    toRef.current = setTimeout(() => {
   370→   370→      ivRef.current = setInterval(() => {
   371→   371→        setRevealCount((c) => {
   372→   372→          if (c >= totalSlots) {
   373→   373→            if (ivRef.current) clearInterval(ivRef.current);
   374→   374→            return c;
   375→   375→          }
   376→   376→          return c + 1;
   377→   377→        });
   378→   378→      }, 240);
   379→   379→    }, 2400);
   380→   380→    return () => {
   381→   381→      cancelAnimationFrame(r0);
   382→   382→      if (toRef.current) clearTimeout(toRef.current);
   383→   383→      if (ivRef.current) clearInterval(ivRef.current);
   384→   384→    };
   385→   385→  }, [status, bracket, totalSlots, mode]);
   386→   386→
   387→   387→  useEffect(() => {
   388→   388→    if (status === "finished") {
   389→   389→      const r = requestAnimationFrame(() => setShowPodium(true));
   390→   390→      return () => cancelAnimationFrame(r);
   391→   391→    }
   392→   392→  }, [status]);
   393→   393→
   394→   394→  /* ----- SPOTLIGHT: ganador marcado / match en juego → corte dramático.
   395→   395→     Compara cada snapshot del bracket con el anterior; al detectar un match
   396→   396→     que pasa a DONE (o cambia de ganador) o a LIVE, lanza el overlay. ----- */
   397→   397→  useEffect(() => {
   398→   398→    const prev = prevBracketRef.current;
   399→   399→    prevBracketRef.current = tid && bracket ? { tid, bracket } : null;
   400→   400→    if (!bracket || !prev || prev.tid !== tid || status !== "live") return;
   401→   401→    const event = findSpotlightEvent(bracket, prev.bracket, tournament?.winsNeeded ?? 1);
   402→   402→    if (event) setSpotlight(event);
   403→   403→  }, [bracket, status, tid, tournament]);
   404→   404→
   405→   405→  /* el corte se va solo tras 5 segundos y regresa a los brackets */
   406→   406→  useEffect(() => {
   407→   407→    if (!spotlight) return;
   408→   408→    const t = setTimeout(() => setSpotlight(null), 5000);

   724→   724→    </div>
   725→   725→  );
   726→   726→}

   728→   728→/* Si no hay matches programados con participantes definidos, cae al ticker */

   730→   730→  bracket,
   731→   731→  schedule,
   732→   732→  tournament,
   733→   733→  tickerItems,
   734→   734→}: {
   735→   735→  bracket: Bracket;
   736→   736→  schedule: Map<string, number>;
   737→   737→  tournament: Tournament;
   738→   738→  tickerItems: string[];
   739→   739→}) {
   740→   740→  const hasAny = useMemo(() => hasRailMatches(bracket, schedule), [bracket, schedule]);
   741→   741→
   742→   742→  if (hasAny) return <MatchRail bracket={bracket} schedule={schedule} tournament={tournament} />;
   743→   743→  return <Ticker items={tickerItems} />;
   744→   744→}
