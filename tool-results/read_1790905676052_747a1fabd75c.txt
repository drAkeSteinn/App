    42→function useClock() {
    43→  const [now, setNow] = useState<Date | null>(null);
    44→  useEffect(() => {
    45→    const first = requestAnimationFrame(() => setNow(new Date()));
    46→    const t = setInterval(() => setNow(new Date()), 1000);
    47→    return () => {
    48→      cancelAnimationFrame(first);
    49→      clearInterval(t);
    50→    };
    51→  }, []);
    52→  return now;
    53→}

   262→/* ---------- Cuerpo principal ---------- */
   263→export function SpectatorView({ tid }: { tid: string | null }) {
   264→  const { data: tournaments, loading, error } = useTournaments();
   265→  const tournament: Tournament | null = useMemo(
   266→    () => tournaments.find((t) => t.id === tid) ?? null,
   267→    [tournaments, tid]
   268→  );
   269→  const { data: players } = usePlayers(tournament ? tid : null);
   270→  const { data: bracket } = useBracket(tournament ? tid : null);
   271→
   272→  const status = tournament?.status ?? "open";
   273→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   274→  const [schedOpen, setSchedOpen] = useState(false);
   275→  const [revealCount, setRevealCount] = useState(0);
   276→  const [showPodium, setShowPodium] = useState(true);
   277→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   278→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   279→
   280→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   281→     se refleja sola en el visor). El toggle manual gana hasta que el status
   282→     vuelva a cambiar, porque la anulación queda atada al status vigente. */
   283→  const autoMode: ViewerMode = status === "open" || status === "closed" ? "reg" : "brackets";
   284→  const mode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   285→
   286→  /* sincroniza el modo en la URL para compartir la vista correcta */
   287→  useEffect(() => {
   288→    try {
   289→      const url = new URL(window.location.href);
   290→      if (mode === "reg") url.searchParams.set("mode", "reg");
   291→      else url.searchParams.delete("mode");
   292→      if (url.toString() !== window.location.href) window.history.replaceState(null, "", url.toString());
   293→    } catch {
   294→      /* ignore */
   295→    }
   296→  }, [mode]);
   297→
   298→  const changeMode = (m: ViewerMode) => {
   299→    setOverride({ status, mode: m });
   300→  };
   301→
   302→  // animación de mix
   303→  const totalSlots = bracket ? bracket.rounds[0].length * 2 : 0;
   304→  const pool = useMemo(() => {
   305→    if (!bracket) return [];
   306→    const labels = bracket.rounds[0].flatMap((m) => m.slots.map((s) => s.label || "TBD"));
   307→    return Array.from(new Set(labels));
   308→  }, [bracket]);
   309→
   310→  useEffect(() => {
   311→    if (status !== "mixing" || !bracket || mode !== "brackets") return;
   312→    const r0 = requestAnimationFrame(() => {
   313→      setRevealCount(0);
   314→      setShowPodium(false);
   315→    });
   316→    toRef.current = setTimeout(() => {
   317→      ivRef.current = setInterval(() => {
   318→        setRevealCount((c) => {
   319→          if (c >= totalSlots) {
   320→            if (ivRef.current) clearInterval(ivRef.current);
   321→            return c;
   322→          }
   323→          return c + 1;
   324→        });
   325→      }, 240);
   326→    }, 2400);
   327→    return () => {
   328→      cancelAnimationFrame(r0);
   329→      if (toRef.current) clearTimeout(toRef.current);
   330→      if (ivRef.current) clearInterval(ivRef.current);
   331→    };
   332→  }, [status, bracket, totalSlots, mode]);
   333→
   334→  useEffect(() => {
   335→    if (status === "finished") {
   336→      const r = requestAnimationFrame(() => setShowPodium(true));
   337→      return () => cancelAnimationFrame(r);
   338→    }
   339→  }, [status]);
   340→
   341→  const schedule = useMemo(
   342→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map<string, number>()),
   343→    [bracket, tournament]
   344→  );
   345→
   346→  const liveMatches = useMemo(() => {
   347→    if (!bracket) return [];
   348→    const out: { tag: string; a: string; b: string }[] = [];
   349→    bracket.rounds.forEach((round, r) =>
   350→      round.forEach((match, m) => {
   351→        if (match.status === "live") {
   352→          out.push({
   353→            tag: matchTag(bracket, r, m),
   354→            a: match.slots[0].label || "—",
   355→            b: match.slots[1].label || "—",
   356→          });
   357→        }
   358→      })
   359→    );
   360→    return out;
   361→  }, [bracket]);
   362→
   363→  const tickerItems = useMemo(() => {
   364→    if (!tournament || !bracket) return [];
   365→    const items: string[] = [];
   366→    const upcoming: { time: number; text: string }[] = [];
   367→    bracket.rounds.forEach((round, r) =>
   368→      round.forEach((match, m) => {
   369→        const st = schedule.get(match.id);
   370→        if (st && match.status === "ready" && match.slots.every((s) => s.pid) && st > Date.now() - 60_000) {
   371→          upcoming.push({
   372→            time: st,
   373→            text: `SIGUIENTE · ${matchTag(bracket, r, m)}: ${match.slots[0].label} VS ${match.slots[1].label} ≈ ${fmtTimeSafe(st)}`,
   374→          });
   375→        }
   376→      })
   377→    );
   378→    upcoming.sort((a, b) => a.time - b.time);
   379→    items.push(...upcoming.slice(0, 4).map((u) => u.text));
   380→    const bankCount = players.filter((p) => p.seat === "bank").length;
   381→    items.push(`BANCO DE RESERVAS: ${bankCount} DISPONIBLES`);
   382→    items.push(`${MODALITY_LABEL[tournament.modality]} · ${totalMatches(bracket)} MATCHES · A ${tournament.winsNeeded} WINS`);
   383→    items.push("JUEGA · COMpite · CONECTA — GAMING SIN LÍMITES");
   384→    return items;
   385→  }, [bracket, tournament, schedule, players]);
   386→
   387→  /* ----- selector de transmisión ----- */
   388→  if (!tid || (!loading && !tournament)) {
   389→    return (
   390→      <div className="fixed inset-0 overflow-hidden" style={{ background: "#070708" }}>
   391→        <Backdrop variant="show" />
   392→        <div className="h-full overflow-y-auto">
   393→          <div className="max-w-4xl mx-auto px-6 py-16">
   394→            <div className="text-center mb-10">
   395→              <span className="text-[11px] font-extrabold tracking-[0.5em] uppercase text-[#ff8095]">Visor de espectadores</span>
   396→              <h1 className="font-display italic text-[38px] sm:text-[52px] uppercase leading-none mt-2">
   397→                <span className="text-silver-grad">Transmisiones</span> <span className="text-red-grad">en vivo</span>
   398→              </h1>
   399→            </div>
   400→            {loading ? (
   401→              <div className="flex justify-center py-16 text-[#8e919c]">
   402→                <Loader2 size={26} className="animate-spin" />
   403→              </div>
   404→            ) : error ? (
   405→              <div className="panel clip-card p-4 text-[12px] font-bold text-[#ff8095]">Error de Firebase: {error}</div>
   406→            ) : tournaments.length === 0 ? (
   407→              <div className="panel clip-card">
   408→                <EmptyState
   409→                  icon={<MonitorPlay size={26} />}
   410→                  title="Sin torneos todavía"
   411→                  message="Cuando el organizador cree un torneo aparecerá aquí para que lo veas en pantalla grande."
   412→                />
   413→              </div>
   414→            ) : (
   415→              <div className="grid grid-cols-1 sm:grid-cols-2 gap-4">
   416→                {[...tournaments]
   417→                  .sort((a, b) => {
   418→                    const order = { live: 0, mixing: 1, closed: 2, open: 3, finished: 4 } as const;
   419→                    return order[a.status] - order[b.status];
   420→                  })
   421→                  .map((t) => (
   422→                    <a
   423→                      key={t.id}
   424→                      href={`/?v=show&t=${t.id}`}
   425→                      className="panel clip-card hover-lift p-5 flex items-center gap-4"
   426→                    >
   427→                      <div className="w-14 h-14 shrink-0 clip-card-sm border border-white/12 bg-black/40 flex items-center justify-center overflow-hidden">
   428→                        {t.logo ? (
   429→                           
   430→                          <img src={t.logo} alt={`Logo ${t.name}`} className="w-full h-full object-contain" />
   431→                        ) : (
   432→                          <Trophy size={20} className="text-[#e8102e]" />
   433→                        )}
   434→                      </div>
   435→                      <div className="min-w-0 flex-1">
   436→                        <div className="font-display italic text-[17px] uppercase truncate">{t.name}</div>
   437→                        <div className="flex items-center gap-2 mt-1.5 flex-wrap">
   438→                          <StatusChip status={t.status} />
   439→                          <ModalityChip modality={t.modality} />
   440→                        </div>
   441→                      </div>
   442→                      <Radio size={18} className="text-[#e8102e] shrink-0" />
   443→                    </a>
   444→                  ))}
   445→              </div>
   446→            )}
   447→            <p className="text-center text-[10px] font-bold tracking-[0.24em] uppercase text-[#6b6e78] mt-10">
   448→              Abre esta vista en otra pantalla para transmitir el torneo
   449→            </p>
   450→          </div>
   451→        </div>
   452→      </div>
   453→    );
   454→  }
   455→
   456→  if (!tournament) {
   457→    return (
   458→      <div className="fixed inset-0 flex items-center justify-center" style={{ background: "#070708" }}>
   459→        <Loader2 size={28} className="animate-spin text-[#e8102e]" />
   460→      </div>
   461→    );
   462→  }
   463→
   464→  const mixing = status === "mixing";
   465→  const mixDone = mixing && totalSlots > 0 && revealCount >= totalSlots;
   466→  const railActive = status === "live" && mode === "brackets";
   467→
   468→  return (
   469→    <div className="fixed inset-0 flex flex-col overflow-hidden" style={{ background: "#070708" }}>
   470→      <Backdrop variant="show" />
   471→      <ShowHeader
   472→        t={tournament}
   473→        mode={mode}
   474→        onMode={changeMode}
   475→        onSchedule={() => setSchedOpen(true)}
   476→        showScheduleBtn={!!bracket && mode === "brackets"}
   477→      />
   478→
   479→      {/* spotlight de matches en juego */}
   480→      <AnimatePresence>
   481→        {status === "live" && mode === "brackets" && liveMatches.length > 0 ? (
   482→          <motion.div
   483→            initial={{ y: -40, opacity: 0 }}
   484→            animate={{ y: 0, opacity: 1 }}
   485→            exit={{ y: -40, opacity: 0 }}
   486→            className="relative z-20 shrink-0 live-glow bg-[linear-gradient(90deg,rgba(110,10,26,0.85),rgba(232,16,46,0.9),rgba(110,10,26,0.85))] flex items-center justify-center gap-3 py-2 px-4"
   487→          >
   488→            <span className="w-2.5 h-2.5 bg-white rounded-full blink shrink-0" aria-hidden />
   489→            <span className="text-[12px] sm:text-[14px] font-extrabold tracking-[0.2em] uppercase text-white truncate">
   490→              {liveMatches
   491→                .slice(0, 2)
   492→                .map((l) => `${l.tag}: ${l.a} VS ${l.b}`)
   493→                .join("  ///  ")}
   494→            </span>
   495→            <span className="w-2.5 h-2.5 bg-white rounded-full blink shrink-0" aria-hidden />
   496→          </motion.div>
   497→        ) : null}
   498→      </AnimatePresence>
   499→
   500→      <main className="relative flex-1 min-h-0">
   501→        <AnimatePresence mode="wait">
   502→          {/* MODO REGISTRO */}
   503→          {mode === "reg" ? (
   504→            <RegistrationBroadcast key="reg" t={tournament} players={players} />
   505→          ) : (
   506→            <>
   507→              {/* PREGAME (registro abierto o cerrado, vista brackets manual) */}
   508→              {status === "open" || status === "closed" ? <Pregame key="pre" t={tournament} players={players} /> : null}
   509→
   510→              {/* MIXING / LIVE / FINISHED: bracket */}
   511→              {status !== "open" && status !== "closed" ? (
   512→                <motion.div
   513→                  key="bracket"
   514→                  initial={{ opacity: 0, scale: 0.97 }}
   515→                  animate={{ opacity: 1, scale: 1 }}
   516→                  exit={{ opacity: 0 }}
   517→                  transition={{ duration: 0.5 }}
   518→                  className={`absolute inset-0 ${mixing && !mixDone ? "opacity-30" : ""} transition-opacity duration-500`}
   519→                >
   520→                  {bracket ? (
   521→                    <FitStage deps={[bracket, tournament.modality, status]} maxScale={1.05} minScale={0.3} className="w-full h-full">
   522→                      <BracketTree
   523→                        bracket={bracket}
   524→                        modality={tournament.modality}
   525→                        mode="viewer"
   526→                        schedule={schedule}
   527→                        showTimes={status === "live"}
   528→                        revealCount={mixing ? revealCount : undefined}
   529→                        pool={pool}
   530→                      />
   531→                    </FitStage>
   532→                  ) : (
   533→                    <div className="h-full flex items-center justify-center text-[#8e919c]">
   534→                      <Loader2 size={24} className="animate-spin" />
   535→                    </div>
   536→                  )}
   537→                </motion.div>
   538→              ) : null}
   539→            </>
   540→          )}
   541→        </AnimatePresence>
   542→
   543→        {/* overlay de mix */}
   544→        <AnimatePresence>
   545→          {mixing && bracket && mode === "brackets" ? (
   546→            <MixOverlay key="mix" locked={revealCount} total={totalSlots} pool={pool} done={mixDone} />
   547→          ) : null}
   548→        </AnimatePresence>
   549→
   550→        {/* podio */}
   551→        <AnimatePresence>
   552→          {status === "finished" && bracket && showPodium && mode === "brackets" ? (
   553→            <Podium key="podium" bracket={bracket} tournament={tournament} onClose={() => setShowPodium(false)} />
   554→          ) : null}
   555→        </AnimatePresence>
   556→
   557→        {/* botón para volver al podio */}
   558→        {status === "finished" && bracket && !showPodium && mode === "brackets" ? (
   559→          <button
   560→            type="button"
   561→            onClick={() => setShowPodium(true)}
   562→            className="btn-press absolute bottom-4 right-4 z-30 clip-btn red-badge px-4 py-2.5 text-[11px] font-extrabold uppercase tracking-[0.16em] flex items-center gap-2"
   563→          >
   564→            <Trophy size={13} />
   565→            Ver podio
   566→          </button>
   567→        ) : null}
   568→
   569→        {/* cartelera de horarios */}
   570→        <AnimatePresence>
   571→          {schedOpen && bracket ? (
   572→            <ScheduleOverlay
   573→              key="sched"
   574→              bracket={bracket}
   575→              schedule={schedule}
   576→              tournament={tournament}
   577→              open
   578→              onClose={() => setSchedOpen(false)}
   579→            />
   580→          ) : null}
   581→        </AnimatePresence>
   582→      </main>
   583→
   584→      {/* banner de espera durante mixing */}
   585→      <AnimatePresence>
   586→        {mixDone && status === "mixing" && mode === "brackets" ? (
   587→          <motion.div
   588→            initial={{ y: 40, opacity: 0 }}
   589→            animate={{ y: 0, opacity: 1 }}
   590→            exit={{ y: 40, opacity: 0 }}
   591→            className="relative z-20 shrink-0 border-t border-[#ffb830]/30 bg-[#171307]/90 flex items-center justify-center gap-2 py-2"
   592→          >
   593→            <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   594→            <span className="text-[10px] font-extrabold tracking-[0.28em] uppercase text-[#ffb830]">
   595→              Brackets listos — el administrador iniciará el torneo
   596→            </span>
   597→            <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   598→          </motion.div>
   599→        ) : null}
   600→      </AnimatePresence>
   601→
   602→      {/* pie: rail de próximos matches / ticker / barra vacía */}
   603→      {railActive && bracket ? (
   604→        <MatchRailFallback bracket={bracket} schedule={schedule} tournament={tournament} tickerItems={tickerItems} />
   605→      ) : status === "live" || status === "finished" ? (
   606→        <Ticker items={tickerItems} />
   607→      ) : (
   608→        <div className="shrink-0 h-10 border-t border-white/10 bg-black/70" />
   609→      )}
   610→    </div>
   611→  );
   612→}

   614→/* Si no hay matches programados con participantes definidos, cae al ticker */
   615→function MatchRailFallback({
   616→  bracket,
   617→  schedule,
   618→  tournament,
   619→  tickerItems,
   620→}: {
   621→  bracket: Bracket;
   622→  schedule: Map<string, number>;
   623→  tournament: Tournament;
   624→  tickerItems: string[];
   625→}) {
   626→  const hasAny = useMemo(() => hasRailMatches(bracket, schedule), [bracket, schedule]);
   627→
   628→  if (hasAny) return <MatchRail bracket={bracket} schedule={schedule} tournament={tournament} />;
   629→  return <Ticker items={tickerItems} />;
   630→}
