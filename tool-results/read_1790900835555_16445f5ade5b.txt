    72→/* ============================================================
    73→   Vista TORNEO EN VIVO — administración del bracket

    79→
    80→type AdminView = "tree" | "rounds";
    81→
    82→export function LiveAdminView({
    83→  tournament,
    84→  players,
    85→  bracket,
    86→  loadingBracket,
    87→}: {
    88→  tournament: Tournament | null;
    89→  players: Player[];
    90→  bracket: Bracket | null;
    91→  loadingBracket: boolean;
    92→}) {
    93→  const [sel, setSel] = useState<{ r: number; m: number } | null>(null);
    94→  const [pickBank, setPickBank] = useState<{ r: number; m: number; slotIdx: number } | null>(null);
    95→  const [confirm, setConfirm] = useState<null | { kind: "remix" | "finish" | "wipe" | "reopen" | "results" | "fill" | "demos" }>(null);
    96→  const [busy, setBusy] = useState(false);
    97→  const [view, setView] = useState<AdminView>("tree");
    98→  const [panelOpen, setPanelOpen] = useState(true);
    99→  const [zoom, setZoom] = useState<number | null>(null); // null = auto-fit
   100→  const [startMins, setStartMins] = useState(1);
   101→
   102→  const bank = useMemo(() => players.filter((p) => p.seat === "bank"), [players]);
   103→  const participants = useMemo(
   104→    () => (tournament ? groupParticipants(players, tournament.modality) : []),
   105→    [players, tournament]
   106→  );
   107→  const officialsCount = useMemo(() => players.filter((p) => p.seat === "off").length, [players]);
   108→  const demoCount = useMemo(() => players.filter((p) => p.demo).length, [players]);
   109→  const missingPlayers = tournament ? tournament.seats * tournament.modality - officialsCount : 0;
   110→  const schedule = useMemo(
   111→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map()),
   112→    [bracket, tournament]
   113→  );
   114→
   115→  /* navegación secuencial de matches (para editar rápido sin recorrer todo) */
   116→  const flatMatches = useMemo(() => {
   117→    if (!bracket) return [];
   118→    const out: { r: number; m: number }[] = [];
   119→    bracket.rounds.forEach((round, r) => round.forEach((_, m) => out.push({ r, m })));
   120→    return out;
   121→  }, [bracket]);
   122→  const flatIdx = useMemo(() => {
   123→    if (!sel || !bracket) return -1;
   124→    return flatMatches.findIndex((x) => x.r === sel.r && x.m === sel.m);
   125→  }, [flatMatches, sel, bracket]);
   126→
   127→  if (!tournament) {
   128→    return (
   129→      <div className="panel clip-card">
   130→        <EmptyState
   131→          icon={<Radio size={26} />}
   132→          title="Selecciona un torneo"
   133→          message="Elige un torneo en el selector superior para administrar el bracket en vivo: ganadores, descalificaciones y reemplazos del banco."
   134→        />
   135→      </div>
   136→    );
   137→  }
   138→
   139→  const guard = async (fn: () => Promise<void>) => {
   140→    setBusy(true);
   141→    try {
   142→      await fn();
   143→    } catch (e) {
   144→      toast.error(e instanceof Error ? e.message : "Error de operación");
   145→    } finally {
   146→      setBusy(false);
   147→    }
   148→  };
   149→
   150→  const selMatch = sel && bracket ? bracket.rounds[sel.r]?.[sel.m] : null;
   151→
   152→  const stepMatch = (dir: 1 | -1) => {
   153→    const nx = flatMatches[flatIdx + dir];
   154→    if (nx) setSel(nx);
   155→  };
   156→
   157→  const zoomStep = (dir: 1 | -1) => {
   158→    setZoom((z) => {
   159→      const base = z ?? 0.8;
   160→      return Math.min(1.8, Math.max(0.3, Math.round((base + dir * 0.1) * 100) / 100));
   161→    });
   162→  };
   163→
   164→  return (
   165→    <div className="flex flex-col gap-3 min-h-0">
   166→      {/* ============ barra de info del torneo ============ */}
   167→      <div className="panel clip-card p-4 flex flex-wrap items-center gap-3">
   168→        <div className="flex items-center gap-3 min-w-0 mr-auto">
   169→          <span className="slashes w-8 h-4 inline-block shrink-0" aria-hidden />
   170→          <div className="min-w-0">
   171→            <div className="font-display italic text-[17px] uppercase truncate">{tournament.name}</div>
   172→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   173→              <StatusChip status={tournament.status} />
   174→              <ModalityChip modality={tournament.modality} />
   175→              {bracket ? (
   176→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   177→                  {doneMatches(bracket)}/{totalMatches(bracket)} MATCHES
   178→                </span>
   179→              ) : (
   180→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   181→                  {officialsCount}/{tournament.seats * tournament.modality} JUGADORES
   182→                </span>
   183→              )}
   184→              {tournament.startAt ? (
   185→                <span className="chip clip-tag bg-[#ffb830]/12 text-[#ffb830] border border-[#ffb830]/35">
   186→                  <TimerReset size={10} />
   187→                  INICIA ≈ {fmtTime(tournament.startAt)}
   188→                </span>
   189→              ) : null}
   190→            </div>
   191→          </div>
   192→        </div>
   193→        <span className="text-[9px] font-extrabold uppercase tracking-[0.18em] text-[#6b6e78] hidden md:inline">
   194→          Cada fase se refleja automáticamente en el visor
   195→        </span>
   196→      </div>
   197→
   198→      {/* ============ FLUJO POR FASES ============ */}
   199→      <PhaseBar
   200→        tournament={tournament}
   201→        bracket={bracket}
   202→        participantsCount={participants.length}
   203→        officialsCount={officialsCount}
   204→        missingPlayers={missingPlayers}
   205→        demoCount={demoCount}
   206→        bankCount={bank.length}
   207→        startMins={startMins}
   208→        setStartMins={setStartMins}
   209→        busy={busy}
   210→        guard={guard}
   211→        onConfirm={(kind) => setConfirm({ kind })}
   212→        onLaunchMix={() =>
   213→          guard(async () => {
   214→            await launchMix(tournament.id, players, tournament.modality);
   215→            setSel(null);
   216→            toast.success("Mix match lanzado — el visor reproduce la animación");
   217→          })
   218→        }
   219→      />
   220→
   221→      {/* ============ cuerpo ============ */}
   222→      {loadingBracket ? (
   223→        <div className="panel clip-card p-14 flex items-center justify-center text-[#8e919c]">
   224→          <Loader2 size={20} className="animate-spin" />
   225→        </div>
   226→      ) : !bracket ? (
   227→        <div className="panel clip-card">
   228→          <EmptyState
   229→            icon={<Zap size={26} />}
   230→            title="Brackets no creados"
   231→            message={`Cuando estés listo, lanza el MIX MATCH: se randomizarán los ${participants.length} participantes y el visor de espectadores mostrará la animación de llenado de brackets.`}
   232→            action={
   233→              <div className="flex flex-wrap gap-2 justify-center">
   234→                <Btn
   235→                  variant="red"
   236→                  disabled={participants.length < 2}
   237→                  onClick={() =>
   238→                    guard(async () => {
   239→                      await launchMix(tournament.id, players, tournament.modality);
   240→                      toast.success("Mix match lanzado");
   241→                    })
   242→                  }
   243→                >
   244→                  <Rocket size={13} />
   245→                  Lanzar mix match
   246→                </Btn>
   247→                <Btn
   248→                  variant="dark"
   249→                  onClick={() =>
   250→                    guard(async () => {
   251→                      await setStartAt(tournament.id, Date.now() + startMins * 60_000);
   252→                      toast.success(`Inicio configurado en ${startMins} min — el visor muestra la cuenta regresiva`);
   253→                    })
   254→                  }
   255→                >
   256→                  <TimerReset size={13} />
   257→                  Inicio en {startMins} min
   258→                </Btn>
   259→              </div>
   260→            }
   261→          />
   262→          {/* vista previa de participantes */}
   263→          {participants.length > 0 ? (
   264→            <div className="px-5 pb-5">
   265→              <div className="label-cap mb-2 flex items-center gap-2">
   266→                <Users size={11} />
   267→                Participantes que entrarán al sorteo ({participants.length})
   268→              </div>
   269→              <div className="flex flex-wrap gap-1.5 max-h-36 overflow-y-auto">
   270→                {participants.map((p, i) => (
   271→                  <span key={p.pid} className="chip clip-tag bg-white/[0.05] text-[#c9cbd3] border border-white/10">
   272→                    {i + 1}. {p.label}
   273→                  </span>
   274→                ))}
   275→              </div>
   276→            </div>
   277→          ) : null}
   278→        </div>
   279→      ) : (
   280→        <>
   281→          {/* ============ toolbar de vistas ============ */}
   282→          <div className="panel clip-card-sm px-3 py-2 flex flex-wrap items-center gap-2">
   283→            <div className="flex gap-1 p-0.5 bg-black/40 border border-white/10" role="radiogroup" aria-label="Vista del bracket">
   284→              <button
   285→                type="button"
   286→                role="radio"
   287→                aria-checked={view === "tree"}
   288→                onClick={() => setView("tree")}
   289→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   290→                  view === "tree" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   291→                }`}
   292→              >
   293→                <LayoutPanelLeft size={12} />
   294→                Bracket
   295→              </button>
   296→              <button
   297→                type="button"
   298→                role="radio"
   299→                aria-checked={view === "rounds"}
   300→                onClick={() => setView("rounds")}
   301→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   302→                  view === "rounds" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   303→                }`}
   304→              >
   305→                <Columns3 size={12} />
   306→                Rondas
   307→              </button>
   308→            </div>
   309→
   310→            {selMatch ? (
   311→              <span className="chip clip-tag bg-[#e8102e]/15 text-[#ff8095] border border-[#e8102e]/40 text-[9px] flex items-center gap-1.5">
   312→                <Hand size={10} />
   313→                {matchTag(bracket, sel!.r, sel!.m)} seleccionado
   314→              </span>
   315→            ) : (
   316→              <span className="text-[9px] font-extrabold uppercase tracking-[0.16em] text-[#6b6e78] hidden sm:inline">
   317→                Clic en un match para administrarlo
   318→              </span>
   319→            )}
   320→
   321→            <div className="ml-auto flex items-center gap-1.5">
   322→              {/* navegación prev/next */}
   323→              <div className="flex items-center gap-1">
   324→                <IconBtn title="Match anterior" disabled={flatIdx <= 0} onClick={() => stepMatch(-1)}>
   325→                  <ChevronLeft size={14} />
   326→                </IconBtn>
   327→                <IconBtn title="Match siguiente" disabled={flatIdx < 0 || flatIdx >= flatMatches.length - 1} onClick={() => stepMatch(1)}>
   328→                  <ChevronRight size={14} />
   329→                </IconBtn>
   330→              </div>
   331→
   332→              {/* zoom (solo vista bracket) */}
   333→              {view === "tree" ? (
   334→                <div className="flex items-center gap-1 pl-2 border-l border-white/10">
   335→                  <IconBtn title="Alejar" onClick={() => zoomStep(-1)}>
   336→                    <ZoomOut size={14} />
   337→                  </IconBtn>
   338→                  <span className="text-[9px] font-extrabold tabular-nums text-[#8e919c] w-10 text-center uppercase">
   339→                    {zoom ? `${Math.round(zoom * 100)}%` : "AUTO"}
   340→                  </span>
   341→                  <IconBtn title="Acercar" onClick={() => zoomStep(1)}>
   342→                    <ZoomIn size={14} />
   343→                  </IconBtn>
   344→                  <IconBtn title="Ajustar a la pantalla" active={zoom === null} onClick={() => setZoom(null)}>
   345→                    <Maximize2 size={13} />
   346→                  </IconBtn>
   347→                </div>
   348→              ) : null}
   349→
   350→              {/* toggle panel lateral */}
   351→              <button
   352→                type="button"
   353→                onClick={() => setPanelOpen((v) => !v)}
   354→                title={panelOpen ? "Ocultar panel del match" : "Mostrar panel del match"}
   355→                aria-pressed={panelOpen}
   356→                className={`btn-press p-2 border transition-colors hidden xl:inline-flex ${
   357→                  panelOpen
   358→                    ? "border-[#e8102e]/60 bg-[#e8102e]/10 text-white"
   359→                    : "border-white/12 text-[#8e919c] hover:text-white"
   360→                }`}
   361→              >
   362→                {panelOpen ? <PanelRightClose size={14} /> : <PanelRightOpen size={14} />}
   363→              </button>
   364→            </div>
   365→          </div>
   366→
   367→          {/* ============ stage + panel lateral colapsable ============ */}
   368→          <div className="flex gap-3 items-stretch min-h-0">
   369→            {/* área principal */}
   370→            <div className="flex-1 min-w-0 min-h-0">
   371→              <div className="panel clip-card p-3 min-h-[420px] h-[calc(100vh-330px)] max-h-[80vh] flex flex-col">
   372→                <div className="flex items-center justify-between px-2 pb-2">
   373→                  <span className="label-cap">
   374→                    {view === "tree" ? "Bracket · clic en un match para administrar" : "Rondas · cards seleccionables con botón de ganador directo"}
   375→                  </span>
   376→                  <span className="label-cap">{MODALITY_LABEL[tournament.modality]} · a {tournament.winsNeeded} wins</span>
   377→                </div>
   378→
   379→                {view === "tree" ? (
   380→                  <div className="flex-1 min-h-0 border border-white/8 bg-black/30 clip-card-sm">
   381→                    <FitStage deps={[bracket, tournament.modality, zoom]} maxScale={0.95} minScale={0.4} zoom={zoom} className="w-full h-full">
   382→                      <BracketTree
   383→                        bracket={bracket}
   384→                        modality={tournament.modality}
   385→                        mode="admin"
   386→                        selectedId={selMatch?.id ?? null}
   387→                        onSelectMatch={(r, m) => setSel({ r, m })}
   388→                        schedule={schedule}
   389→                        showTimes
   390→                      />
   391→                    </FitStage>
   392→                  </div>
   393→                ) : (
   394→                  <RoundColumns
   395→                    bracket={bracket}
   396→                    times={schedule}
   397→                    selectedId={selMatch?.id ?? null}
   398→                    onSelectMatch={(r, m) => setSel({ r, m })}
   399→                    onPickWinner={(r, m, slotIdx) =>
   400→                      guard(async () => {
   401→                        await setWinner(tournament.id, r, m, slotIdx);
   402→                        setSel({ r, m });
   403→                        toast.success(`${bracket.rounds[r][m].slots[slotIdx].label} avanza`);
   404→                      })
   405→                    }
   406→                  />
   407→                )}
   408→              </div>
   409→            </div>
   410→
   411→            {/* panel lateral (desktop, colapsable) */}
   412→            <aside
   413→              className={`hidden xl:block shrink-0 overflow-hidden transition-[width] duration-300 ease-out ${
   414→                panelOpen ? "w-[376px]" : "w-0"
   415→              }`}
   416→              aria-hidden={!panelOpen}
   417→            >
   418→              <div className="w-[376px] h-full">
   419→                <MatchPanel
   420→                  tournament={tournament}
   421→                  bracket={bracket}
   422→                  sel={sel}
   423→                  setSel={setSel}
   424→                  selMatch={selMatch}
   425→                  schedule={schedule}
   426→                  bank={bank}
   427→                  onClose={() => setPanelOpen(false)}
   428→                  guard={guard}
   429→                  setPickBank={setPickBank}
   430→                  busy={busy}
   431→                  stepMatch={stepMatch}
   432→                />
   433→              </div>
   434→            </aside>
   435→          </div>
   436→
   437→          {/* panel del match (móvil/tablet: debajo) */}
   438→          <div className="xl:hidden">
   439→            <MatchPanel
   440→              tournament={tournament}
   441→              bracket={bracket}
   442→              sel={sel}
   443→              setSel={setSel}
   444→              selMatch={selMatch}
   445→              schedule={schedule}
   446→              bank={bank}
   447→              onClose={() => setSel(null)}
   448→              guard={guard}
   449→              setPickBank={setPickBank}
   450→              busy={busy}
   451→              stepMatch={stepMatch}
   452→            />
   453→          </div>
   454→
   455→          {/* pestaña flotante para reabrir el panel en desktop */}
   456→          {!panelOpen && selMatch ? (
   457→            <button
   458→              type="button"
   459→              onClick={() => setPanelOpen(true)}
   460→              className="btn-press fixed right-0 top-1/2 -translate-y-1/2 z-40 hidden xl:flex flex-col items-center gap-2 red-badge clip-badge-l px-2 py-4 text-[10px] font-extrabold uppercase tracking-[0.2em] shadow-[0_10px_30px_rgba(232,16,46,0.4)]"
   461→              style={{ writingMode: "vertical-rl" }}
   462→            >
   463→              <ChevronLeft size={12} />
   464→              Panel del match
   465→            </button>
   466→          ) : null}
   467→        </>
   468→      )}
   469→
   470→      {/* ============ modal banco ============ */}
   471→      <Modal open={!!pickBank} onClose={() => setPickBank(null)} title="Reemplazar desde el banco">
   472→        <p className="text-[11px] font-bold text-[#8e919c] uppercase tracking-wider mb-3">
   473→          Selecciona la reserva que entrará al match
   474→        </p>
   475→        <div className="space-y-2 max-h-72 overflow-y-auto">
   476→          {bank.length === 0 ? (
   477→            <p className="text-[11px] font-bold text-[#6b6e78] uppercase">No hay jugadores en el banco</p>
   478→          ) : (
   479→            bank.map((p) => (
   480→              <button
   481→                key={p.id}
   482→                type="button"
   483→                onClick={async () => {
   484→                  if (!pickBank || !tournament) return;
   485→                  await guard(async () => {
   486→                    await replaceFromBank(tournament.id, pickBank.r, pickBank.m, pickBank.slotIdx, p, tournament.modality);
   487→                    toast.success(`${p.nick} entra al match`);
   488→                    setPickBank(null);
   489→                  });
   490→                }}
   491→                className="w-full panel-2 clip-card-sm flex items-center gap-3 px-3 py-2.5 hover:border-[#ffb830]/50 text-left transition-colors"
   492→              >
   493→                <span className="w-10 h-7 clip-badge red-badge flex items-center justify-center">
   494→                  <SkipForward size={12} />
   495→                </span>
   496→                <span className="min-w-0 flex-1">
   497→                  <span className="block text-[13px] font-extrabold uppercase text-white truncate">{p.nick}</span>
   498→                  <span className="block text-[10px] font-semibold text-[#8e919c] truncate">{p.name}</span>
   499→                </span>
   500→              </button>
   501→            ))
   502→          )}
   503→        </div>
   504→      </Modal>
   505→
   506→      {/* ============ confirmaciones de flujo ============ */}
   507→      <Confirm
   508→        open={confirm?.kind === "remix"}
   509→        onClose={() => setConfirm(null)}
   510→        danger
   511→        confirmLabel="Lanzar mix"
   512→        title="Lanzar mix — reinicio completo"
   513→        message="Se borran por completo los resultados anteriores (marcadores, ganadores, descalificaciones) y se crean brackets 100% nuevos con los participantes randomizados. El visor reproducirá la animación de mix desde cero."
   514→        onConfirm={() =>
   515→          tournament &&
   516→          guard(async () => {
   517→            await launchMix(tournament.id, players, tournament.modality);
   518→            setSel(null);
   519→            toast.success("Mix lanzado — brackets nuevos y limpios");
   520→          })
   521→        }
   522→      />
   523→      <Confirm
   524→        open={confirm?.kind === "finish"}
   525→        onClose={() => setConfirm(null)}
   526→        confirmLabel="Finalizar"
   527→        title="Finalizar torneo"
   528→        message="El visor de espectadores mostrará el podio con el campeón. Podrás reabrir el torneo si necesitas corregir algo."
   529→        onConfirm={() =>
   530→          tournament &&
   531→          guard(async () => {

   871→function MiniCountdown({ target }: { target: number }) {
   872→  const [now, setNow] = useState(() => Date.now());
   873→  useEffect(() => {
   874→    const t = setInterval(() => setNow(Date.now()), 500);
   875→    return () => clearInterval(t);
   876→  }, []);
   877→  const total = Math.max(0, Math.floor((target - now) / 1000));
   878→  const h = Math.floor(total / 3600);
   879→  const m = Math.floor((total % 3600) / 60);
   880→  const s = total % 60;
   881→  const pad = (x: number) => String(x).padStart(2, "0");
   882→  return (
   883→    <span className="font-display italic text-[20px] leading-none text-silver-grad tabular-nums">
   884→      {h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`}
   885→    </span>
   886→  );
   887→}

  1007→              small
  1008→              variant="dark"
