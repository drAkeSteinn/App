    91→
    92→export function LiveAdminView({
    93→  tournament,
    94→  players,
    95→  bracket,
    96→  loadingBracket,
    97→}: {
    98→  tournament: Tournament | null;
    99→  players: Player[];
   100→  bracket: Bracket | null;
   101→  loadingBracket: boolean;
   102→}) {
   103→  const [sel, setSel] = useState<{ r: number; m: number } | null>(null);
   104→  const [pickBank, setPickBank] = useState<{ r: number; m: number; slotIdx: number } | null>(null);
   105→  const [confirm, setConfirm] = useState<null | { kind: "remix" | "finish" | "wipe" | "reopen" | "results" | "fill" | "demos" }>(null);
   106→  const [busy, setBusy] = useState(false);
   107→  const [view, setView] = useState<AdminView>("tree");
   108→  const [panelOpen, setPanelOpen] = useState(true);
   109→  const [zoom, setZoom] = useState<number | null>(null); // null = auto-fit
   110→  const [startMins, setStartMins] = useState(1);
   111→  const [forceModal, setForceModal] = useState<{ blockingName: string } | null>(null);
   112→  const { data: allTournaments } = useTournaments();
   113→  const { data: allContests } = useContests();
   114→
   115→  /* banco efectivo: reservas del registro antes de la primera eliminatoria;
   116→     después, los eliminados de la última ronda completada (banco rotativo) */
   117→  const bankInfo = useMemo(() => computeBank(bracket, players), [bracket, players]);
   118→  const arenaState = useArenaState(tournament?.id ?? null);
   119→  const schedOpen = arenaState?.schedOpen ?? false;
   120→  /** Emite la señal de sonido al visor. NO reproduce localmente en el admin
   121→      para evitar duplicación — el visor es el único que debe sonar. */
   122→  const fireSound = (event: SoundEvent) => {
   123→    if (tournament) emitSound(tournament.id, event).catch(() => {});
   124→  };
   125→  /* Torneo activo para transmisión (visor + OBS). Solo uno a la vez. */
   126→  const active = useActiveTournament();
   127→  const isActive = !!tournament && active.tid === tournament.id;
   128→  const anotherActive = !!active.tid && active.tid !== tournament?.id;
   129→  const activeTournamentName = active.tournament?.name ?? null;
   130→  const participants = useMemo(
   131→    () => (tournament ? groupParticipants(players, tournament.modality) : []),
   132→    [players, tournament]
   133→  );
   134→  const officialsCount = useMemo(() => players.filter((p) => p.seat === "off").length, [players]);
   135→  const demoCount = useMemo(() => players.filter((p) => p.demo).length, [players]);
   136→  const missingPlayers = tournament ? playersCapacity(tournament) - officialsCount : 0;
   137→  const schedule = useMemo(
   138→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map()),
   139→    [bracket, tournament]
   140→  );
   141→  /* capacidad del escenario: matches simultáneos según las 4 PCs */
   142→  const maxSim = tournament ? maxSimultaneous(tournament.modality) : 1;
   143→  const liveCount = useMemo(
   144→    () => (bracket ? bracket.rounds.reduce((acc, round) => acc + round.filter((x) => x.status === "live").length, 0) : 0),
   145→    [bracket]
   146→  );
   147→
   148→  /* navegación secuencial de matches (para editar rápido sin recorrer todo) */
   149→  const flatMatches = useMemo(() => {
   150→    if (!bracket) return [];
   151→    const out: { r: number; m: number }[] = [];
   152→    bracket.rounds.forEach((round, r) => round.forEach((_, m) => out.push({ r, m })));
   153→    if (bracket.thirdPlace) out.push({ r: -1, m: 0 });
   154→    return out;
   155→  }, [bracket]);
   156→  const flatIdx = useMemo(() => {
   157→    if (!sel || !bracket) return -1;
   158→    return flatMatches.findIndex((x) => x.r === sel.r && x.m === sel.m);
   159→  }, [flatMatches, sel, bracket]);
   160→
   161→  if (!tournament) {
   162→    return (
   163→      <div className="panel clip-card">
   164→        <EmptyState
   165→          icon={<Radio size={26} />}
   166→          title="Selecciona un torneo"
   167→          message="Elige un torneo en el selector superior para administrar el bracket en vivo: ganadores, descalificaciones y reemplazos del banco."
   168→        />
   169→      </div>
   170→    );
   171→  }
   172→
   173→  const guard = async (fn: () => Promise<void>) => {
   174→    setBusy(true);
   175→    try {
   176→      await fn();
   177→    } catch (e) {
   178→      toast.error(e instanceof Error ? e.message : "Error de operación");
   179→    } finally {
   180→      setBusy(false);
   181→    }
   182→  };
   183→
   184→  const selMatch = sel && bracket ? (sel.r === -1 ? bracket.thirdPlace : bracket.rounds[sel.r]?.[sel.m]) ?? null : null;
   185→
   186→  const stepMatch = (dir: 1 | -1) => {
   187→    const nx = flatMatches[flatIdx + dir];
   188→    if (nx) setSel(nx);
   189→  };
   190→
   191→  const zoomStep = (dir: 1 | -1) => {
   192→    setZoom((z) => {
   193→      const base = z ?? 0.8;
   194→      return Math.min(1.8, Math.max(0.3, Math.round((base + dir * 0.1) * 100) / 100));
   195→    });
   196→  };
   197→
   198→  return (
   199→    <div className="flex flex-col gap-3 min-h-0">
   200→      {/* ============ barra de info del torneo ============ */}
   201→      <div className="panel clip-card p-4 flex flex-wrap items-center gap-3">
   202→        <div className="flex items-center gap-3 min-w-0 mr-auto">
   203→          <span className="slashes w-8 h-4 inline-block shrink-0" aria-hidden />
   204→          <div className="min-w-0">
   205→            <div className="font-display italic text-[17px] uppercase truncate">{tournament.name}</div>
   206→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   207→              <StatusChip status={tournament.status} />
   208→              <ModalityChip modality={tournament.modality} />
   209→              {bracket ? (
   210→                <span
   211→                  className={`chip clip-tag border ${
   212→                    liveCount >= maxSim
   213→                      ? "bg-[#e8102e]/15 text-[#ff8095] border-[#e8102e]/40"
   214→                      : "bg-[#e8102e]/12 text-[#ff8095] border-[#e8102e]/35"
   215→                  }`}
   216→                  title="Matches en juego / máximo simultáneo según las PCs del escenario (4 PCs)"
   217→                >
   218→                  <Monitor size={10} />
   219→                  EN JUEGO {liveCount}/{maxSim}
   220→                </span>
   221→              ) : null}
   222→              {bracket ? (
   223→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   224→                  {doneMatches(bracket)}/{totalMatches(bracket)} MATCHES
   225→                </span>
   226→              ) : (
   227→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   228→                  {officialsCount}/{playersCapacity(tournament)} JUGADORES
   229→                </span>
   230→              )}
   231→              {tournament.startAt ? (
   232→                <span className="chip clip-tag bg-[#ffb830]/12 text-[#ffb830] border border-[#ffb830]/35">
   233→                  <TimerReset size={10} />
   234→                  INICIA ≈ {fmtTime(tournament.startAt)}
   235→                </span>
   236→              ) : null}
   237→              {isActive ? (
   238→                <span className="chip clip-tag bg-[#e8102e] text-white text-[9px] blink">
   239→                  <span className="w-1.5 h-1.5 bg-white rounded-full" aria-hidden />
   240→                  EN TRANSMISIÓN
   241→                </span>
   242→              ) : null}
   243→            </div>
   244→          </div>
   245→        </div>
   246→        {/* control de transmisión: Abrir / Cerrar torneo para el visor + OBS */}
   247→        <div className="flex items-center gap-2 shrink-0">
   248→          {isActive ? (
   249→            <Btn
   250→              small
   251→              variant="dark"
   252→              onClick={() =>
   253→                guard(async () => {
   254→                  await closeTournament();
   255→                  toast.success("Torneo cerrado — el visor y las cards de OBS dejan de transmitir");
   256→                })
   257→              }
   258→              title="Cerrar torneo: el visor y las cards de OBS dejan de mostrar este torneo"
   259→            >
   260→              <Lock size={12} />
   261→              Cerrar torneo
   262→            </Btn>
   263→          ) : (
   264→            <Btn
   265→              small
   266→              variant={anotherActive ? "dark" : "red"}
   267→              title={
   268→                anotherActive
   269→                  ? `Ya hay otra transmisión activa: ${activeTournamentName}. Clic para forzar.`
   270→                  : "En vivo: el visor y las cards de OBS mostrarán este torneo"
   271→              }
   272→              onClick={() =>
   273→                guard(async () => {
   274→                  if (!tournament) return;
   275→                  /* verificar si hay otra transmisión activa */
   276→                  const info = await getActiveInfo();
   277→                  const hasOther = (info.tid && info.tid !== tournament.id) || info.contestId;
   278→                  if (hasOther) {
   279→                    /* determinar el nombre del bloqueante */
   280→                    let blockingName = "otra transmisión";
   281→                    if (info.tid) {
   282→                      const t = allTournaments.find((x) => x.id === info.tid);
   283→                      blockingName = t?.name ?? "un torneo";
   284→                    } else if (info.contestId) {
   285→                      const c = allContests.find((x) => x.id === info.contestId);
   286→                      blockingName = c?.name ?? "un concurso";
   287→                    }
   288→                    setForceModal({ blockingName });
   289→                    return;
   290→                  }
   291→                  await openTournament(tournament.id, tournament.name);
   292→                  toast.success("Torneo en vivo — el visor y las cards de OBS lo muestran");
   293→                })
   294→              }
   295→            >
   296→              <Radio size={12} />
   297→              {anotherActive ? `Ocupado: ${activeTournamentName}` : "En vivo"}
   298→            </Btn>
   299→          )}
   300→        </div>
   301→      </div>
   302→
   303→      {/* ============ FLUJO POR FASES ============ */}
   304→      <PhaseBar
   305→        tournament={tournament}
   306→        bracket={bracket}
   307→        participantsCount={participants.length}
   308→        officialsCount={officialsCount}
   309→        missingPlayers={missingPlayers}
   310→        demoCount={demoCount}
   311→        bankCount={bankInfo.entries.length}
   312→        startMins={startMins}
   313→        setStartMins={setStartMins}
   314→        busy={busy}
   315→        guard={guard}
   316→        onConfirm={(kind) => setConfirm({ kind })}
   317→        onFireSound={fireSound}
   318→        onLaunchMix={() =>
   319→          guard(async () => {
   320→            await launchMix(tournament.id, players, tournament);
   321→            setSel(null);
   322→            fireSound("mixLaunch");
   323→            toast.success("Mix match lanzado — el visor reproduce la animación");
   324→          })
   325→        }
   326→      />
   327→
   328→      {/* ============ cuerpo ============ */}
   329→      {loadingBracket ? (
   330→        <div className="panel clip-card p-14 flex items-center justify-center text-[#8e919c]">
   331→          <Loader2 size={20} className="animate-spin" />
   332→        </div>
   333→      ) : !bracket ? (
   334→        <div className="panel clip-card">
   335→          <EmptyState
   336→            icon={<Zap size={26} />}
   337→            title="Brackets no creados"
   338→            message={`Cuando estés listo, lanza el MIX MATCH: se randomizarán los ${participants.length} participantes y el visor de espectadores mostrará la animación de llenado de brackets.`}
   339→            action={
   340→              <div className="flex flex-wrap gap-2 justify-center">
   341→                <Btn
   342→                  variant="red"
   343→                  disabled={participants.length < 2}
   344→                  onClick={() =>
   345→                    guard(async () => {
   346→                      await launchMix(tournament.id, players, tournament);
   347→                      fireSound("mixLaunch");
   348→                      toast.success("Mix match lanzado");
   349→                    })
   350→                  }
   351→                >
   352→                  <Rocket size={13} />
   353→                  Lanzar mix match
   354→                </Btn>
   355→                <Btn
   356→                  variant="dark"
   357→                  onClick={() =>
   358→                    guard(async () => {
   359→                      await setStartAt(tournament.id, Date.now() + startMins * 60_000);
   360→                      toast.success(`Inicio configurado en ${startMins} min — el visor cambia a brackets con el cronómetro`);
   361→                    })
   362→                  }
   363→                >
   364→                  <TimerReset size={13} />
   365→                  Inicio en {startMins} min
   366→                </Btn>
   367→              </div>
   368→            }
   369→          />
   370→          {/* vista previa de participantes */}
   371→          {participants.length > 0 ? (
   372→            <div className="px-5 pb-5">
   373→              <div className="label-cap mb-2 flex items-center gap-2">
   374→                <Users size={11} />
   375→                Participantes que entrarán al sorteo ({participants.length})
   376→              </div>
   377→              <div className="flex flex-wrap gap-1.5 max-h-36 overflow-y-auto">
   378→                {participants.map((p, i) => (
   379→                  <span key={p.pid} className="chip clip-tag bg-white/[0.05] text-[#c9cbd3] border border-white/10">
   380→                    {i + 1}. {p.label}
   381→                  </span>
   382→                ))}
   383→              </div>
   384→            </div>
   385→          ) : null}
   386→        </div>
   387→      ) : (
   388→        <>
   389→          {/* ============ toolbar de vistas ============ */}
   390→          <div className="panel clip-card-sm px-3 py-2 flex flex-wrap items-center gap-2">
   391→            <div className="flex gap-1 p-0.5 bg-black/40 border border-white/10" role="radiogroup" aria-label="Vista del bracket">
   392→              <button
   393→                type="button"
   394→                role="radio"
   395→                aria-checked={view === "tree"}
   396→                onClick={() => setView("tree")}
   397→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   398→                  view === "tree" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   399→                }`}
   400→              >
   401→                <LayoutPanelLeft size={12} />
   402→                Bracket
   403→              </button>
   404→              <button
   405→                type="button"
   406→                role="radio"
   407→                aria-checked={view === "rounds"}
   408→                onClick={() => setView("rounds")}
   409→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   410→                  view === "rounds" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   411→                }`}
   412→              >
   413→                <Columns3 size={12} />
   414→                Rondas
   415→              </button>
   416→            </div>
   417→
   418→            {selMatch ? (
   419→              <span className="chip clip-tag bg-[#e8102e]/15 text-[#ff8095] border border-[#e8102e]/40 text-[9px] flex items-center gap-1.5">
   420→                <Hand size={10} />
   421→                {matchTag(bracket, sel!.r, sel!.m)} seleccionado
   422→              </span>
   423→            ) : (
   424→              <span className="text-[9px] font-extrabold uppercase tracking-[0.16em] text-[#6b6e78] hidden sm:inline">
   425→                Clic en un match para administrarlo
   426→              </span>
   427→            )}
   428→
   429→            <div className="ml-auto flex items-center gap-1.5">
   430→              {/* navegación prev/next */}
   431→              <div className="flex items-center gap-1">
   432→                <IconBtn title="Match anterior" disabled={flatIdx <= 0} onClick={() => stepMatch(-1)}>
   433→                  <ChevronLeft size={14} />
   434→                </IconBtn>
   435→                <IconBtn title="Match siguiente" disabled={flatIdx < 0 || flatIdx >= flatMatches.length - 1} onClick={() => stepMatch(1)}>
   436→                  <ChevronRight size={14} />
   437→                </IconBtn>
   438→              </div>
   439→
   440→              {/* zoom (solo vista bracket) */}
   441→              {view === "tree" ? (
   442→                <div className="flex items-center gap-1 pl-2 border-l border-white/10">
   443→                  <IconBtn title="Alejar" onClick={() => zoomStep(-1)}>
   444→                    <ZoomOut size={14} />
   445→                  </IconBtn>
   446→                  <span className="text-[9px] font-extrabold tabular-nums text-[#8e919c] w-10 text-center uppercase">
   447→                    {zoom ? `${Math.round(zoom * 100)}%` : "AUTO"}
   448→                  </span>
   449→                  <IconBtn title="Acercar" onClick={() => zoomStep(1)}>
   450→                    <ZoomIn size={14} />
   451→                  </IconBtn>
   452→                  <IconBtn title="Ajustar a la pantalla" active={zoom === null} onClick={() => setZoom(null)}>
   453→                    <Maximize2 size={13} />
   454→                  </IconBtn>
   455→                </div>
   456→              ) : null}
   457→
   458→              {/* toggle cartelera en el visor (proyectado) */}
   459→              <button
   460→                type="button"
   461→                onClick={() => {
   462→                  if (!tournament) return;
   463→                  setSchedOpen(tournament.id, !schedOpen).catch(() => {});
   464→                }}
   465→                title={schedOpen ? "Ocultar cartelera en el visor" : "Mostrar cartelera de horarios en el visor"}
   466→                aria-pressed={schedOpen}
   467→                className={`btn-press clip-tag px-2.5 py-2 text-[10px] font-extrabold uppercase tracking-[0.1em] flex items-center gap-1.5 border transition-colors ${
   468→                  schedOpen
   469→                    ? "border-[#ffb830]/60 bg-[#ffb830]/12 text-[#ffb830]"
   470→                    : "border-white/12 text-[#8e919c] hover:text-white hover:border-[#ffb830]/50"
   471→                }`}
   472→              >
   473→                <CalendarClock size={13} />
   474→                <span className="hidden md:inline">Cartelera</span>
   475→              </button>
   476→
   477→              {/* editar posición de cards de OBS (abre el editor draggable) */}
   478→              <button
   479→                type="button"
   480→                onClick={() => window.open("/?obs=1&edit=1", "_blank", "noopener")}
   481→                title="Arrastrar las cards de OBS (PC1-PC4) a posiciones personalizadas — se guarda para todos los torneos"
   482→                className="btn-press clip-tag px-2.5 py-2 text-[10px] font-extrabold uppercase tracking-[0.1em] flex items-center gap-1.5 border border-white/12 text-[#8e919c] hover:text-white hover:border-[#e8102e]/60 transition-colors"
   483→              >
   484→                <MonitorPlay size={13} />
   485→                <span className="hidden md:inline">Editar cards OBS</span>
   486→              </button>
   487→
   488→              {/* toggle panel lateral */}
   489→              <button
   490→                type="button"
   491→                onClick={() => setPanelOpen((v) => !v)}
   492→                title={panelOpen ? "Ocultar panel del match" : "Mostrar panel del match"}
   493→                aria-pressed={panelOpen}
   494→                className={`btn-press p-2 border transition-colors hidden xl:inline-flex ${
   495→                  panelOpen
   496→                    ? "border-[#e8102e]/60 bg-[#e8102e]/10 text-white"
   497→                    : "border-white/12 text-[#8e919c] hover:text-white"
   498→                }`}
   499→              >
   500→                {panelOpen ? <PanelRightClose size={14} /> : <PanelRightOpen size={14} />}
   501→              </button>
   502→            </div>
   503→          </div>
   504→
   505→          {/* ============ stage + panel lateral colapsable ============ */}
   506→          <div className="flex gap-3 items-stretch min-h-0">
   507→            {/* área principal */}
   508→            <div className="flex-1 min-w-0 min-h-0">
