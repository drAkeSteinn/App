    45→    45→/* Detecta el primer evento "dramático" entre dos versiones del bracket:
    46→    46→   match que pasa a DONE (ganador marcado o cambiado) o a LIVE (en juego).
    47→    47→   Las PCs del match se leen del match actual (al poner en juego ya tienen
    48→    48→   PCs asignadas) o del snapshot anterior (al definir ganador ya se liberaron). */
    49→    49→function findSpotlightEvent(
    50→    50→  cur: Bracket,
    51→    51→  prev: Bracket,
    52→    52→  winsNeeded: number,
    53→    53→  modality: Modality
    54→    54→): SpotlightData | null {
    55→    55→  for (let r = 0; r < cur.rounds.length; r++) {
    56→    56→    const round = cur.rounds[r];
    57→    57→    for (let m = 0; m < round.length; m++) {
    58→    58→      const match = round[m];
    59→    59→      const old = prev.rounds[r]?.[m];
    60→    60→      if (!match || !old || match.bye) continue;
    61→    61→      const becameDone = match.status === "done" && old.status !== "done";
    62→    62→      const winnerChanged = match.status === "done" && old.status === "done" && match.w !== old.w;
    63→    63→      const becameLive = match.status === "live" && old.status !== "live";
    64→    64→      if (becameDone || winnerChanged || becameLive) {
    65→    65→        const pcs = match.pcs?.length ? match.pcs : old.pcs;
    66→    66→        return {
    67→    67→          id: `${match.id}-${match.status}-${match.w}-${Date.now()}`,
    68→    68→          kind: match.status === "live" ? "live" : "win",
    69→    69→          tag: matchTag(cur, r, m),
    70→    70→          players: match.slots.map((s, i) => ({
    71→    71→            name: s.label || "—",
    72→    72→            score: s.score ?? 0,
    73→    73→            pcs: pcsForSide(modality, pcs, i),
    74→    74→          })),
    75→    75→          winner: match.w,
    76→    76→          winsNeeded,
    77→    77→        };
    78→    78→      }
    79→    79→    }
    80→    80→  }
    81→    81→  return null;
    82→    82→}

   320→   320→export function SpectatorView({ tid }: { tid: string | null }) {
   321→   321→  const { data: tournaments, loading, error } = useTournaments();
   322→   322→  const tournament: Tournament | null = useMemo(
   323→   323→    () => tournaments.find((t) => t.id === tid) ?? null,
   324→   324→    [tournaments, tid]
   325→   325→  );
   326→   326→  const { data: players } = usePlayers(tournament ? tid : null);
   327→   327→  const { data: bracket } = useBracket(tournament ? tid : null);
   328→   328→
   329→   329→  const status = tournament?.status ?? "open";
   330→   330→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   331→   331→  const [schedOpen, setSchedOpen] = useState(false);
   332→   332→  const [revealCount, setRevealCount] = useState(0);
   333→   333→  const [showPodium, setShowPodium] = useState(true);
   334→   334→  const [spotlight, setSpotlight] = useState<SpotlightData | null>(null);
   335→   335→  const prevBracketRef = useRef<{ tid: string | null; bracket: Bracket } | null>(null);
   336→   336→  const rootRef = useRef<HTMLDivElement | null>(null);
   337→   337→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   338→   338→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   339→   339→
   340→   340→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   341→   341→     se refleja sola en el visor). El toggle manual gana hasta que el status
   342→   342→     vuelva a cambiar, porque la anulación queda atada al status vigente.
   343→   343→     CUENTA REGRESIVA: en cuanto el admin programa el inicio (startAt), el
   344→   344→     visor salta solo a la pestaña de brackets, donde el cronómetro de inicio
   345→   345→     se ve en grande; si cancela la cuenta regresiva regresa al modo registro. */
   346→   346→  const inRegisterPhase = status === "open" || status === "closed";
   347→   347→  const autoMode: ViewerMode = inRegisterPhase ? (tournament?.startAt ? "brackets" : "reg") : "brackets";
   348→   348→  const mode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   349→   349→
   350→   350→  /* sincroniza el modo en la URL para compartir la vista correcta */
   351→   351→  useEffect(() => {
   352→   352→    try {
   353→   353→      const url = new URL(window.location.href);
   354→   354→      if (mode === "reg") url.searchParams.set("mode", "reg");
   355→   355→      else url.searchParams.delete("mode");
   356→   356→      if (url.toString() !== window.location.href) window.history.replaceState(null, "", url.toString());
   357→   357→    } catch {
   358→   358→      /* ignore */
   359→   359→    }
   360→   360→  }, [mode]);
   361→   361→
   362→   362→  const changeMode = (m: ViewerMode) => {
   363→   363→    setOverride({ status, mode: m });
   364→   364→  };
   365→   365→
   366→   366→  // animación de mix
   367→   367→  const totalSlots = bracket ? bracket.rounds[0].length * (bracket.s ?? 2) : 0;
   368→   368→  const pool = useMemo(() => {
   369→   369→    if (!bracket) return [];
   370→   370→    const labels = bracket.rounds[0].flatMap((m) => m.slots.map((s) => s.label || "TBD"));
   371→   371→    return Array.from(new Set(labels));
   372→   372→  }, [bracket]);
   373→   373→
   374→   374→  useEffect(() => {
   375→   375→    if (status !== "mixing" || !bracket || mode !== "brackets") return;
   376→   376→    const r0 = requestAnimationFrame(() => {
   377→   377→      setRevealCount(0);
   378→   378→      setShowPodium(false);
   379→   379→    });
   380→   380→    toRef.current = setTimeout(() => {
   381→   381→      ivRef.current = setInterval(() => {
   382→   382→        setRevealCount((c) => {
   383→   383→          if (c >= totalSlots) {
   384→   384→            if (ivRef.current) clearInterval(ivRef.current);
   385→   385→            return c;
   386→   386→          }
   387→   387→          return c + 1;
   388→   388→        });
   389→   389→      }, 240);
   390→   390→    }, 2400);
   391→   391→    return () => {
   392→   392→      cancelAnimationFrame(r0);
   393→   393→      if (toRef.current) clearTimeout(toRef.current);
   394→   394→      if (ivRef.current) clearInterval(ivRef.current);
   395→   395→    };
   396→   396→  }, [status, bracket, totalSlots, mode]);
   397→   397→
   398→   398→  useEffect(() => {
   399→   399→    if (status === "finished") {
   400→   400→      const r = requestAnimationFrame(() => setShowPodium(true));
   401→   401→      return () => cancelAnimationFrame(r);
   402→   402→    }
   403→   403→  }, [status]);
   404→   404→
   405→   405→  /* ----- SPOTLIGHT: ganador marcado / match en juego → corte dramático.
   406→   406→     Compara cada snapshot del bracket con el anterior; al detectar un match
   407→   407→     que pasa a DONE (o cambia de ganador) o a LIVE, lanza el overlay. ----- */
   408→   408→  useEffect(() => {
   409→   409→    const prev = prevBracketRef.current;

   751→   751→/* Si no hay matches programados con participantes definidos, cae al ticker */

   753→   753→  bracket,
   754→   754→  schedule,
   755→   755→  tournament,
   756→   756→  tickerItems,
   757→   757→}: {
   758→   758→  bracket: Bracket;
   759→   759→  schedule: Map<string, number>;
   760→   760→  tournament: Tournament;
   761→   761→  tickerItems: string[];
   762→   762→}) {
   763→   763→  const hasAny = useMemo(() => hasRailMatches(bracket, schedule), [bracket, schedule]);
   764→   764→
   765→   765→  if (hasAny) return <MatchRail bracket={bracket} schedule={schedule} tournament={tournament} />;
   766→   766→  return <Ticker items={tickerItems} />;
   767→   767→}

   771→   771→}
