    65→    65→function useClock() {
    66→    66→  const [now, setNow] = useState<Date | null>(null);
    67→    67→  useEffect(() => {
    68→    68→    const first = requestAnimationFrame(() => setNow(new Date()));
    69→    69→    const t = setInterval(() => setNow(new Date()), 1000);
    70→    70→    return () => {
    71→    71→      cancelAnimationFrame(first);
    72→    72→      clearInterval(t);
    73→    73→    };
    74→    74→  }, []);
    75→    75→  return now;
    76→    76→}

   281→   281→/* ---------- Cuerpo principal ---------- */
   282→   282→export function SpectatorView({ tid }: { tid: string | null }) {
   283→   283→  const { data: tournaments, loading, error } = useTournaments();
   284→   284→  /* El visor siempre es la misma URL (/?v=show). Si no viene ?t=, se
   285→   285→     resuelve el torneo ACTIVO desde Firestore (arena/active). Solo puede
   286→   286→     haber uno activo a la vez — lo controla el admin desde "Torneo en vivo". */
   287→   287→  const active = useActiveTournament();
   288→   288→  const effectiveTid = tid ?? active.tid;
   289→   289→  const tournament: Tournament | null = useMemo(
   290→   290→    () => tournaments.find((t) => t.id === effectiveTid) ?? null,
   291→   291→    [tournaments, effectiveTid]
   292→   292→  );
   293→   293→  const { data: players } = usePlayers(tournament ? effectiveTid : null);
   294→   294→  const { data: bracket } = useBracket(tournament ? effectiveTid : null);
   295→   295→  const arenaState = useArenaState(tournament ? effectiveTid : null);
   296→   296→
   297→   297→  const status = tournament?.status ?? "open";
   298→   298→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   299→   299→  const [revealCount, setRevealCount] = useState(0);
   300→   300→  const rootRef = useRef<HTMLDivElement | null>(null);
   301→   301→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   302→   302→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   303→   303→
   304→   304→  /* cartelera de horarios: la controla el admin desde "Torneo en vivo".
   305→   305→     El visor solo refleja el estado (pantalla proyectada, sin interacción). */
   306→   306→  const schedOpen = arenaState?.schedOpen ?? false;
   307→   307→  /* la transmisión del OBS solo se muestra si el admin la activó (fase 5) */
   308→   308→  const transmissionVisible = arenaState?.transmissionVisible ?? false;
   309→   309→  /* reproductor de sonidos (carga la config global + reproduce) */
   310→   310→  const playSound = useSoundPlayer();
   311→   311→
   312→   312→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   313→   313→     se refleja sola en el visor). El toggle manual gana hasta que el status
   314→   314→     vuelva a cambiar, porque la anulación queda atada al status vigente.
   315→   315→     CUENTA REGRESIVA: en cuanto el admin programa el inicio (startAt), el
   316→   316→     visor salta solo a la pestaña de brackets, donde el cronómetro de inicio
   317→   317→     se ve en grande; si cancela la cuenta regresiva regresa al modo registro. */
   318→   318→  const inRegisterPhase = status === "open" || status === "closed";
   319→   319→  const autoMode: ViewerMode = inRegisterPhase ? (tournament?.startAt ? "brackets" : "reg") : "brackets";
   320→   320→  const baseMode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   321→   321→  /* El admin abrió el selector de banco → el visor pasa a la vista de Banco
   322→   322→     para que el caster vea las reservas. Es prioritario y transitorio:
   323→   323→     en cuanto se confirma/cancela (bankPick = null) regresa al modo base. */
   324→   324→  const arenaBankPick = arenaState?.bankPick ?? null;
   325→   325→  const forceBank =
   326→   326→    !!arenaBankPick && (status === "live" || status === "mixing" || status === "stopped" || status === "finished");
   327→   327→  const mode: ViewerMode = forceBank ? "bank" : baseMode;
   328→   328→
   329→   329→  /* sincroniza el modo en la URL para compartir la vista correcta */
   330→   330→  useEffect(() => {
   331→   331→    try {
   332→   332→      const url = new URL(window.location.href);
   333→   333→      if (mode === "reg") url.searchParams.set("mode", "reg");
   334→   334→      else url.searchParams.delete("mode");
   335→   335→      if (url.toString() !== window.location.href) window.history.replaceState(null, "", url.toString());
   336→   336→    } catch {
   337→   337→      /* ignore */
   338→   338→    }
   339→   339→  }, [mode]);
   340→   340→
   341→   341→  const changeMode = (m: ViewerMode) => {
   342→   342→    setOverride({ status, mode: m });
   343→   343→  };
   344→   344→
   345→   345→  // animación de mix
   346→   346→  const totalSlots = bracket ? bracket.rounds[0].length * (bracket.s ?? 2) : 0;
   347→   347→  const pool = useMemo(() => {
   348→   348→    if (!bracket) return [];
   349→   349→    const labels = bracket.rounds[0].flatMap((m) => m.slots.map((s) => s.label || "TBD"));
   350→   350→    return Array.from(new Set(labels));
   351→   351→  }, [bracket]);
   352→   352→
   353→   353→  useEffect(() => {
   354→   354→    if (status !== "mixing" || !bracket || mode !== "brackets") return;
   355→   355→    const r0 = requestAnimationFrame(() => {
   356→   356→      setRevealCount(0);
   357→   357→    });
   358→   358→    // el sonido de inicio del mix (mixLaunch) ya lo dispara el admin
   359→   359→    // al lanzar el mix — no duplicar aquí
   360→   360→    toRef.current = setTimeout(() => {
   361→   361→      ivRef.current = setInterval(() => {
   362→   362→        setRevealCount((c) => {
   363→   363→          if (c >= totalSlots) {
   364→   364→            if (ivRef.current) clearInterval(ivRef.current);
   365→   365→            return c;
   366→   366→          }
   367→   367→          // el sonido de cada tick (mixPlace) lo dispara el effect
   368→   368→          // que vigila revealCount — no duplicar aquí
   369→   369→          return c + 1;
   370→   370→        });
   371→   371→      }, 240);
   372→   372→    }, 2400);
   373→   373→    return () => {
   374→   374→      cancelAnimationFrame(r0);
   375→   375→      if (toRef.current) clearTimeout(toRef.current);
   376→   376→      if (ivRef.current) clearInterval(ivRef.current);
   377→   377→    };
   378→   378→  }, [status, bracket, totalSlots, mode]);
   379→   379→
   380→   380→  /* ----- PODIO — controlado por el admin (arenaState.showPodium).
   381→   381→     DOBLE SEGURIDAD anti-residual: la señal del admin se ignora si el
   382→   382→     bracket actual NO tiene campeón real (p. ej. señal vieja en la base
   383→   383→     de datos tras reiniciar el torneo, o al actualizar el visor).
   384→   384→     Sin campeón → jamás se dibuja el podio, aunque el flag venga true. ----- */
   385→   385→  const podiumFlag = arenaState?.showPodium ?? false;
   386→   386→  const hasChampion = useMemo(() => {
   387→   387→    if (!bracket) return false;
   388→   388→    return !!getPodium(bracket).champion?.pid;
   389→   389→  }, [bracket]);
   390→   390→  const showPodium = podiumFlag && hasChampion;
   391→   391→
   392→   392→  /* sonido de celebración: solo en la transición oculto → visible */
   393→   393→  const prevPodiumFlag = useRef(false);
   394→   394→  useEffect(() => {
   395→   395→    if (podiumFlag && !prevPodiumFlag.current && hasChampion) playSound("tournamentFinish");
   396→   396→    prevPodiumFlag.current = podiumFlag;
   397→   397→  }, [podiumFlag, hasChampion, playSound]);
   398→   398→
