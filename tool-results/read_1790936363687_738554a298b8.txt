    83→    83→
    84→    84→export function LiveAdminView({
    85→    85→  tournament,
    86→    86→  players,
    87→    87→  bracket,
    88→    88→  loadingBracket,
    89→    89→}: {
    90→    90→  tournament: Tournament | null;
    91→    91→  players: Player[];
    92→    92→  bracket: Bracket | null;
    93→    93→  loadingBracket: boolean;
    94→    94→}) {

    96→    96→  const [pickBank, setPickBank] = useState<{ r: number; m: number; slotIdx: number } | null>(null);
    97→    97→  const [confirm, setConfirm] = useState<null | { kind: "remix" | "finish" | "wipe" | "reopen" | "results" | "fill" | "demos" }>(null);

    99→    99→  const [view, setView] = useState<AdminView>("tree");
   100→   100→  const [panelOpen, setPanelOpen] = useState(true);
   101→   101→  const [zoom, setZoom] = useState<number | null>(null); // null = auto-fit
   102→   102→  const [startMins, setStartMins] = useState(1);
   103→   103→
   104→   104→  const bank = useMemo(() => players.filter((p) => p.seat === "bank"), [players]);
   105→   105→  const participants = useMemo(
   106→   106→    () => (tournament ? groupParticipants(players, tournament.modality) : []),
   107→   107→    [players, tournament]
   108→   108→  );
   109→   109→  const officialsCount = useMemo(() => players.filter((p) => p.seat === "off").length, [players]);
   110→   110→  const demoCount = useMemo(() => players.filter((p) => p.demo).length, [players]);
   111→   111→  const missingPlayers = tournament ? playersCapacity(tournament) - officialsCount : 0;
   112→   112→  const schedule = useMemo(
   113→   113→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map()),
   114→   114→    [bracket, tournament]
   115→   115→  );
   116→   116→  /* capacidad del escenario: matches simultáneos según las 4 PCs */
   117→   117→  const maxSim = tournament ? maxSimultaneous(tournament.modality) : 1;
   118→   118→  const liveCount = useMemo(
   119→   119→    () => (bracket ? bracket.rounds.reduce((acc, round) => acc + round.filter((x) => x.status === "live").length, 0) : 0),
   120→   120→    [bracket]
   121→   121→  );
   122→   122→
   123→   123→  /* navegación secuencial de matches (para editar rápido sin recorrer todo) */

   125→   125→    if (!bracket) return [];
   126→   126→    const out: { r: number; m: number }[] = [];
   127→   127→    bracket.rounds.forEach((round, r) => round.forEach((_, m) => out.push({ r, m })));
   128→   128→    return out;
   129→   129→  }, [bracket]);
   130→   130→  const flatIdx = useMemo(() => {
   131→   131→    if (!sel || !bracket) return -1;
   132→   132→    return flatMatches.findIndex((x) => x.r === sel.r && x.m === sel.m);
   133→   133→  }, [flatMatches, sel, bracket]);
   134→   134→
   135→   135→  if (!tournament) {
   136→   136→    return (
   137→   137→      <div className="panel clip-card">
   138→   138→        <EmptyState
   139→   139→          icon={<Radio size={26} />}
   140→   140→          title="Selecciona un torneo"
   141→   141→          message="Elige un torneo en el selector superior para administrar el bracket en vivo: ganadores, descalificaciones y reemplazos del banco."
   142→   142→        />
   143→   143→      </div>
   144→   144→    );
   145→   145→  }
   146→   146→
   147→   147→  const guard = async (fn: () => Promise<void>) => {
   148→   148→    setBusy(true);
   149→   149→    try {
   150→   150→      await fn();
   151→   151→    } catch (e) {
   152→   152→      toast.error(e instanceof Error ? e.message : "Error de operación");
   153→   153→    } finally {
   154→   154→      setBusy(false);
   155→   155→    }
   156→   156→  };
   157→   157→
   158→   158→  const selMatch = sel && bracket ? bracket.rounds[sel.r]?.[sel.m] : null;
   159→   159→
   160→   160→  const stepMatch = (dir: 1 | -1) => {
   161→   161→    const nx = flatMatches[flatIdx + dir];
   162→   162→    if (nx) setSel(nx);
   163→   163→  };
   164→   164→
   165→   165→  const zoomStep = (dir: 1 | -1) => {
   166→   166→    setZoom((z) => {
   167→   167→      const base = z ?? 0.8;
   168→   168→      return Math.min(1.8, Math.max(0.3, Math.round((base + dir * 0.1) * 100) / 100));
   169→   169→    });
   170→   170→  };
   171→   171→
   172→   172→  return (
   173→   173→    <div className="flex flex-col gap-3 min-h-0">
   174→   174→      {/* ============ barra de info del torneo ============ */}
   175→   175→      <div className="panel clip-card p-4 flex flex-wrap items-center gap-3">
   176→   176→        <div className="flex items-center gap-3 min-w-0 mr-auto">
   177→   177→          <span className="slashes w-8 h-4 inline-block shrink-0" aria-hidden />
   178→   178→          <div className="min-w-0">
   179→   179→            <div className="font-display italic text-[17px] uppercase truncate">{tournament.name}</div>
   180→   180→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   181→   181→              <StatusChip status={tournament.status} />
   182→   182→              <ModalityChip modality={tournament.modality} />
   183→   183→              {bracket ? (
   184→   184→                <span
   185→   185→                  className={`chip clip-tag border ${
   186→   186→                    liveCount >= maxSim
   187→   187→                      ? "bg-[#e8102e]/15 text-[#ff8095] border-[#e8102e]/40"
   188→   188→                      : "bg-[#e8102e]/12 text-[#ff8095] border-[#e8102e]/35"
   189→   189→                  }`}
   190→   190→                  title="Matches en juego / máximo simultáneo según las PCs del escenario (4 PCs)"
   191→   191→                >
   192→   192→                  <Monitor size={10} />
   193→   193→                  EN JUEGO {liveCount}/{maxSim}
   194→   194→                </span>
   195→   195→              ) : null}
   196→   196→              {bracket ? (
   197→   197→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   198→   198→                  {doneMatches(bracket)}/{totalMatches(bracket)} MATCHES
   199→   199→                </span>
   200→   200→              ) : (
   201→   201→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   202→   202→                  {officialsCount}/{playersCapacity(tournament)} JUGADORES
   203→   203→                </span>
   204→   204→              )}
   205→   205→              {tournament.startAt ? (
   206→   206→                <span className="chip clip-tag bg-[#ffb830]/12 text-[#ffb830] border border-[#ffb830]/35">
   207→   207→                  <TimerReset size={10} />
   208→   208→                  INICIA ≈ {fmtTime(tournament.startAt)}
   209→   209→                </span>
   210→   210→              ) : null}
   211→   211→            </div>
   212→   212→          </div>
   213→   213→        </div>
   214→   214→        <span className="text-[9px] font-extrabold uppercase tracking-[0.18em] text-[#6b6e78] hidden md:inline">
   215→   215→          Cada fase se refleja automáticamente en el visor
   216→   216→        </span>
   217→   217→      </div>
   218→   218→
   219→   219→      {/* ============ FLUJO POR FASES ============ */}
   220→   220→      <PhaseBar
   221→   221→        tournament={tournament}
   222→   222→        bracket={bracket}
   223→   223→        participantsCount={participants.length}
   224→   224→        officialsCount={officialsCount}
   225→   225→        missingPlayers={missingPlayers}
   226→   226→        demoCount={demoCount}
   227→   227→        bankCount={bank.length}
   228→   228→        startMins={startMins}
   229→   229→        setStartMins={setStartMins}
   230→   230→        busy={busy}
   231→   231→        guard={guard}
   232→   232→        onConfirm={(kind) => setConfirm({ kind })}
   233→   233→        onLaunchMix={() =>
   234→   234→          guard(async () => {
   235→   235→            await launchMix(tournament.id, players, tournament);
   236→   236→            setSel(null);
   237→   237→            toast.success("Mix match lanzado — el visor reproduce la animación");
   238→   238→          })
   239→   239→        }
   240→   240→      />
   241→   241→
   242→   242→      {/* ============ cuerpo ============ */}
   243→   243→      {loadingBracket ? (
   244→   244→        <div className="panel clip-card p-14 flex items-center justify-center text-[#8e919c]">
   245→   245→          <Loader2 size={20} className="animate-spin" />
   246→   246→        </div>
   247→   247→      ) : !bracket ? (
   248→   248→        <div className="panel clip-card">
   249→   249→          <EmptyState
   250→   250→            icon={<Zap size={26} />}
   251→   251→            title="Brackets no creados"
   252→   252→            message={`Cuando estés listo, lanza el MIX MATCH: se randomizarán los ${participants.length} participantes y el visor de espectadores mostrará la animación de llenado de brackets.`}
   253→   253→            action={
   254→   254→              <div className="flex flex-wrap gap-2 justify-center">
   255→   255→                <Btn
   256→   256→                  variant="red"
   257→   257→                  disabled={participants.length < 2}
   258→   258→                  onClick={() =>
   259→   259→                    guard(async () => {
   260→   260→                      await launchMix(tournament.id, players, tournament);
   261→   261→                      toast.success("Mix match lanzado");
   262→   262→                    })
   263→   263→                  }
   264→   264→                >
   265→   265→                  <Rocket size={13} />
   266→   266→                  Lanzar mix match
   267→   267→                </Btn>
   268→   268→                <Btn
   269→   269→                  variant="dark"
   270→   270→                  onClick={() =>
   271→   271→                    guard(async () => {
   272→   272→                      await setStartAt(tournament.id, Date.now() + startMins * 60_000);
   273→   273→                      toast.success(`Inicio configurado en ${startMins} min — el visor cambia a brackets con el cronómetro`);
   274→   274→                    })
   275→   275→                  }
   276→   276→                >
   277→   277→                  <TimerReset size={13} />
   278→   278→                  Inicio en {startMins} min
   279→   279→                </Btn>
   280→   280→              </div>
   281→   281→            }
   282→   282→          />
   283→   283→          {/* vista previa de participantes */}
   284→   284→          {participants.length > 0 ? (
   285→   285→            <div className="px-5 pb-5">
   286→   286→              <div className="label-cap mb-2 flex items-center gap-2">
   287→   287→                <Users size={11} />
   288→   288→                Participantes que entrarán al sorteo ({participants.length})
   289→   289→              </div>
   290→   290→              <div className="flex flex-wrap gap-1.5 max-h-36 overflow-y-auto">
   291→   291→                {participants.map((p, i) => (
   292→   292→                  <span key={p.pid} className="chip clip-tag bg-white/[0.05] text-[#c9cbd3] border border-white/10">
   293→   293→                    {i + 1}. {p.label}
   294→   294→                  </span>
   295→   295→                ))}
   296→   296→              </div>
   297→   297→            </div>
   298→   298→          ) : null}
   299→   299→        </div>
   300→   300→      ) : (
   301→   301→        <>
   302→   302→          {/* ============ toolbar de vistas ============ */}
   303→   303→          <div className="panel clip-card-sm px-3 py-2 flex flex-wrap items-center gap-2">
   304→   304→            <div className="flex gap-1 p-0.5 bg-black/40 border border-white/10" role="radiogroup" aria-label="Vista del bracket">
   305→   305→              <button
   306→   306→                type="button"
   307→   307→                role="radio"
   308→   308→                aria-checked={view === "tree"}
   309→   309→                onClick={() => setView("tree")}
   310→   310→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   311→   311→                  view === "tree" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   312→   312→                }`}
   313→   313→              >
   314→   314→                <LayoutPanelLeft size={12} />
   315→   315→                Bracket
   316→   316→              </button>
   317→   317→              <button
   318→   318→                type="button"
   319→   319→                role="radio"
   320→   320→                aria-checked={view === "rounds"}
   321→   321→                onClick={() => setView("rounds")}
   322→   322→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   323→   323→                  view === "rounds" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   324→   324→                }`}
   325→   325→              >
   326→   326→                <Columns3 size={12} />
   327→   327→                Rondas
   328→   328→              </button>
   329→   329→            </div>
   330→   330→
   331→   331→            {selMatch ? (
   332→   332→              <span className="chip clip-tag bg-[#e8102e]/15 text-[#ff8095] border border-[#e8102e]/40 text-[9px] flex items-center gap-1.5">
   333→   333→                <Hand size={10} />
   334→   334→                {matchTag(bracket, sel!.r, sel!.m)} seleccionado
   335→   335→              </span>
   336→   336→            ) : (
   337→   337→              <span className="text-[9px] font-extrabold uppercase tracking-[0.16em] text-[#6b6e78] hidden sm:inline">
   338→   338→                Clic en un match para administrarlo
   339→   339→              </span>
   340→   340→            )}
   341→   341→
   342→   342→            <div className="ml-auto flex items-center gap-1.5">
   343→   343→              {/* navegación prev/next */}
   344→   344→              <div className="flex items-center gap-1">
   345→   345→                <IconBtn title="Match anterior" disabled={flatIdx <= 0} onClick={() => stepMatch(-1)}>
   346→   346→                  <ChevronLeft size={14} />
   347→   347→                </IconBtn>
   348→   348→                <IconBtn title="Match siguiente" disabled={flatIdx < 0 || flatIdx >= flatMatches.length - 1} onClick={() => stepMatch(1)}>
   349→   349→                  <ChevronRight size={14} />
   350→   350→                </IconBtn>
   351→   351→              </div>
   352→   352→
   353→   353→              {/* zoom (solo vista bracket) */}
   354→   354→              {view === "tree" ? (
   355→   355→                <div className="flex items-center gap-1 pl-2 border-l border-white/10">
   356→   356→                  <IconBtn title="Alejar" onClick={() => zoomStep(-1)}>
   357→   357→                    <ZoomOut size={14} />
   358→   358→                  </IconBtn>
   359→   359→                  <span className="text-[9px] font-extrabold tabular-nums text-[#8e919c] w-10 text-center uppercase">
   360→   360→                    {zoom ? `${Math.round(zoom * 100)}%` : "AUTO"}

   362→   362→                  <IconBtn title="Acercar" onClick={() => zoomStep(1)}>
   363→   363→                    <ZoomIn size={14} />
   364→   364→                  </IconBtn>
   365→   365→                  <IconBtn title="Ajustar a la pantalla" active={zoom === null} onClick={() => setZoom(null)}>

   367→   367→                  </IconBtn>
   368→   368→                </div>
   369→   369→              ) : null}
   370→   370→
   371→   371→              {/* toggle panel lateral */}

   373→   373→                type="button"
   374→   374→                onClick={() => setPanelOpen((v) => !v)}
   375→   375→                title={panelOpen ? "Ocultar panel del match" : "Mostrar panel del match"}

   379→   379→                    ? "border-[#e8102e]/60 bg-[#e8102e]/10 text-white"
   380→   380→                    : "border-white/12 text-[#8e919c] hover:text-white"
   381→   381→                }`}
   382→   382→              >
   383→   383→                {panelOpen ? <PanelRightClose size={14} /> : <PanelRightOpen size={14} />}
   384→   384→              </button>
   385→   385→            </div>
   386→   386→          </div>
   387→   387→
   388→   388→          {/* ============ stage + panel lateral colapsable ============ */}
   389→   389→          <div className="flex gap-3 items-stretch min-h-0">
   390→   390→            {/* área principal */}
   391→   391→            <div className="flex-1 min-w-0 min-h-0">
   392→   392→              <div className="panel clip-card p-3 min-h-[420px] h-[calc(100vh-330px)] max-h-[80vh] flex flex-col">
   393→   393→                <div className="flex items-center justify-between px-2 pb-2">
   394→   394→                  <span className="label-cap">

   397→   397→                  <span className="label-cap">{MODALITY_LABEL[tournament.modality]} · a {tournament.winsNeeded} wins</span>

   407→   407→                        bracket={bracket}
   408→   408→                        modality={tournament.modality}
   409→   409→                        mode="admin"
   410→   410→                        selectedId={selMatch?.id ?? null}
   411→   411→                        onSelectMatch={(r, m) => setSel({ r, m })}
   412→   412→                        schedule={schedule}
   413→   413→                        showTimes
   414→   414→                      />
   415→   415→                    </FitStage>
   416→   416→                  </div>
   417→   417→                ) : (
   418→   418→                  <RoundColumns
   419→   419→                    bracket={bracket}
   420→   420→                    modality={tournament.modality}
   421→   421→                    times={schedule}
   422→   422→                    selectedId={selMatch?.id ?? null}
   423→   423→                    onSelectMatch={(r, m) => setSel({ r, m })}
   424→   424→                    onPickWinner={(r, m, slotIdx) =>
   425→   425→                      guard(async () => {

   430→   430→                    }
   431→   431→                  />

   433→   433→              </div>
   434→   434→            </div>
   435→   435→
   436→   436→            {/* panel lateral (desktop, colapsable) */}
   437→   437→            <aside
