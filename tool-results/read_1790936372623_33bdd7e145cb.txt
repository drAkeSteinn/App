  1045→function MiniCountdown({ target }: { target: number }) {
  1046→  const [now, setNow] = useState(() => Date.now());
  1047→  useEffect(() => {
  1048→    const t = setInterval(() => setNow(Date.now()), 500);
  1049→    return () => clearInterval(t);
  1050→  }, []);
  1051→  const total = Math.max(0, Math.floor((target - now) / 1000));
  1052→  const h = Math.floor(total / 3600);
  1053→  const m = Math.floor((total % 3600) / 60);
  1054→  const s = total % 60;
  1055→  const pad = (x: number) => String(x).padStart(2, "0");
  1056→  return (
  1057→    <span className="font-display italic text-[20px] leading-none text-silver-grad tabular-nums">
  1058→      {h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`}
  1059→    </span>
  1060→  );
  1061→}
