    49→/* Detecta el primer evento "dramático" entre dos versiones del bracket:
    50→   match que pasa a DONE (ganador marcado o cambiado) o a LIVE (en juego).
    51→   Las PCs del match se leen del match actual (al poner en juego ya tienen
    52→   PCs asignadas) o del snapshot anterior (al definir ganador ya se liberaron). */
    53→function findSpotlightEvent(
    54→  cur: Bracket,
    55→  prev: Bracket,
    56→  winsNeeded: number,
    57→  modality: Modality
    58→): SpotlightData | null {
    59→  for (let r = 0; r < cur.rounds.length; r++) {
    60→    const round = cur.rounds[r];
    61→    for (let m = 0; m < round.length; m++) {
    62→      const match = round[m];
    63→      const old = prev.rounds[r]?.[m];
    64→      if (!match || !old || match.bye) continue;
    65→      const becameDone = match.status === "done" && old.status !== "done";
    66→      const winnerChanged = match.status === "done" && old.status === "done" && match.w !== old.w;
    67→      const becameLive = match.status === "live" && old.status !== "live";
    68→      if (becameDone || winnerChanged || becameLive) {
    69→        const pcs = match.pcs?.length ? match.pcs : old.pcs;
    70→        return {
    71→          id: `${match.id}-${match.status}-${match.w}-${Date.now()}`,
    72→          kind: match.status === "live" ? "live" : "win",
    73→          tag: matchTag(cur, r, m),
    74→          players: match.slots.map((s, i) => ({
    75→            name: s.label || "—",
    76→            score: s.score ?? 0,
    77→            pcs: pcsForSide(modality, pcs, i),
    78→          })),
    79→          winner: match.w,
    80→          winsNeeded,
    81→        };
    82→      }
    83→    }
    84→  }
    85→  // verificar también el match de 3er lugar
    86→  if (cur.thirdPlace && prev.thirdPlace) {
    87→    const tp = cur.thirdPlace;
    88→    const oldTp = prev.thirdPlace;
    89→    const becameDone = tp.status === "done" && oldTp.status !== "done";
    90→    const becameLive = tp.status === "live" && oldTp.status !== "live";
    91→    if (becameDone || becameLive) {
    92→      const pcs = tp.pcs?.length ? tp.pcs : oldTp.pcs;
    93→      return {
    94→        id: `${tp.id}-${tp.status}-${tp.w}-${Date.now()}`,
    95→        kind: tp.status === "live" ? "live" : "win",
    96→        tag: "3ER LUGAR",
    97→        players: tp.slots.map((s, i) => ({
    98→          name: s.label || "—",
    99→          score: s.score ?? 0,
   100→          pcs: pcsForSide(modality, pcs, i),
   101→        })),
   102→        winner: tp.w,
   103→        winsNeeded,
   104→      };
   105→    }
   106→  }
   107→  return null;
   108→}

   125→function useClock() {
   126→  const [now, setNow] = useState<Date | null>(null);
   127→  useEffect(() => {
   128→    const first = requestAnimationFrame(() => setNow(new Date()));
   129→    const t = setInterval(() => setNow(new Date()), 1000);
   130→    return () => {
   131→      cancelAnimationFrame(first);
   132→      clearInterval(t);
   133→    };
   134→  }, []);
   135→  return now;
   136→}

   341→/* ---------- Cuerpo principal ---------- */
   342→export function SpectatorView({ tid }: { tid: string | null }) {
   343→  const { data: tournaments, loading, error } = useTournaments();
   344→  /* El visor siempre es la misma URL (/?v=show). Si no viene ?t=, se
   345→     resuelve el torneo ACTIVO desde Firestore (arena/active). Solo puede
   346→     haber uno activo a la vez — lo controla el admin desde "Torneo en vivo". */
   347→  const active = useActiveTournament();
   348→  const effectiveTid = tid ?? active.tid;
   349→  const tournament: Tournament | null = useMemo(
   350→    () => tournaments.find((t) => t.id === effectiveTid) ?? null,
   351→    [tournaments, effectiveTid]
   352→  );
   353→  const { data: players } = usePlayers(tournament ? effectiveTid : null);
   354→  const { data: bracket } = useBracket(tournament ? effectiveTid : null);
   355→  const arenaState = useArenaState(tournament ? effectiveTid : null);
   356→
   357→  const status = tournament?.status ?? "open";
   358→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   359→  const [revealCount, setRevealCount] = useState(0);
   360→  const [showPodium, setShowPodium] = useState(false);
   361→  const [spotlight, setSpotlight] = useState<SpotlightData | null>(null);
   362→  const [subEnter, setSubEnter] = useState<SubEnterData | null>(null);
   363→  const prevBracketRef = useRef<{ tid: string | null; bracket: Bracket } | null>(null);
   364→  const prevBankPickRef = useRef<{ id: number; slotPid: string | null } | null>(null);
   365→  const rootRef = useRef<HTMLDivElement | null>(null);
   366→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   367→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   368→
   369→  /* cartelera de horarios: la controla el admin desde "Torneo en vivo".
   370→     El visor solo refleja el estado (pantalla proyectada, sin interacción). */
   371→  const schedOpen = arenaState?.schedOpen ?? false;
   372→  /* reproductor de sonidos (carga la config global + reproduce) */
   373→  const playSound = useSoundPlayer();
   374→
   375→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   376→     se refleja sola en el visor). El toggle manual gana hasta que el status
   377→     vuelva a cambiar, porque la anulación queda atada al status vigente.
   378→     CUENTA REGRESIVA: en cuanto el admin programa el inicio (startAt), el
   379→     visor salta solo a la pestaña de brackets, donde el cronómetro de inicio
   380→     se ve en grande; si cancela la cuenta regresiva regresa al modo registro. */
   381→  const inRegisterPhase = status === "open" || status === "closed";
   382→  const autoMode: ViewerMode = inRegisterPhase ? (tournament?.startAt ? "brackets" : "reg") : "brackets";
   383→  const baseMode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   384→  /* El admin abrió el selector de banco → el visor pasa a la vista de Banco
   385→     para que el caster vea las reservas. Es prioritario y transitorio:
   386→     en cuanto se confirma/cancela (bankPick = null) regresa al modo base. */
   387→  const arenaBankPick = arenaState?.bankPick ?? null;
   388→  const forceBank =
   389→    !!arenaBankPick && (status === "live" || status === "mixing" || status === "finished");
   390→  const mode: ViewerMode = forceBank ? "bank" : baseMode;
   391→
   392→  /* sincroniza el modo en la URL para compartir la vista correcta */
   393→  useEffect(() => {
   394→    try {
   395→      const url = new URL(window.location.href);
   396→      if (mode === "reg") url.searchParams.set("mode", "reg");
   397→      else url.searchParams.delete("mode");
   398→      if (url.toString() !== window.location.href) window.history.replaceState(null, "", url.toString());
   399→    } catch {
   400→      /* ignore */
   401→    }
   402→  }, [mode]);
   403→
   404→  const changeMode = (m: ViewerMode) => {
   405→    setOverride({ status, mode: m });
   406→  };
   407→
   408→  // animación de mix
   409→  const totalSlots = bracket ? bracket.rounds[0].length * (bracket.s ?? 2) : 0;
   410→  const pool = useMemo(() => {
   411→    if (!bracket) return [];
   412→    const labels = bracket.rounds[0].flatMap((m) => m.slots.map((s) => s.label || "TBD"));
   413→    return Array.from(new Set(labels));
   414→  }, [bracket]);
   415→
   416→  useEffect(() => {
   417→    if (status !== "mixing" || !bracket || mode !== "brackets") return;
   418→    const r0 = requestAnimationFrame(() => {
   419→      setRevealCount(0);
   420→      setShowPodium(false);
   421→    });
   422→    // el sonido de inicio del mix (mixLaunch) ya lo dispara el admin
   423→    // al lanzar el mix — no duplicar aquí
   424→    toRef.current = setTimeout(() => {
   425→      ivRef.current = setInterval(() => {
   426→        setRevealCount((c) => {
   427→          if (c >= totalSlots) {
   428→            if (ivRef.current) clearInterval(ivRef.current);
   429→            return c;
   430→          }
   431→          // el sonido de cada tick (mixPlace) lo dispara el effect
   432→          // que vigila revealCount — no duplicar aquí
   433→          return c + 1;
   434→        });
   435→      }, 240);
   436→    }, 2400);
   437→    return () => {
   438→      cancelAnimationFrame(r0);
   439→      if (toRef.current) clearTimeout(toRef.current);
   440→      if (ivRef.current) clearInterval(ivRef.current);
   441→    };
   442→  }, [status, bracket, totalSlots, mode]);
   443→
   444→  useEffect(() => {
   445→    // el podio se controla desde el admin (arenaState.showPodium)
   446→    // puede activarse en cualquier momento, no solo cuando el torneo está finished
   447→    if (arenaState?.showPodium) {
   448→      const r = requestAnimationFrame(() => {
   449→        setShowPodium(true);
   450→        playSound("tournamentFinish");
   451→      });
   452→      return () => cancelAnimationFrame(r);
   453→    }
   454→    const r = requestAnimationFrame(() => setShowPodium(false));
   455→    return () => cancelAnimationFrame(r);
   456→  }, [arenaState?.showPodium, playSound]);
   457→
   458→  /* ----- SPOTLIGHT: ganador marcado / match en juego → corte dramático.
   459→     Compara cada snapshot del bracket con el anterior; al detectar un match
   460→     que pasa a DONE (o cambia de ganador) o a LIVE, lanza el overlay. ----- */
   461→  useEffect(() => {
   462→    const prev = prevBracketRef.current;
   463→    prevBracketRef.current = effectiveTid && bracket ? { tid: effectiveTid, bracket } : null;
   464→    if (!bracket || !prev || prev.tid !== effectiveTid || status !== "live") return;
   465→    const event = findSpotlightEvent(bracket, prev.bracket, tournament?.winsNeeded ?? 1, tournament?.modality ?? 1);
   466→    if (event) {
   467→      setSpotlight(event);
   468→      // reproducir sonido según el tipo de evento
   469→      if (event.kind === "win") {
   470→        playSound("winner");
   471→      } else if (event.kind === "live") {
   472→        playSound("matchLive");
   473→      }
   474→    }
   475→  }, [bracket, status, effectiveTid, tournament, playSound]);
   476→
   477→  /* el corte se va solo tras 10 segundos (para ver bien quién ganó y qué
   478→     match está en juego) y regresa a los brackets */
   479→  useEffect(() => {
   480→    if (!spotlight) return;
   481→    const t = setTimeout(() => setSpotlight(null), 10_000);

   485→  /* glitch sutil de toda la pantalla al disparar el corte (Web Animations API:

   505→      { duration: 1900, easing: "linear" }

   508→      anim.cancel();

   510→  }, [spotlight]);
   511→
   512→  /* ref siempre fresca del bracket (para leer el estado más reciente dentro
   513→     de un setTimeout sin closure stalado) */
   514→  const bracketRef = useRef<Bracket | null>(null);
   515→  useEffect(() => {
   516→    bracketRef.current = bracket;
   517→  }, [bracket]);
   518→

   520→     el visor muestra la animación del reserva que entra al match.
   521→     Detección: al abrir el bankPick se snapshot el pid del slot; al cerrar
   522→     (bankPick → null) se compara. Si el pid cambió → confirmó (animación);
   523→     si es el mismo → canceló (nada). Un grace de 500ms deja llegar la
   524→     actualización del bracket por si el orden de onSnapshot se invierte. ----- */
   525→  useEffect(() => {
   526→    const pick = arenaBankPick;
   527→    if (pick) {
   528→      if (!prevBankPickRef.current || prevBankPickRef.current.id !== pick.id) {
   529→        const slotPid =
   530→          bracketRef.current?.rounds[pick.r]?.[pick.m]?.slots[pick.slotIdx]?.pid ?? null;
   531→        prevBankPickRef.current = {
   532→          r: pick.r,
   533→          m: pick.m,
   534→          slotIdx: pick.slotIdx,
   535→          id: pick.id,
   536→          slotPid,
   537→        };
   538→      }
   539→      return;
   540→    }
   541→    const prev = prevBankPickRef.current;
   542→    if (!prev) return;
   543→    prevBankPickRef.current = null;
   544→    const captured = prev;
   545→    const t = setTimeout(() => {
   546→      const b = bracketRef.current;
   547→      const curSlot = b?.rounds[captured.r]?.[captured.m]?.slots[captured.slotIdx];
   548→      const curPid = curSlot?.pid ?? null;
   549→      if (curPid && curPid !== captured.slotPid) {
   550→        const nick = curSlot?.label || curSlot?.members[0]?.nick || "—";
   551→        const isFinal = b ? captured.r === b.rounds.length - 1 : false;
   552→        const tag = isFinal ? "GRAN FINAL" : b ? matchTag(b, captured.r, captured.m) : "MATCH";
   553→        setSubEnter({ id: `${captured.id}-${Date.now()}`, nick, matchTag: tag });
   554→      }
   555→    }, 500);
   556→    return () => clearTimeout(t);
   557→  }, [arenaBankPick]);
   558→
   559→  /* la animación de entrada se descarta sola tras 4.5 s (o al hacer click) */
   560→  useEffect(() => {
   561→    if (!subEnter) return;
   562→    const t = setTimeout(() => setSubEnter(null), 4500);
   563→    return () => clearTimeout(t);
   564→  }, [subEnter]);

   567→     para el público (arenaState.sound). mixPlace es interno del visor:

   570→  useEffect(() => {
   571→    const sig = arenaState?.sound;
   572→    if (!sig || sig.id === lastSoundId.current) return;
   573→    lastSoundId.current = sig.id;
   574→    // validar que el evento existe antes de reproducirlo
   575→    const validEvents = ["scoreUp", "winner", "matchLive", "bankSwap", "tournamentFinish", "mixLaunch", "tournamentStart"];
   576→    if (validEvents.includes(sig.event)) {
   577→      playSound(sig.event as SoundEvent);
   578→    }
   579→  }, [arenaState?.sound, playSound]);
   580→
   581→  const prevReveal = useRef(0);
   582→  useEffect(() => {
   583→    if (status !== "mixing" || mode !== "brackets") {
   584→      prevReveal.current = revealCount;
   585→      return;
   586→    }
   587→    // en cada tick del random → sonido de colocación
   588→    if (revealCount > prevReveal.current && revealCount < totalSlots) {
   589→      playSound("mixPlace");
   590→    }

   603→      round.forEach((match, m) => {
   604→        if (match.status === "live") {
   605→          out.push({

   607→            names:

   609→                .filter((s) => s.pid)

   624→      round.forEach((match, m) => {
   625→        const st = schedule.get(match.id);
   626→        if (st && match.status === "ready" && match.slots.every((s) => s.pid) && st > Date.now() - 60_000) {
   627→          upcoming.push({
   628→            time: st,
   629→            text: `SIGUIENTE · ${matchTag(bracket, r, m)}: ${match.slots
   630→              .map((s) => s.label || "—")
   631→              .join(" VS ")} ≈ ${fmtTimeSafe(st)}`,
   632→          });
   633→        }
   634→      })
   635→    );
   636→    upcoming.sort((a, b) => a.time - b.time);
   637→    items.push(...upcoming.slice(0, 4).map((u) => u.text));
   638→    /* banco efectivo: reservas del registro antes de la 1ª eliminatoria;
   639→       después, los eliminados de la última ronda completada (rotativo) */
   640→    const bankInfo = computeBank(bracket, players);
   641→    items.push(
   642→      bankInfo.fromRound === null
   643→        ? `BANCO DE RESERVAS: ${bankInfo.entries.length} DISPONIBLES`
   644→        : `BANCO · ELIMINADOS DE ${bankInfo.sourceLabel}: ${bankInfo.entries.length} RESERVAS`
   645→    );
   646→    items.push(`${MODALITY_LABEL[tournament.modality]} · ${totalMatches(bracket)} MATCHES · A ${tournament.winsNeeded} WINS`);
   647→    items.push("JUEGA · COMpite · CONECTA — GAMING SIN LÍMITES");
   648→    return items;
   649→  }, [bracket, tournament, schedule, players]);
   650→
   651→  /* ----- pantalla de espera: no hay torneo activo para transmitir ----- */
   652→  if (!effectiveTid || (!loading && !tournament)) {
   653→    return (
   654→      <div className="fixed inset-0 overflow-hidden flex items-center justify-center" style={{ background: "#070708" }}>
   655→        <Backdrop variant="show" />
   656→        <ShowBackground dim={0.14} />
   657→        <div className="relative z-10 max-w-xl mx-auto px-6 text-center">
   658→          {loading ? (
   659→            <div className="flex justify-center py-16 text-[#8e919c]">
   660→              <Loader2 size={26} className="animate-spin" />
   661→            </div>
   662→          ) : error ? (
   663→            <div className="panel clip-card p-4 text-[12px] font-bold text-[#ff8095]">Error de Firebase: {error}</div>
   664→          ) : (
   665→            <>
   666→              <div className="flex justify-center mb-6">
   667→                <div className="w-16 h-16 clip-card border border-white/15 bg-black/40 flex items-center justify-center">
   668→                  <Radio size={28} className="text-[#e8102e]" />
   669→                </div>
   670→              </div>
   671→              <span className="text-[11px] font-extrabold tracking-[0.5em] uppercase text-[#ff8095]">Visor de espectadores</span>
   672→              <h1 className="font-display italic text-[34px] sm:text-[44px] uppercase leading-none mt-3">
   673→                <span className="text-silver-grad">Esperando</span> <span className="text-red-grad">transmisión</span>
   674→              </h1>
   675→              <p className="text-[12px] font-bold tracking-[0.16em] uppercase text-[#8e919c] mt-6 leading-relaxed">
   676→                No hay un torneo abierto para transmitir.
   677→                <br />
   678→                El administrador debe abrir un torneo desde
   679→                <br />
   680→                <span className="text-white">Torneo en vivo → Abrir torneo</span>.
   681→              </p>
   682→              <p className="text-[9px] font-bold tracking-[0.3em] uppercase text-[#6b6e78] mt-8">
   683→                Esta pantalla se actualiza sola cuando se abra un torneo

   685→            </>
   686→          )}
   687→        </div>
   688→      </div>
   689→    );

   691→
   692→  if (!tournament) {
   693→    return (
   694→      <div className="fixed inset-0 flex items-center justify-center" style={{ background: "#070708" }}>
   695→        <Loader2 size={28} className="animate-spin text-[#e8102e]" />
   696→      </div>
   697→    );
   698→  }
   699→
   700→  const mixing = status === "mixing";
   701→  const mixDone = mixing && totalSlots > 0 && revealCount >= totalSlots;
   702→  const railActive = status === "live" && mode === "brackets";
   703→
   704→  return (
   705→    <div
   706→      ref={rootRef}
   707→      className="fixed inset-0 flex flex-col overflow-hidden"
   708→      style={{ background: "#070708" }}
   709→    >
   710→      <Backdrop variant="show" />
   711→      <ShowBackground />
   712→      <ShowHeader
   713→        t={tournament}
   714→        mode={mode}
   715→        onMode={changeMode}
   716→        schedIndicator={schedOpen}
   717→      />
   718→
   719→      {/* spotlight de matches en juego */}
   720→      <AnimatePresence>
   721→        {status === "live" && mode === "brackets" && liveMatches.length > 0 ? (
   722→          <motion.div
   723→            initial={{ y: -40, opacity: 0 }}
   724→            animate={{ y: 0, opacity: 1 }}
   725→            exit={{ y: -40, opacity: 0 }}
   726→            className="relative z-20 shrink-0 live-glow bg-[linear-gradient(90deg,rgba(110,10,26,0.85),rgba(232,16,46,0.9),rgba(110,10,26,0.85))] flex items-center justify-center gap-3 py-2 px-4"
   727→          >
   728→            <span className="w-2.5 h-2.5 bg-white rounded-full blink shrink-0" aria-hidden />
   729→            <span className="text-[12px] sm:text-[14px] font-extrabold tracking-[0.2em] uppercase text-white truncate">
   730→              {liveMatches
   731→                .slice(0, 2)
   732→                .map((l) => `${l.tag}: ${l.names}`)

   734→            </span>
   735→            <span className="w-2.5 h-2.5 bg-white rounded-full blink shrink-0" aria-hidden />

   737→        ) : null}
   738→      </AnimatePresence>
   739→
   740→      <main className="relative flex-1 min-h-0">
   741→        <AnimatePresence mode="wait">
   742→          {/* MODO REGISTRO */}
   743→          {mode === "reg" ? (
   744→            <RegistrationBroadcast
   745→              key="reg"
   746→              t={tournament}
   747→              players={players}
   748→              onCountdownClick={() => changeMode("brackets")}
   749→            />
   750→          ) : mode === "bank" ? (
   751→            /* MODO BANCO — el admin abrió el selector de reservas: el visor
   752→               muestra las disponibles para que el caster llame a los jugadores */
   753→            <BankView
   754→              key="bank"
   755→              t={tournament}
   756→              bracket={bracket}
   757→              players={players}
   758→              matchTag={arenaBankPick?.tag ?? null}
   759→            />
   760→          ) : (
   761→            <>
   762→              {/* PREGAME (registro abierto o cerrado, vista brackets manual) */}
   763→              {status === "open" || status === "closed" ? <Pregame key="pre" t={tournament} players={players} /> : null}
   764→
   765→              {/* MIXING / LIVE / FINISHED: bracket */}
   766→              {status !== "open" && status !== "closed" ? (
   767→                <motion.div
   768→                  key="bracket"
   769→                  initial={{ opacity: 0, scale: 0.97 }}
   770→                  animate={{ opacity: 1, scale: 1 }}
   771→                  exit={{ opacity: 0 }}
   772→                  transition={{ duration: 0.5 }}
   773→                  className={`absolute inset-0 ${mixing && !mixDone ? "opacity-30" : ""} transition-opacity duration-500`}
   774→                >
   775→                  {bracket ? (
   776→                    <FitStage deps={[bracket, tournament.modality, status]} maxScale={1.05} minScale={0.3} className="w-full h-full">
   777→                      <BracketTree
   778→                        bracket={bracket}
   779→                        modality={tournament.modality}
   780→                        mode="viewer"
   781→                        schedule={schedule}
   782→                        showTimes={status === "live"}
   783→                        revealCount={mixing ? revealCount : undefined}
   784→                        pool={pool}
   785→                      />
   786→                    </FitStage>
   787→                  ) : (
   788→                    <div className="h-full flex items-center justify-center text-[#8e919c]">
   789→                      <Loader2 size={24} className="animate-spin" />
   790→                    </div>
   791→                  )}
   792→                </motion.div>
   793→              ) : null}
   794→            </>
   795→          )}
   796→        </AnimatePresence>
   797→
   798→        {/* overlay de mix */}
   799→        <AnimatePresence>
   800→          {mixing && bracket && mode === "brackets" ? (
   801→            <MixOverlay key="mix" locked={revealCount} total={totalSlots} pool={pool} done={mixDone} />
   802→          ) : null}
   803→        </AnimatePresence>
   804→
   805→        {/* podio — cuando el admin lo activa (showPodium) */}
   806→        <AnimatePresence>
   807→          {bracket && showPodium && mode === "brackets" ? (
   808→            <Podium key="podium" bracket={bracket} tournament={tournament} onClose={() => setShowPodium(false)} />
   809→          ) : null}
   810→        </AnimatePresence>
   811→
   812→        {/* banner "Torneo detenido" cuando está finalizado sin podio */}
   813→        <AnimatePresence>
   814→          {status === "finished" && !showPodium && mode === "brackets" ? (
   815→            <motion.div
   816→              initial={{ y: 40, opacity: 0 }}
   817→              animate={{ y: 0, opacity: 1 }}
   818→              exit={{ y: 40, opacity: 0 }}
   819→              className="relative z-20 shrink-0 border-t border-[#ffb830]/30 bg-[#171307]/90 flex items-center justify-center gap-2 py-2"
   820→            >
   821→              <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   822→              <span className="text-[10px] font-extrabold tracking-[0.28em] uppercase text-[#ffb830]">
   823→                Torneo detenido
   824→              </span>
   825→              <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   826→            </motion.div>
   827→          ) : null}
   828→        </AnimatePresence>
   829→
   830→        {/* cartelera de horarios */}
   831→        <AnimatePresence>
   832→          {schedOpen && bracket ? (
   833→            <ScheduleOverlay
   834→              key="sched"
   835→              bracket={bracket}
   836→              schedule={schedule}
   837→              tournament={tournament}
   838→              open
   839→              onClose={() => setSchedOpen(false)}
   840→            />
   841→          ) : null}
   842→        </AnimatePresence>
   843→      </main>
   844→
   845→      {/* corte dramático: ganador marcado / match en juego — 5 s y regresa a brackets */}
   846→      <AnimatePresence>
   847→        {spotlight ? <MatchSpotlight key={spotlight.id} data={spotlight} onDone={() => setSpotlight(null)} /> : null}
   848→      </AnimatePresence>
   849→
   850→      {/* animación de reserva entrando al bracket desde el banco */}
   851→      <AnimatePresence>
   852→        {subEnter ? <SubEnterOverlay key={subEnter.id} data={subEnter} onDone={() => setSubEnter(null)} /> : null}
   853→      </AnimatePresence>
   854→
   855→      {/* banner de espera durante mixing */}
   856→      <AnimatePresence>
   857→        {mixDone && status === "mixing" && mode === "brackets" ? (
   858→          <motion.div
   859→            initial={{ y: 40, opacity: 0 }}
   860→            animate={{ y: 0, opacity: 1 }}
   861→            exit={{ y: 40, opacity: 0 }}
   862→            className="relative z-20 shrink-0 border-t border-[#ffb830]/30 bg-[#171307]/90 flex items-center justify-center gap-2 py-2"
   863→          >
   864→            <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   865→            <span className="text-[10px] font-extrabold tracking-[0.28em] uppercase text-[#ffb830]">
   866→              Brackets listos — el administrador iniciará el torneo
   867→            </span>
   868→            <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   869→          </motion.div>
   870→        ) : null}
   871→      </AnimatePresence>
   872→
   873→      {/* pie: rail de próximos matches / ticker / barra vacía */}
   874→      {railActive && bracket ? (
   875→        <MatchRailFallback bracket={bracket} schedule={schedule} tournament={tournament} tickerItems={tickerItems} />
   876→      ) : status === "live" || status === "finished" ? (
   877→        <Ticker items={tickerItems} />
   878→      ) : (
   879→        <div className="relative shrink-0 h-10 border-t border-white/10 bg-black/70" />
   880→      )}
   881→    </div>
   882→  );
   883→}

   885→/* Si no hay matches programados con participantes definidos, cae al ticker */
   886→function MatchRailFallback({
   887→  bracket,
   888→  schedule,
   889→  tournament,
   890→  tickerItems,
   891→}: {
   892→  bracket: Bracket;
   893→  schedule: Map<string, number>;
   894→  tournament: Tournament;
   895→  tickerItems: string[];
   896→}) {
   897→  const hasAny = useMemo(() => hasRailMatches(bracket, schedule), [bracket, schedule]);
   898→
   899→  if (hasAny) return <MatchRail bracket={bracket} schedule={schedule} tournament={tournament} />;
   900→  return <Ticker items={tickerItems} />;
   901→}
   902→
   903→function fmtTimeSafe(ts: number): string {
   904→  return new Date(ts).toLocaleTimeString("es-MX", { hour: "2-digit", minute: "2-digit", hour12: true });
   905→}
   906→
