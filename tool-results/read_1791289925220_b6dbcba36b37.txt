    49→    49→/* Detecta el primer evento "dramático" entre dos versiones del bracket:
    50→    50→   match que pasa a DONE (ganador marcado o cambiado) o a LIVE (en juego).
    51→    51→   Las PCs del match se leen del match actual (al poner en juego ya tienen
    52→    52→   PCs asignadas) o del snapshot anterior (al definir ganador ya se liberaron). */
    53→    53→function findSpotlightEvent(
    54→    54→  cur: Bracket,
    55→    55→  prev: Bracket,
    56→    56→  winsNeeded: number,
    57→    57→  modality: Modality
    58→    58→): SpotlightData | null {
    59→    59→  for (let r = 0; r < cur.rounds.length; r++) {
    60→    60→    const round = cur.rounds[r];
    61→    61→    for (let m = 0; m < round.length; m++) {
    62→    62→      const match = round[m];
    63→    63→      const old = prev.rounds[r]?.[m];
    64→    64→      if (!match || !old || match.bye) continue;
    65→    65→      const becameDone = match.status === "done" && old.status !== "done";
    66→    66→      const winnerChanged = match.status === "done" && old.status === "done" && match.w !== old.w;
    67→    67→      const becameLive = match.status === "live" && old.status !== "live";
    68→    68→      if (becameDone || winnerChanged || becameLive) {
    69→    69→        const pcs = match.pcs?.length ? match.pcs : old.pcs;
    70→    70→        return {
    71→    71→          id: `${match.id}-${match.status}-${match.w}-${Date.now()}`,
    72→    72→          kind: match.status === "live" ? "live" : "win",
    73→    73→          tag: matchTag(cur, r, m),
    74→    74→          players: match.slots.map((s, i) => ({
    75→    75→            name: s.label || "—",
    76→    76→            score: s.score ?? 0,
    77→    77→            pcs: pcsForSide(modality, pcs, i),
    78→    78→          })),
    79→    79→          winner: match.w,
    80→    80→          winsNeeded,
    81→    81→        };
    82→    82→      }
    83→    83→    }
    84→    84→  }
    85→    85→  // verificar también el match de 3er lugar
    86→    86→  if (cur.thirdPlace && prev.thirdPlace) {
    87→    87→    const tp = cur.thirdPlace;
    88→    88→    const oldTp = prev.thirdPlace;
    89→    89→    const becameDone = tp.status === "done" && oldTp.status !== "done";
    90→    90→    const becameLive = tp.status === "live" && oldTp.status !== "live";
    91→    91→    if (becameDone || becameLive) {
    92→    92→      const pcs = tp.pcs?.length ? tp.pcs : oldTp.pcs;
    93→    93→      return {
    94→    94→        id: `${tp.id}-${tp.status}-${tp.w}-${Date.now()}`,
    95→    95→        kind: tp.status === "live" ? "live" : "win",
    96→    96→        tag: "3ER LUGAR",
    97→    97→        players: tp.slots.map((s, i) => ({
    98→    98→          name: s.label || "—",
    99→    99→          score: s.score ?? 0,
   100→   100→          pcs: pcsForSide(modality, pcs, i),
   101→   101→        })),
   102→   102→        winner: tp.w,
   103→   103→        winsNeeded,
   104→   104→      };
   105→   105→    }
   106→   106→  }
   107→   107→  return null;
   108→   108→}

   342→   342→export function SpectatorView({ tid }: { tid: string | null }) {
   343→   343→  const { data: tournaments, loading, error } = useTournaments();
   344→   344→  /* El visor siempre es la misma URL (/?v=show). Si no viene ?t=, se
   345→   345→     resuelve el torneo ACTIVO desde Firestore (arena/active). Solo puede
   346→   346→     haber uno activo a la vez — lo controla el admin desde "Torneo en vivo". */
   347→   347→  const active = useActiveTournament();
   348→   348→  const effectiveTid = tid ?? active.tid;
   349→   349→  const tournament: Tournament | null = useMemo(
   350→   350→    () => tournaments.find((t) => t.id === effectiveTid) ?? null,
   351→   351→    [tournaments, effectiveTid]
   352→   352→  );
   353→   353→  const { data: players } = usePlayers(tournament ? effectiveTid : null);
   354→   354→  const { data: bracket } = useBracket(tournament ? effectiveTid : null);
   355→   355→  const arenaState = useArenaState(tournament ? effectiveTid : null);
   356→   356→
   357→   357→  const status = tournament?.status ?? "open";
   358→   358→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   359→   359→  const [revealCount, setRevealCount] = useState(0);
   360→   360→  const [spotlight, setSpotlight] = useState<SpotlightData | null>(null);
   361→   361→  const [subEnter, setSubEnter] = useState<SubEnterData | null>(null);
   362→   362→  const prevBracketRef = useRef<{ tid: string | null; bracket: Bracket } | null>(null);
   363→   363→  const prevBankPickRef = useRef<{ id: number; slotPid: string | null } | null>(null);
   364→   364→  const rootRef = useRef<HTMLDivElement | null>(null);
   365→   365→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   366→   366→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   367→   367→
   368→   368→  /* cartelera de horarios: la controla el admin desde "Torneo en vivo".
   369→   369→     El visor solo refleja el estado (pantalla proyectada, sin interacción). */
   370→   370→  const schedOpen = arenaState?.schedOpen ?? false;
   371→   371→  /* reproductor de sonidos (carga la config global + reproduce) */
   372→   372→  const playSound = useSoundPlayer();
   373→   373→
   374→   374→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   375→   375→     se refleja sola en el visor). El toggle manual gana hasta que el status
   376→   376→     vuelva a cambiar, porque la anulación queda atada al status vigente.
   377→   377→     CUENTA REGRESIVA: en cuanto el admin programa el inicio (startAt), el
   378→   378→     visor salta solo a la pestaña de brackets, donde el cronómetro de inicio
   379→   379→     se ve en grande; si cancela la cuenta regresiva regresa al modo registro. */
   380→   380→  const inRegisterPhase = status === "open" || status === "closed";
   381→   381→  const autoMode: ViewerMode = inRegisterPhase ? (tournament?.startAt ? "brackets" : "reg") : "brackets";
   382→   382→  const baseMode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   383→   383→  /* El admin abrió el selector de banco → el visor pasa a la vista de Banco
   384→   384→     para que el caster vea las reservas. Es prioritario y transitorio:
   385→   385→     en cuanto se confirma/cancela (bankPick = null) regresa al modo base. */
   386→   386→  const arenaBankPick = arenaState?.bankPick ?? null;
   387→   387→  const forceBank =
   388→   388→    !!arenaBankPick && (status === "live" || status === "mixing" || status === "stopped" || status === "finished");
   389→   389→  const mode: ViewerMode = forceBank ? "bank" : baseMode;
   390→   390→
   391→   391→  /* sincroniza el modo en la URL para compartir la vista correcta */

   906→   906→    </div>
   907→   907→  );

   911→   911→function MatchRailFallback({
   912→   912→  bracket,
   913→   913→  schedule,
   914→   914→  tournament,
   915→   915→  tickerItems,
   916→   916→}: {
   917→   917→  bracket: Bracket;
   918→   918→  schedule: Map<string, number>;
   919→   919→  tournament: Tournament;
   920→   920→  tickerItems: string[];

   926→   926→}
