    49→/* Detecta el primer evento "dramático" entre dos versiones del bracket:
    50→   match que pasa a DONE (ganador marcado o cambiado) o a LIVE (en juego).
    51→   Las PCs del match se leen del match actual (al poner en juego ya tienen
    52→   PCs asignadas) o del snapshot anterior (al definir ganador ya se liberaron). */
    53→function findSpotlightEvent(
    54→  cur: Bracket,
    55→  prev: Bracket,
    56→  winsNeeded: number,
    57→  modality: Modality
    58→): SpotlightData | null {
    59→  for (let r = 0; r < cur.rounds.length; r++) {
    60→    const round = cur.rounds[r];
    61→    for (let m = 0; m < round.length; m++) {
    62→      const match = round[m];
    63→      const old = prev.rounds[r]?.[m];
    64→      if (!match || !old || match.bye) continue;
    65→      const becameDone = match.status === "done" && old.status !== "done";
    66→      const winnerChanged = match.status === "done" && old.status === "done" && match.w !== old.w;
    67→      const becameLive = match.status === "live" && old.status !== "live";
    68→      if (becameDone || winnerChanged || becameLive) {
    69→        const pcs = match.pcs?.length ? match.pcs : old.pcs;
    70→        return {
    71→          id: `${match.id}-${match.status}-${match.w}-${Date.now()}`,
    72→          kind: match.status === "live" ? "live" : "win",
    73→          tag: matchTag(cur, r, m),
    74→          players: match.slots.map((s, i) => ({
    75→            name: s.label || "—",
    76→            score: s.score ?? 0,
    77→            pcs: pcsForSide(modality, pcs, i),
    78→          })),
    79→          winner: match.w,
    80→          winsNeeded,
    81→        };
    82→      }
    83→    }
    84→  }
    85→  // verificar también el match de 3er lugar
    86→  if (cur.thirdPlace && prev.thirdPlace) {
    87→    const tp = cur.thirdPlace;
    88→    const oldTp = prev.thirdPlace;
    89→    const becameDone = tp.status === "done" && oldTp.status !== "done";
    90→    const becameLive = tp.status === "live" && oldTp.status !== "live";
    91→    if (becameDone || becameLive) {
    92→      const pcs = tp.pcs?.length ? tp.pcs : oldTp.pcs;
    93→      return {
    94→        id: `${tp.id}-${tp.status}-${tp.w}-${Date.now()}`,
    95→        kind: tp.status === "live" ? "live" : "win",
    96→        tag: "3ER LUGAR",
    97→        players: tp.slots.map((s, i) => ({
    98→          name: s.label || "—",
    99→          score: s.score ?? 0,
   100→          pcs: pcsForSide(modality, pcs, i),
   101→        })),
   102→        winner: tp.w,
   103→        winsNeeded,
   104→      };
   105→    }
   106→  }
   107→  return null;
   108→}

   125→function useClock() {
   126→  const [now, setNow] = useState<Date | null>(null);
   127→  useEffect(() => {
   128→    const first = requestAnimationFrame(() => setNow(new Date()));
   129→    const t = setInterval(() => setNow(new Date()), 1000);
   130→    return () => {
   131→      cancelAnimationFrame(first);
   132→      clearInterval(t);
   133→    };
   134→  }, []);
   135→  return now;
   136→}

   341→/* ---------- Cuerpo principal ---------- */
   342→export function SpectatorView({ tid }: { tid: string | null }) {
   343→  const { data: tournaments, loading, error } = useTournaments();
   344→  /* El visor siempre es la misma URL (/?v=show). Si no viene ?t=, se
   345→     resuelve el torneo ACTIVO desde Firestore (arena/active). Solo puede
   346→     haber uno activo a la vez — lo controla el admin desde "Torneo en vivo". */
   347→  const active = useActiveTournament();
   348→  const effectiveTid = tid ?? active.tid;
   349→  const tournament: Tournament | null = useMemo(
   350→    () => tournaments.find((t) => t.id === effectiveTid) ?? null,
   351→    [tournaments, effectiveTid]
   352→  );
   353→  const { data: players } = usePlayers(tournament ? effectiveTid : null);
   354→  const { data: bracket } = useBracket(tournament ? effectiveTid : null);
   355→  const arenaState = useArenaState(tournament ? effectiveTid : null);
   356→
   357→  const status = tournament?.status ?? "open";
   358→  const [override, setOverride] = useState<{ status: Tournament["status"]; mode: ViewerMode } | null>(null);
   359→  const [revealCount, setRevealCount] = useState(0);
   360→  const [spotlight, setSpotlight] = useState<SpotlightData | null>(null);
   361→  const [subEnter, setSubEnter] = useState<SubEnterData | null>(null);
   362→  const prevBracketRef = useRef<{ tid: string | null; bracket: Bracket } | null>(null);
   363→  const prevBankPickRef = useRef<{ id: number; slotPid: string | null } | null>(null);
   364→  const rootRef = useRef<HTMLDivElement | null>(null);
   365→  const ivRef = useRef<ReturnType<typeof setInterval> | null>(null);
   366→  const toRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   367→
   368→  /* cartelera de horarios: la controla el admin desde "Torneo en vivo".
   369→     El visor solo refleja el estado (pantalla proyectada, sin interacción). */
   370→  const schedOpen = arenaState?.schedOpen ?? false;
   371→  /* reproductor de sonidos (carga la config global + reproduce) */
   372→  const playSound = useSoundPlayer();
   373→
   374→  /* AUTO-MODO: el modo se deriva del status del torneo (cada fase del admin
   375→     se refleja sola en el visor). El toggle manual gana hasta que el status
   376→     vuelva a cambiar, porque la anulación queda atada al status vigente.
   377→     CUENTA REGRESIVA: en cuanto el admin programa el inicio (startAt), el
   378→     visor salta solo a la pestaña de brackets, donde el cronómetro de inicio
   379→     se ve en grande; si cancela la cuenta regresiva regresa al modo registro. */
   380→  const inRegisterPhase = status === "open" || status === "closed";
   381→  const autoMode: ViewerMode = inRegisterPhase ? (tournament?.startAt ? "brackets" : "reg") : "brackets";
   382→  const baseMode: ViewerMode = override && override.status === status ? override.mode : autoMode;
   383→  /* El admin abrió el selector de banco → el visor pasa a la vista de Banco
   384→     para que el caster vea las reservas. Es prioritario y transitorio:
   385→     en cuanto se confirma/cancela (bankPick = null) regresa al modo base. */
   386→  const arenaBankPick = arenaState?.bankPick ?? null;
   387→  const forceBank =
   388→    !!arenaBankPick && (status === "live" || status === "mixing" || status === "stopped" || status === "finished");
   389→  const mode: ViewerMode = forceBank ? "bank" : baseMode;
   390→
   391→  /* sincroniza el modo en la URL para compartir la vista correcta */
   392→  useEffect(() => {
   393→    try {
   394→      const url = new URL(window.location.href);
   395→      if (mode === "reg") url.searchParams.set("mode", "reg");
   396→      else url.searchParams.delete("mode");
   397→      if (url.toString() !== window.location.href) window.history.replaceState(null, "", url.toString());
   398→    } catch {
   399→      /* ignore */
   400→    }
   401→  }, [mode]);
   402→
   403→  const changeMode = (m: ViewerMode) => {
   404→    setOverride({ status, mode: m });
   405→  };
   406→
   407→  // animación de mix
   408→  const totalSlots = bracket ? bracket.rounds[0].length * (bracket.s ?? 2) : 0;
   409→  const pool = useMemo(() => {
   410→    if (!bracket) return [];
   411→    const labels = bracket.rounds[0].flatMap((m) => m.slots.map((s) => s.label || "TBD"));
   412→    return Array.from(new Set(labels));
   413→  }, [bracket]);
   414→
   415→  useEffect(() => {
   416→    if (status !== "mixing" || !bracket || mode !== "brackets") return;
   417→    const r0 = requestAnimationFrame(() => {
   418→      setRevealCount(0);
   419→    });
   420→    // el sonido de inicio del mix (mixLaunch) ya lo dispara el admin
   421→    // al lanzar el mix — no duplicar aquí
   422→    toRef.current = setTimeout(() => {
   423→      ivRef.current = setInterval(() => {
   424→        setRevealCount((c) => {
   425→          if (c >= totalSlots) {
   426→            if (ivRef.current) clearInterval(ivRef.current);
   427→            return c;
   428→          }
   429→          // el sonido de cada tick (mixPlace) lo dispara el effect
   430→          // que vigila revealCount — no duplicar aquí
   431→          return c + 1;
   432→        });
   433→      }, 240);
   434→    }, 2400);
   435→    return () => {
   436→      cancelAnimationFrame(r0);
   437→      if (toRef.current) clearTimeout(toRef.current);
   438→      if (ivRef.current) clearInterval(ivRef.current);
   439→    };
   440→  }, [status, bracket, totalSlots, mode]);
   441→
   442→  /* ----- PODIO — controlado por el admin (arenaState.showPodium).
   443→     DOBLE SEGURIDAD anti-residual: la señal del admin se ignora si el
   444→     bracket actual NO tiene campeón real (p. ej. señal vieja en la base
   445→     de datos tras reiniciar el torneo, o al actualizar el visor).
   446→     Sin campeón → jamás se dibuja el podio, aunque el flag venga true. ----- */
   447→  const podiumFlag = arenaState?.showPodium ?? false;
   448→  const hasChampion = useMemo(() => {
   449→    if (!bracket) return false;
   450→    return !!getPodium(bracket).champion?.pid;
   451→  }, [bracket]);
   452→  const showPodium = podiumFlag && hasChampion;
   453→
   454→  /* sonido de celebración: solo en la transición oculto → visible */
   455→  const prevPodiumFlag = useRef(false);
   456→  useEffect(() => {
   457→    if (podiumFlag && !prevPodiumFlag.current && hasChampion) playSound("tournamentFinish");
   458→    prevPodiumFlag.current = podiumFlag;
   459→  }, [podiumFlag, hasChampion, playSound]);
   460→
   461→  /* ----- SPOTLIGHT: ganador marcado / match en juego → corte dramático.
   462→     Compara cada snapshot del bracket con el anterior; al detectar un match
   463→     que pasa a DONE (o cambia de ganador) o a LIVE, lanza el overlay. ----- */
   464→  useEffect(() => {
   465→    const prev = prevBracketRef.current;

   879→

   881→      <AnimatePresence>
   882→        {mixDone && status === "mixing" && mode === "brackets" ? (
   883→          <motion.div
   884→            initial={{ y: 40, opacity: 0 }}
   885→            animate={{ y: 0, opacity: 1 }}
   886→            exit={{ y: 40, opacity: 0 }}

   888→          >
   889→            <span className="w-2 h-2 bg-[#ffb830] blink" aria-hidden />
   890→            <span className="text-[10px] font-extrabold tracking-[0.28em] uppercase text-[#ffb830]">

   903→      ) : (
   904→        <div className="relative shrink-0 h-10 border-t border-white/10 bg-black/70" />
   905→      )}
   906→    </div>
   907→  );
   908→}

   910→/* Si no hay matches programados con participantes definidos, cae al ticker */
   911→function MatchRailFallback({
   912→  bracket,
   913→  schedule,
   914→  tournament,
   915→  tickerItems,
   916→}: {
   917→  bracket: Bracket;
   918→  schedule: Map<string, number>;
   919→  tournament: Tournament;
   920→  tickerItems: string[];
   921→}) {
   922→  const hasAny = useMemo(() => hasRailMatches(bracket, schedule), [bracket, schedule]);
   923→
   924→  if (hasAny) return <MatchRail bracket={bracket} schedule={schedule} tournament={tournament} />;
   925→  return <Ticker items={tickerItems} />;
   926→}

   928→function fmtTimeSafe(ts: number): string {
   929→  return new Date(ts).toLocaleTimeString("es-MX", { hour: "2-digit", minute: "2-digit", hour12: true });
   930→}
   931→
