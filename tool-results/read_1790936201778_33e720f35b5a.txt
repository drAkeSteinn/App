    74→/* ============================================================
    75→   Vista TORNEO EN VIVO — administración del bracket
    76→   · Dos vistas: Bracket (árbol) y Rondas (columnas de cards)

    81→
    82→type AdminView = "tree" | "rounds";
    83→
    84→export function LiveAdminView({
    85→  tournament,
    86→  players,
    87→  bracket,
    88→  loadingBracket,
    89→}: {
    90→  tournament: Tournament | null;
    91→  players: Player[];
    92→  bracket: Bracket | null;
    93→  loadingBracket: boolean;
    94→}) {
    95→  const [sel, setSel] = useState<{ r: number; m: number } | null>(null);
    96→  const [pickBank, setPickBank] = useState<{ r: number; m: number; slotIdx: number } | null>(null);
    97→  const [confirm, setConfirm] = useState<null | { kind: "remix" | "finish" | "wipe" | "reopen" | "results" | "fill" | "demos" }>(null);
    98→  const [busy, setBusy] = useState(false);
    99→  const [view, setView] = useState<AdminView>("tree");
   100→  const [panelOpen, setPanelOpen] = useState(true);
   101→  const [zoom, setZoom] = useState<number | null>(null); // null = auto-fit
   102→  const [startMins, setStartMins] = useState(1);
   103→
   104→  const bank = useMemo(() => players.filter((p) => p.seat === "bank"), [players]);
   105→  const participants = useMemo(
   106→    () => (tournament ? groupParticipants(players, tournament.modality) : []),
   107→    [players, tournament]
   108→  );
   109→  const officialsCount = useMemo(() => players.filter((p) => p.seat === "off").length, [players]);
   110→  const demoCount = useMemo(() => players.filter((p) => p.demo).length, [players]);
   111→  const missingPlayers = tournament ? playersCapacity(tournament) - officialsCount : 0;
   112→  const schedule = useMemo(
   113→    () => (tournament && bracket ? computeSchedule(bracket, tournament.startAt, tournament.matchMins) : new Map()),
   114→    [bracket, tournament]
   115→  );
   116→  /* capacidad del escenario: matches simultáneos según las 4 PCs */
   117→  const maxSim = tournament ? maxSimultaneous(tournament.modality) : 1;
   118→  const liveCount = useMemo(
   119→    () => (bracket ? bracket.rounds.reduce((acc, round) => acc + round.filter((x) => x.status === "live").length, 0) : 0),
   120→    [bracket]
   121→  );
   122→
   123→  /* navegación secuencial de matches (para editar rápido sin recorrer todo) */
   124→  const flatMatches = useMemo(() => {
   125→    if (!bracket) return [];
   126→    const out: { r: number; m: number }[] = [];
   127→    bracket.rounds.forEach((round, r) => round.forEach((_, m) => out.push({ r, m })));
   128→    return out;
   129→  }, [bracket]);
   130→  const flatIdx = useMemo(() => {
   131→    if (!sel || !bracket) return -1;
   132→    return flatMatches.findIndex((x) => x.r === sel.r && x.m === sel.m);
   133→  }, [flatMatches, sel, bracket]);
   134→
   135→  if (!tournament) {
   136→    return (
   137→      <div className="panel clip-card">
   138→        <EmptyState
   139→          icon={<Radio size={26} />}
   140→          title="Selecciona un torneo"
   141→          message="Elige un torneo en el selector superior para administrar el bracket en vivo: ganadores, descalificaciones y reemplazos del banco."
   142→        />
   143→      </div>
   144→    );
   145→  }
   146→
   147→  const guard = async (fn: () => Promise<void>) => {
   148→    setBusy(true);
   149→    try {
   150→      await fn();
   151→    } catch (e) {
   152→      toast.error(e instanceof Error ? e.message : "Error de operación");
   153→    } finally {
   154→      setBusy(false);
   155→    }
   156→  };
   157→
   158→  const selMatch = sel && bracket ? bracket.rounds[sel.r]?.[sel.m] : null;
   159→
   160→  const stepMatch = (dir: 1 | -1) => {
   161→    const nx = flatMatches[flatIdx + dir];
   162→    if (nx) setSel(nx);
   163→  };
   164→
   165→  const zoomStep = (dir: 1 | -1) => {
   166→    setZoom((z) => {
   167→      const base = z ?? 0.8;
   168→      return Math.min(1.8, Math.max(0.3, Math.round((base + dir * 0.1) * 100) / 100));
   169→    });
   170→  };
   171→
   172→  return (
   173→    <div className="flex flex-col gap-3 min-h-0">
   174→      {/* ============ barra de info del torneo ============ */}
   175→      <div className="panel clip-card p-4 flex flex-wrap items-center gap-3">
   176→        <div className="flex items-center gap-3 min-w-0 mr-auto">
   177→          <span className="slashes w-8 h-4 inline-block shrink-0" aria-hidden />
   178→          <div className="min-w-0">
   179→            <div className="font-display italic text-[17px] uppercase truncate">{tournament.name}</div>
   180→            <div className="flex items-center gap-2 mt-1 flex-wrap">
   181→              <StatusChip status={tournament.status} />
   182→              <ModalityChip modality={tournament.modality} />
   183→              {bracket ? (
   184→                <span
   185→                  className={`chip clip-tag border ${
   186→                    liveCount >= maxSim
   187→                      ? "bg-[#e8102e]/15 text-[#ff8095] border-[#e8102e]/40"
   188→                      : "bg-[#e8102e]/12 text-[#ff8095] border-[#e8102e]/35"
   189→                  }`}
   190→                  title="Matches en juego / máximo simultáneo según las PCs del escenario (4 PCs)"
   191→                >
   192→                  <Monitor size={10} />
   193→                  EN JUEGO {liveCount}/{maxSim}
   194→                </span>
   195→              ) : null}
   196→              {bracket ? (
   197→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   198→                  {doneMatches(bracket)}/{totalMatches(bracket)} MATCHES
   199→                </span>
   200→              ) : (
   201→                <span className="chip clip-tag bg-white/[0.06] text-[#c9cbd3] border border-white/10">
   202→                  {officialsCount}/{playersCapacity(tournament)} JUGADORES
   203→                </span>
   204→              )}
   205→              {tournament.startAt ? (
   206→                <span className="chip clip-tag bg-[#ffb830]/12 text-[#ffb830] border border-[#ffb830]/35">
   207→                  <TimerReset size={10} />
   208→                  INICIA ≈ {fmtTime(tournament.startAt)}
   209→                </span>
   210→              ) : null}
   211→            </div>
   212→          </div>
   213→        </div>
   214→        <span className="text-[9px] font-extrabold uppercase tracking-[0.18em] text-[#6b6e78] hidden md:inline">
   215→          Cada fase se refleja automáticamente en el visor
   216→        </span>
   217→      </div>
   218→
   219→      {/* ============ FLUJO POR FASES ============ */}
   220→      <PhaseBar
   221→        tournament={tournament}
   222→        bracket={bracket}
   223→        participantsCount={participants.length}
   224→        officialsCount={officialsCount}
   225→        missingPlayers={missingPlayers}
   226→        demoCount={demoCount}
   227→        bankCount={bank.length}
   228→        startMins={startMins}
   229→        setStartMins={setStartMins}
   230→        busy={busy}
   231→        guard={guard}
   232→        onConfirm={(kind) => setConfirm({ kind })}
   233→        onLaunchMix={() =>
   234→          guard(async () => {
   235→            await launchMix(tournament.id, players, tournament);
   236→            setSel(null);
   237→            toast.success("Mix match lanzado — el visor reproduce la animación");
   238→          })
   239→        }
   240→      />
   241→
   242→      {/* ============ cuerpo ============ */}
   243→      {loadingBracket ? (
   244→        <div className="panel clip-card p-14 flex items-center justify-center text-[#8e919c]">
   245→          <Loader2 size={20} className="animate-spin" />
   246→        </div>
   247→      ) : !bracket ? (
   248→        <div className="panel clip-card">
   249→          <EmptyState
   250→            icon={<Zap size={26} />}
   251→            title="Brackets no creados"
   252→            message={`Cuando estés listo, lanza el MIX MATCH: se randomizarán los ${participants.length} participantes y el visor de espectadores mostrará la animación de llenado de brackets.`}
   253→            action={
   254→              <div className="flex flex-wrap gap-2 justify-center">
   255→                <Btn
   256→                  variant="red"
   257→                  disabled={participants.length < 2}
   258→                  onClick={() =>
   259→                    guard(async () => {
   260→                      await launchMix(tournament.id, players, tournament);
   261→                      toast.success("Mix match lanzado");
   262→                    })
   263→                  }
   264→                >
   265→                  <Rocket size={13} />
   266→                  Lanzar mix match
   267→                </Btn>
   268→                <Btn
   269→                  variant="dark"
   270→                  onClick={() =>
   271→                    guard(async () => {
   272→                      await setStartAt(tournament.id, Date.now() + startMins * 60_000);
   273→                      toast.success(`Inicio configurado en ${startMins} min — el visor cambia a brackets con el cronómetro`);
   274→                    })
   275→                  }
   276→                >
   277→                  <TimerReset size={13} />
   278→                  Inicio en {startMins} min
   279→                </Btn>
   280→              </div>
   281→            }
   282→          />
   283→          {/* vista previa de participantes */}
   284→          {participants.length > 0 ? (
   285→            <div className="px-5 pb-5">
   286→              <div className="label-cap mb-2 flex items-center gap-2">
   287→                <Users size={11} />
   288→                Participantes que entrarán al sorteo ({participants.length})
   289→              </div>
   290→              <div className="flex flex-wrap gap-1.5 max-h-36 overflow-y-auto">
   291→                {participants.map((p, i) => (
   292→                  <span key={p.pid} className="chip clip-tag bg-white/[0.05] text-[#c9cbd3] border border-white/10">
   293→                    {i + 1}. {p.label}
   294→                  </span>
   295→                ))}
   296→              </div>
   297→            </div>
   298→          ) : null}
   299→        </div>
   300→      ) : (
   301→        <>
   302→          {/* ============ toolbar de vistas ============ */}
   303→          <div className="panel clip-card-sm px-3 py-2 flex flex-wrap items-center gap-2">
   304→            <div className="flex gap-1 p-0.5 bg-black/40 border border-white/10" role="radiogroup" aria-label="Vista del bracket">
   305→              <button
   306→                type="button"
   307→                role="radio"
   308→                aria-checked={view === "tree"}
   309→                onClick={() => setView("tree")}
   310→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   311→                  view === "tree" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   312→                }`}
   313→              >
   314→                <LayoutPanelLeft size={12} />
   315→                Bracket
   316→              </button>
   317→              <button
   318→                type="button"
   319→                role="radio"
   320→                aria-checked={view === "rounds"}
   321→                onClick={() => setView("rounds")}
   322→                className={`btn-press clip-tag px-3 py-1.5 text-[10px] font-extrabold uppercase tracking-[0.12em] flex items-center gap-1.5 transition-colors ${
   323→                  view === "rounds" ? "bg-[linear-gradient(160deg,#ff2440,#a30d24)] text-white" : "text-[#8e919c] hover:text-white"
   324→                }`}
   325→              >
   326→                <Columns3 size={12} />
   327→                Rondas
   328→              </button>
   329→            </div>
   330→
   331→            {selMatch ? (
   332→              <span className="chip clip-tag bg-[#e8102e]/15 text-[#ff8095] border border-[#e8102e]/40 text-[9px] flex items-center gap-1.5">
   333→                <Hand size={10} />
   334→                {matchTag(bracket, sel!.r, sel!.m)} seleccionado
   335→              </span>
   336→            ) : (
   337→              <span className="text-[9px] font-extrabold uppercase tracking-[0.16em] text-[#6b6e78] hidden sm:inline">
   338→                Clic en un match para administrarlo
   339→              </span>
   340→            )}
   341→
   342→            <div className="ml-auto flex items-center gap-1.5">
   343→              {/* navegación prev/next */}
   344→              <div className="flex items-center gap-1">
   345→                <IconBtn title="Match anterior" disabled={flatIdx <= 0} onClick={() => stepMatch(-1)}>
   346→                  <ChevronLeft size={14} />
   347→                </IconBtn>
   348→                <IconBtn title="Match siguiente" disabled={flatIdx < 0 || flatIdx >= flatMatches.length - 1} onClick={() => stepMatch(1)}>
   349→                  <ChevronRight size={14} />
   350→                </IconBtn>
   351→              </div>
   352→
   353→              {/* zoom (solo vista bracket) */}
   354→              {view === "tree" ? (
   355→                <div className="flex items-center gap-1 pl-2 border-l border-white/10">
   356→                  <IconBtn title="Alejar" onClick={() => zoomStep(-1)}>
   357→                    <ZoomOut size={14} />
   358→                  </IconBtn>
   359→                  <span className="text-[9px] font-extrabold tabular-nums text-[#8e919c] w-10 text-center uppercase">
   360→                    {zoom ? `${Math.round(zoom * 100)}%` : "AUTO"}
   361→                  </span>
   362→                  <IconBtn title="Acercar" onClick={() => zoomStep(1)}>
   363→                    <ZoomIn size={14} />
   364→                  </IconBtn>
   365→                  <IconBtn title="Ajustar a la pantalla" active={zoom === null} onClick={() => setZoom(null)}>
   366→                    <Maximize2 size={13} />
   367→                  </IconBtn>
   368→                </div>
   369→              ) : null}
   370→
   371→              {/* toggle panel lateral */}
   372→              <button
   373→                type="button"
   374→                onClick={() => setPanelOpen((v) => !v)}
   375→                title={panelOpen ? "Ocultar panel del match" : "Mostrar panel del match"}
   376→                aria-pressed={panelOpen}
   377→                className={`btn-press p-2 border transition-colors hidden xl:inline-flex ${
   378→                  panelOpen
   379→                    ? "border-[#e8102e]/60 bg-[#e8102e]/10 text-white"
   380→                    : "border-white/12 text-[#8e919c] hover:text-white"
   381→                }`}
   382→              >
   383→                {panelOpen ? <PanelRightClose size={14} /> : <PanelRightOpen size={14} />}
   384→              </button>
   385→            </div>
   386→          </div>
   387→
   388→          {/* ============ stage + panel lateral colapsable ============ */}
   389→          <div className="flex gap-3 items-stretch min-h-0">
   390→            {/* área principal */}
   391→            <div className="flex-1 min-w-0 min-h-0">
   392→              <div className="panel clip-card p-3 min-h-[420px] h-[calc(100vh-330px)] max-h-[80vh] flex flex-col">
   393→                <div className="flex items-center justify-between px-2 pb-2">
   394→                  <span className="label-cap">
   395→                    {view === "tree" ? "Bracket · clic en un match para administrar" : "Rondas · cards seleccionables con botón de ganador directo"}
   396→                  </span>
   397→                  <span className="label-cap">{MODALITY_LABEL[tournament.modality]} · a {tournament.winsNeeded} wins</span>
   398→                </div>
   399→
   400→                {/* monitor del escenario: asignación de las 4 PCs por match */}
   401→                <PcDock tournament={tournament} bracket={bracket} onSelectMatch={(r, m) => setSel({ r, m })} />
   402→
   403→                {view === "tree" ? (
   404→                  <div className="flex-1 min-h-0 border border-white/8 bg-black/30 clip-card-sm">
   405→                    <FitStage deps={[bracket, tournament.modality, zoom]} maxScale={0.95} minScale={0.4} zoom={zoom} className="w-full h-full">
   406→                      <BracketTree
   407→                        bracket={bracket}
   408→                        modality={tournament.modality}
   409→                        mode="admin"
   410→                        selectedId={selMatch?.id ?? null}
   411→                        onSelectMatch={(r, m) => setSel({ r, m })}
   412→                        schedule={schedule}
   413→                        showTimes
   414→                      />
   415→                    </FitStage>
   416→                  </div>
   417→                ) : (
   418→                  <RoundColumns
   419→                    bracket={bracket}
   420→                    modality={tournament.modality}
   421→                    times={schedule}
   422→                    selectedId={selMatch?.id ?? null}
   423→                    onSelectMatch={(r, m) => setSel({ r, m })}
   424→                    onPickWinner={(r, m, slotIdx) =>
   425→                      guard(async () => {
   426→                        await setWinner(tournament.id, r, m, slotIdx);
   427→                        setSel({ r, m });
   428→                        toast.success(`${bracket.rounds[r][m].slots[slotIdx].label} avanza`);
   429→                      })
   430→                    }
   431→                  />
   432→                )}
   433→              </div>
   434→            </div>
   435→
   436→            {/* panel lateral (desktop, colapsable) */}
   437→            <aside
   438→              className={`hidden xl:block shrink-0 overflow-hidden transition-[width] duration-300 ease-out ${
   439→                panelOpen ? "w-[376px]" : "w-0"
   440→              }`}
   441→              aria-hidden={!panelOpen}
   442→            >
   443→              <div className="w-[376px] h-full">
   444→                <MatchPanel
   445→                  tournament={tournament}
   446→                  bracket={bracket}
   447→                  sel={sel}
   448→                  setSel={setSel}
   449→                  selMatch={selMatch}
   450→                  schedule={schedule}
   451→                  bank={bank}
   452→                  onClose={() => setPanelOpen(false)}
   453→                  guard={guard}
   454→                  setPickBank={setPickBank}
   455→                  busy={busy}
   456→                  stepMatch={stepMatch}
   457→                />
   458→              </div>
   459→            </aside>
   460→          </div>
   461→
   462→          {/* panel del match (móvil/tablet: debajo) */}
   463→          <div className="xl:hidden">
   464→            <MatchPanel
   465→              tournament={tournament}
   466→              bracket={bracket}
   467→              sel={sel}
   468→              setSel={setSel}
   469→              selMatch={selMatch}
   470→              schedule={schedule}
   471→              bank={bank}
   472→              onClose={() => setSel(null)}
   473→              guard={guard}
   474→              setPickBank={setPickBank}
   475→              busy={busy}
   476→              stepMatch={stepMatch}
   477→            />
   478→          </div>
   479→
   480→          {/* pestaña flotante para reabrir el panel en desktop */}
   481→          {!panelOpen && selMatch ? (
   482→            <button
   483→              type="button"
   484→              onClick={() => setPanelOpen(true)}
   485→              className="btn-press fixed right-0 top-1/2 -translate-y-1/2 z-40 hidden xl:flex flex-col items-center gap-2 red-badge clip-badge-l px-2 py-4 text-[10px] font-extrabold uppercase tracking-[0.2em] shadow-[0_10px_30px_rgba(232,16,46,0.4)]"
   486→              style={{ writingMode: "vertical-rl" }}
   487→            >
   488→              <ChevronLeft size={12} />
   489→              Panel del match
   490→            </button>
   491→          ) : null}
   492→        </>
   493→      )}
   494→
   495→      {/* ============ modal banco ============ */}
   496→      <Modal open={!!pickBank} onClose={() => setPickBank(null)} title="Reemplazar desde el banco">
   497→        <p className="text-[11px] font-bold text-[#8e919c] uppercase tracking-wider mb-3">
   498→          Selecciona la reserva que entrará al match
   499→        </p>
   500→        <div className="space-y-2 max-h-72 overflow-y-auto">
   501→          {bank.length === 0 ? (
   502→            <p className="text-[11px] font-bold text-[#6b6e78] uppercase">No hay jugadores en el banco</p>
   503→          ) : (
   504→            bank.map((p) => (
   505→              <button
   506→                key={p.id}
   507→                type="button"
   508→                onClick={async () => {
   509→                  if (!pickBank || !tournament) return;
   510→                  await guard(async () => {
   511→                    await replaceFromBank(tournament.id, pickBank.r, pickBank.m, pickBank.slotIdx, p, tournament.modality);
   512→                    toast.success(`${p.nick} entra al match`);
   513→                    setPickBank(null);
   514→                  });
   515→                }}
   516→                className="w-full panel-2 clip-card-sm flex items-center gap-3 px-3 py-2.5 hover:border-[#ffb830]/50 text-left transition-colors"
   517→              >
   518→                <span className="w-10 h-7 clip-badge red-badge flex items-center justify-center">
   519→                  <SkipForward size={12} />
   520→                </span>
   521→                <span className="min-w-0 flex-1">
   522→                  <span className="block text-[13px] font-extrabold uppercase text-white truncate">{p.nick}</span>
   523→                  <span className="block text-[10px] font-semibold text-[#8e919c] truncate">{p.name}</span>
   524→                </span>
   525→              </button>
   526→            ))

  1045→function MiniCountdown({ target }: { target: number }) {
  1046→  const [now, setNow] = useState(() => Date.now());
  1047→  useEffect(() => {
  1048→    const t = setInterval(() => setNow(Date.now()), 500);
  1049→    return () => clearInterval(t);
  1050→  }, []);
  1051→  const total = Math.max(0, Math.floor((target - now) / 1000));
  1052→  const h = Math.floor(total / 3600);
  1053→  const m = Math.floor((total % 3600) / 60);
  1054→  const s = total % 60;
  1055→  const pad = (x: number) => String(x).padStart(2, "0");
  1056→  return (
  1057→    <span className="font-display italic text-[20px] leading-none text-silver-grad tabular-nums">
  1058→      {h > 0 ? `${h}:${pad(m)}:${pad(s)}` : `${pad(m)}:${pad(s)}`}
  1059→    </span>
  1060→  );
  1061→}
